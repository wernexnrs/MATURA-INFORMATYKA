Zadanie 62. Liczby ósemkowe

ZBIÓR ZADAŃDANEROZWIĄZANIA

62.1

PythonC++23Java 25
Treść zadania

Plik liczby1.txt zawiera 1000 liczb całkowitych dodatnich zapisanych ósemkowo (co najwyżej sześciocyfrowych), a plik liczby2.txt — 1000 liczb całkowitych dodatnich zapisanych dziesiętnie (co najwyżej sześciocyfrowych); w obu plikach każda liczba jest w osobnym wierszu. Rozwiązania zapisz w pliku wyniki.txt.

Znajdź w pliku liczby1.txt najmniejszą i największą liczbę. Podaj je w zapisie ósemkowym.

Python
liczby = []

with open("liczby1.txt") as file:
    for line in file:
        liczby.append(int(line, 8))

print("min:", oct(min(liczby))[2:], "max:", oct(max(liczby))[2:])
C++23
#include <algorithm>
#include <fstream>
#include <print>
#include <string>
#include <vector>

int main() {
    std::vector<int> liczby;

    std::ifstream plik("liczby1.txt");
    std::string wiersz;
    while (plik >> wiersz) {
        liczby.push_back(std::stoi(wiersz, nullptr, 8));  // zapis ósemkowy
    }

    // {:o} - zapis ósemkowy, jak oct(...)[2:] w Pythonie
    std::println("min: {:o} max: {:o}", std::ranges::min(liczby), std::ranges::max(liczby));
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Zadanie {
    public static void main(String[] args) throws Exception {
        List<Integer> liczby = new ArrayList<>();

        for (String wiersz : Files.readAllLines(Path.of("liczby1.txt"))) {
            liczby.add(Integer.parseInt(wiersz.trim(), 8));  // zapis ósemkowy
        }

        System.out.println("min: " + Integer.toOctalString(Collections.min(liczby))
                + " max: " + Integer.toOctalString(Collections.max(liczby)));
    }
}

Odpowiedź: najmniejsza: 1002, największa: 777044

Treść zadania

Dane jak w zadaniu 62.1 (pliki liczby1.txt i liczby2.txt).

Znajdź najdłuższy niemalejący ciąg liczb stojących w kolejnych wierszach pliku liczby2.txt. Podaj jego pierwszy element i liczbę elementów (można założyć, że taki ciąg jest jeden).

Przykład: dla danych 23156, 1231, 1345, 1456, 1456, 897 najdłuższy ciąg zaczyna się od 1231 i ma 4 elementy.

Python
liczby = []

with open("liczby2.txt") as file:
    for line in file:
        liczby.append(int(line))

dlugosc = 1
poczatek = 0
najdluzszy = 1
najlepszy_poczatek = 0

for i in range(1, len(liczby)):
    if liczby[i] >= liczby[i - 1]:
        dlugosc += 1
    else:
        dlugosc = 1
        poczatek = i
    if dlugosc > najdluzszy:
        najdluzszy = dlugosc
        najlepszy_poczatek = poczatek

print(liczby[najlepszy_poczatek], najdluzszy)
C++23
#include <fstream>
#include <print>
#include <vector>

int main() {
    std::vector<int> liczby;

    std::ifstream plik("liczby2.txt");
    int x;
    while (plik >> x) {
        liczby.push_back(x);
    }

    int dlugosc = 1;
    int poczatek = 0;
    int najdluzszy = 1;
    int najlepszy_poczatek = 0;

    for (int i = 1; i < (int)liczby.size(); i++) {
        if (liczby[i] >= liczby[i - 1]) {
            dlugosc += 1;
        } else {
            dlugosc = 1;
            poczatek = i;
        }
        if (dlugosc > najdluzszy) {
            najdluzszy = dlugosc;
            najlepszy_poczatek = poczatek;
        }
    }

    std::println("{} {}", liczby[najlepszy_poczatek], najdluzszy);
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    public static void main(String[] args) throws Exception {
        List<Integer> liczby = new ArrayList<>();

        for (String wiersz : Files.readAllLines(Path.of("liczby2.txt"))) {
            liczby.add(Integer.parseInt(wiersz.trim()));
        }

        int dlugosc = 1;
        int poczatek = 0;
        int najdluzszy = 1;
        int najlepszyPoczatek = 0;

        for (int i = 1; i < liczby.size(); i++) {

                dlugosc += 1;
            } else {
                dlugosc = 1;
                poczatek = i;
            }
            if (dlugosc > najdluzszy) {
                najdluzszy = dlugosc;
                najlepszyPoczatek = poczatek;
            }
        }

        System.out.println(liczby.get(najlepszyPoczatek) + " " + najdluzszy);
    }
}

62.3

PythonC++23Java 25
Treść zadania

Dane jak w zadaniu 62.1.

Porównaj wartości liczb z wierszy o tych samych numerach w plikach liczby1.txt (ósemkowo) i liczby2.txt (dziesiętnie). Podaj, w ilu wierszach:

  • a) obie liczby mają tę samą wartość,
  • b) liczba z liczby1.txt jest większa od liczby z liczby2.txt.

Przykład: dla par (11456₈, 1302), (22666₈, 9654), (546₈, 499) odpowiedź a) to 1 (22666₈ = 9654), a b) to 1 (11456₈ > 1302).

Python
rowne = 0
wieksze = 0

with open("liczby1.txt") as file1:
    with open("liczby2.txt") as file2:
        for x, y in zip(file1, file2):
            x = int(x, 8)  # liczby1.txt - zapis ósemkowy
            y = int(y)  # liczby2.txt - zapis dziesiętny
            if x == y:
                rowne += 1
            elif x > y:
                wieksze += 1

print("a)", rowne)
print("b)", wieksze)
C++23
#include <fstream>
#include <print>
#include <string>

int main() {
    int rowne = 0;
    int wieksze = 0;

    std::ifstream plik1("liczby1.txt");
    std::ifstream plik2("liczby2.txt");
    std::string napis_x, napis_y;
    // czytamy parami, dopóki oba pliki mają dane (jak zip w Pythonie)
    while (plik1 >> napis_x && plik2 >> napis_y) {
        int x = std::stoi(napis_x, nullptr, 8);  // liczby1.txt - zapis ósemkowy
        int y = std::stoi(napis_y);              // liczby2.txt - zapis dziesiętny
        if (x == y) {
            rowne += 1;
        } else if (x > y) {
            wieksze += 1;
        }
    }

    std::println("a) {}", rowne);
    std::println("b) {}", wieksze);
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;

public class Zadanie {
    public static void main(String[] args) throws Exception {
        int rowne = 0;
        int wieksze = 0;

        List<String> plik1 = Files.readAllLines(Path.of("liczby1.txt"));
        List<String> plik2 = Files.readAllLines(Path.of("liczby2.txt"));
        // czytamy parami, dopóki oba pliki mają dane (jak zip w Pythonie)
        for (int i = 0; i < Math.min(plik1.size(), plik2.size()); i++) {
            int x = Integer.parseInt(plik1.get(i).trim(), 8);  // liczby1.txt - zapis ósemkowy
            int y = Integer.parseInt(plik2.get(i).trim());     // liczby2.txt - zapis dziesiętny
            if (x == y) {
                rowne += 1;
            } else if (x > y) {
                wieksze += 1;
            }
        }

        System.out.println("a) " + rowne);
        System.out.println("b) " + wieksze);
    }
}

Odpowiedź: a) 160, b) 357

W pliku wyniki.txt z oficjalnych rozwiązań CKE w podpunkcie a) jest 639, ale algorytm wzorcowy CKE (zadanie 3.cc, taki sam jak powyżej) daje na tych danych 160 – liczba 639 to omyłkowo przepisany wynik zadania 62.2.

62.4

PythonC++23Java 25
Treść zadania

Dane jak w zadaniu 62.1.

Podaj, ile razy cyfra 6 występuje w zapisie dziesiętnym wszystkich liczb z pliku liczby2.txt oraz ile razy wystąpiłaby, gdyby te same liczby zapisać w systemie ósemkowym.

Python
dziesietnie = 0
osemkowo = 0

with open("liczby2.txt") as file:
    for line in file:
        liczba = int(line)
        dziesietnie += str(liczba).count("6")
        osemkowo += oct(liczba)[2:].count("6")

print("dziesiętnie:", dziesietnie)
print("ósemkowo:", osemkowo)
C++23
#include <algorithm>
#include <format>
#include <fstream>
#include <print>
#include <string>

int main() {
    int dziesietnie = 0;
    int osemkowo = 0;

    std::ifstream plik("liczby2.txt");
    int liczba;
    while (plik >> liczba) {
        dziesietnie += std::ranges::count(std::to_string(liczba), '6');
        osemkowo += std::ranges::count(std::format("{:o}", liczba), '6');
    }

    std::println("dziesiętnie: {}", dziesietnie);
    std::println("ósemkowo: {}", osemkowo);
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;

public class Zadanie {
    static int ileSzostek(String s) {
        int wynik = 0;
        for (char c : s.toCharArray()) {
            if (c == '6') {
                wynik++;
            }
        }
        return wynik;
    }

    public static void main(String[] args) throws Exception {
        int dziesietnie = 0;
        int osemkowo = 0;

        for (String wiersz : Files.readAllLines(Path.of("liczby2.txt"))) {
            int liczba = Integer.parseInt(wiersz.trim());
            dziesietnie += ileSzostek(String.valueOf(liczba));
            osemkowo += ileSzostek(Integer.toOctalString(liczba));
        }

        System.out.println("dziesiętnie: " + dziesietnie);
        System.out.println("ósemkowo: " + osemkowo);
    }
}

Odpowiedź: w zapisie dziesiętnym: 411, w zapisie ósemkowym: 625

W pliku wyniki.txt CKE wartości są zamienione miejscami (dziesiętnie 625, ósemkowo 411) – program wzorcowy CKE (zadanie 4.cc) wypisuje najpierw wynik ósemkowy, a potem dziesiętny.
