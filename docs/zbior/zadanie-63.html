Treść zadania

Plik ciagi.txt zawiera 1000 różnych ciągów zerojedynkowych o długości od 2 do 18, każdy w osobnym wierszu. Odpowiedzi zapisz w pliku wyniki_ciagi.txt, poprzedzając wyniki każdego zadania jego numerem.

Ciąg dwucykliczny to ciąg zerojedynkowy o parzystej długości, który składa się z dwóch identycznych połówek (x = AB, gdzie A = B), np. 11001100. Wypisz wszystkie ciągi dwucykliczne z pliku.

Python
with open("ciagi.txt") as file:
    for line in file:
        ciag = line.strip()
        polowa = len(ciag) // 2
        if len(ciag) % 2 == 0 and ciag[:polowa] == ciag[polowa:]:
            print(ciag)
C++23
#include <fstream>
#include <print>

int main() {
    std::ifstream plik("ciagi.txt");
    std::string ciag;
    while (plik >> ciag) {  // wczytanie słowa pomija białe znaki - jak strip()
        size_t polowa = ciag.size() / 2;
        if (ciag.size() % 2 == 0 && ciag.substr(0, polowa) == ciag.substr(polowa)) {
            std::println("{}", ciag);
        }
    }
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;

public class Zadanie {
    public static void main(String[] args) throws Exception {
        for (String linia : Files.readAllLines(Path.of("ciagi.txt"))) {
            String ciag = linia.strip();
            int polowa = ciag.length() / 2;
            if (ciag.length() % 2 == 0 && ciag.substring(0, polowa).equals(ciag.substring(polowa))) {
                System.out.println(ciag);
            }
        }
    }
}

63.2

PythonC++23Java 25
Treść zadania

Dane jak w zadaniu 63.1 (1000 ciągów zerojedynkowych w pliku ciagi.txt).

Podaj, w ilu ciągach z pliku nigdzie nie stoją obok siebie dwie jedynki.

Przykład: spośród ciągów 11001101001, 10001000, 000011 warunek spełnia tylko 10001000.

Python
count = 0

with open("ciagi.txt") as file:
    for line in file:
        if "11" not in line:
            count += 1

print(count)
C++23
#include <fstream>
#include <print>
#include <string>

int main() {
    int count = 0;

    std::ifstream plik("ciagi.txt");
    std::string linia;
    while (std::getline(plik, linia)) {
        if (!linia.contains("11")) {
            count += 1;
        }
    }

    std::println("{}", count);
}

Dane jak w zadaniu 63.1.

Liczba półpierwsza to iloczyn dwóch liczb pierwszych (mogą być równe). Podaj, ile ciągów z pliku jest zapisem binarnym liczby półpierwszej, oraz największą i najmniejszą taką liczbę w zapisie dziesiętnym.

Przykład: 100010 = 34 = 2·17 i 1101001001 = 841 = 29·29 są półpierwsze; 1100101 = 101 (liczba pierwsza) i 1111111111 = 1023 = 3·11·31 — nie.

Python
def liczba_czynnikow(n):  # liczba czynników pierwszych (z powtórzeniami)
    ile = 0
    i = 2
    while i * i <= n:
        while n % i == 0:
            ile += 1
            n //= i
        i += 1
    if n > 1:
        ile += 1
    return ile


polpierwsze = []

with open("ciagi.txt") as file:
    for line in file:
        liczba = int(line.strip(), 2)
        if liczba_czynnikow(liczba) == 2:
            polpierwsze.append(liczba)

print(len(polpierwsze))
print("min:", min(polpierwsze), "max:", max(polpierwsze))
C++23
#include <algorithm>
#include <fstream>
#include <print>
#include <string>
#include <vector>

int liczba_czynnikow(long long n) {  // liczba czynników pierwszych (z powtórzeniami)
    int ile = 0;
    long long i = 2;
    while (i * i <= n) {
        while (n % i == 0) {
            ile += 1;
            n /= i;
        }
        i += 1;
    }
    if (n > 1) {
        ile += 1;
    }
    return ile;
}

int main() {
    std::vector<long long> polpierwsze;

    std::ifstream plik("ciagi.txt");
    std::string ciag;
    while (plik >> ciag) {
        long long liczba = std::stoll(ciag, nullptr, 2);  // zapis dwójkowy
        if (liczba_czynnikow(liczba) == 2) {
            polpierwsze.push_back(liczba);
        }
    }

    std::println("{}", polpierwsze.size());
    std::println("min: {} max: {}", std::ranges::min(polpierwsze), std::ranges::max(polpierwsze));
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Zadanie {
    static int liczbaCzynnikow(long n) {  // liczba czynników pierwszych (z powtórzeniami)
        int ile = 0;
        long i = 2;
        while (i * i <= n) {
            while (n % i == 0) {
                ile += 1;
                n /= i;
            }
            i += 1;
        }
        if (n > 1) {
            ile += 1;
        }
        return ile;
    }

    public static void main(String[] args) throws Exception {
        List<Long> polpierwsze = new ArrayList<>();

        for (String linia : Files.readAllLines(Path.of("ciagi.txt"))) {
            long liczba = Long.parseLong(linia.strip(), 2);  // zapis dwójkowy
            if (liczbaCzynnikow(liczba) == 2) {
                polpierwsze.add(liczba);
            }
        }

        System.out.println(polpierwsze.size());
        System.out.println("min: " + Collections.min(polpierwsze) + " max: " + Collections.max(polpierwsze));
    }
}
