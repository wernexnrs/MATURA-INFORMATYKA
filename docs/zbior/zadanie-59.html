Treść zadania

Plik liczby.txt zawiera 1000 różnych liczb (każda ma od 2 do 9 cyfr), po jednej w wierszu. Odpowiedzi zapisz w pliku wyniki_liczby.txt, poprzedzając każdą numerem zadania.

Podaj, ile liczb z pliku ma w rozkładzie na czynniki pierwsze dokładnie trzy różne czynniki, z których każdy jest nieparzysty (czynniki mogą się powtarzać).

Przykład: 105 = 3·5·7 i 1287 = 3·3·11·13 spełniają warunek; 210 = 2·3·5·7, 1331 = 11³ i 3465 = 3·3·5·7·11 — nie.

Python
def rozne_czynniki(n):
    czynniki = set()
    i = 2
    while i * i <= n:
        while n % i == 0:
            czynniki.add(i)
            n //= i
        i += 1
    if n > 1:
        czynniki.add(n)
    return czynniki


count = 0

with open("liczby.txt") as file:
    for line in file:
        czynniki = rozne_czynniki(int(line))
        if len(czynniki) == 3 and 2 not in czynniki:
            count += 1

print(count)

import java.nio.file.Files;

import java.util.HashSet;

public class Zadanie {
    static Set<Long> rozneCzynniki(long n) {
        Set<Long> czynniki = new HashSet<>();
        long i = 2;
        while (i * i <= n) {
            while (n % i == 0) {
                czynniki.add(i);
                n /= i;
            }
            i += 1;
        }
        if (n > 1) {
            czynniki.add(n);
        }
        return czynniki;

        int count = 0;

        for (String linia : Files.readAllLines(Path.of("liczby.txt"))) {
            Set<Long> czynniki = rozneCzynniki(Long.parseLong(linia.trim()));
            if (czynniki.size() == 3 && !czynniki.contains(2L)) {
                count += 1;
            }
        }

        System.out.println(count);
    }
}

Treść zadania

Dane jak w zadaniu 59.1 (1000 liczb w pliku liczby.txt).

Podaj, ile jest liczb, dla których suma liczby i liczby powstałej z odwrócenia kolejności jej cyfr jest palindromem (w zapisie dziesiętnym).

Przykład: 45 + 54 = 99 — tak; 8542 + 2458 = 11000 — nie.

Python
count = 0

with open("liczby.txt") as file:
    for line in file:
        liczba = line.strip()
        suma = str(int(liczba) + int(liczba[::-1]))
        if suma == suma[::-1]:
            count += 1

print(count)
C++23
#include <fstream>
#include <print>
#include <string>

std::string odwroc(const std::string& s) {
    return std::string(s.rbegin(), s.rend());
}

int main() {
    int count = 0;

    std::ifstream plik("liczby.txt");
    std::string liczba;
    while (plik >> liczba) {
        // suma może przekroczyć zakres int - używamy long long
        std::string suma = std::to_string(std::stoll(liczba) + std::stoll(odwroc(liczba)));
        if (suma == odwroc(suma)) {
            count += 1;
        }
    }

    std::println("{}", count);
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;

public class Zadanie {
    static String odwroc(String s) {
        return new StringBuilder(s).reverse().toString();
    }

    public static void main(String[] args) throws Exception {
        int count = 0;

        for (String linia : Files.readAllLines(Path.of("liczby.txt"))) {
            String liczba = linia.strip();
            // suma może przekroczyć zakres int - używamy long
            String suma = String.valueOf(Long.parseLong(liczba) + Long.parseLong(odwroc(liczba)));
            if (suma.equals(odwroc(suma))) {
                count += 1;
            }
        }

        System.out.println(count);
    }
}

Dane jak w zadaniu 59.1.

Niech w(n) oznacza iloczyn cyfr liczby n. Dla liczby n tworzymy ciąg n1 = w(n), n2 = w(n1), …, aż otrzymamy liczbę jednocyfrową nk; wtedy mocą liczby n jest k. Podaj, ile liczb w pliku ma moc 1, 2, 3, …, 8. Dodatkowo podaj najmniejszą i największą liczbę o mocy 1.

Przykład: 678 → 336 → 54 → 20 → 0, więc moc 678 wynosi 4; 1991 → 81 → 8, więc moc 1991 wynosi 2.

Python
def iloczyn_cyfr(x):
    wynik = 1
    for cyfra in str(x):
        wynik *= int(cyfra)
    return wynik


def moc(x):
    k = 0
    while x > 9:
        x = iloczyn_cyfr(x)
        k += 1
    return k


ile = [0] * 9
moc_1 = []

with open("liczby.txt") as file:
    for line in file:
        liczba = int(line)
        m = moc(liczba)
        ile[m] += 1
        if m == 1:
            moc_1.append(liczba)

for m in range(1, 9):
    print(m, ile[m])
print("min:", min(moc_1), "max:", max(moc_1))

long long iloczyn_cyfr(long long x) {
    long long wynik = 1;
    for (char cyfra : std::to_string(x)) {
        wynik *= cyfra - '0';
    }
    return wynik;
}

int moc(long long x) {
    int k = 0;
    while (x > 9) {
        x = iloczyn_cyfr(x);
        k += 1;
    }
    return k;
}

int main() {
    std::vector<int> ile(9, 0);
    std::vector<long long> moc_1;

    std::ifstream plik("liczby.txt");
    long long liczba;
    while (plik >> liczba) {
        int m = moc(liczba);
        ile[m] += 1;
        if (m == 1) {
            moc_1.push_back(liczba);
        }
    }

    for (int m = 1; m < 9; m++) {
        std::println("{} {}", m, ile[m]);
    }
    std::println("min: {} max: {}", std::ranges::min(moc_1), std::ranges::max(moc_1));
}
Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Zadanie {
    static long iloczynCyfr(long x) {
        long wynik = 1;
        for (char cyfra : String.valueOf(x).toCharArray()) {
            wynik *= cyfra - '0';
        }
        return wynik;
    }

    static int moc(long x) {
        int k = 0;
        while (x > 9) {
            x = iloczynCyfr(x);
            k += 1;
        }
        return k;
    }

    public static void main(String[] args) throws Exception {
        int[] ile = new int[9];
        List<Long> moc1 = new ArrayList<>();

        for (String linia : Files.readAllLines(Path.of("liczby.txt"))) {
            long liczba = Long.parseLong(linia.trim());
            int m = moc(liczba);
            ile[m] += 1;
            if (m == 1) {
                moc1.add(liczba);
            }
        }

        for (int m = 1; m < 9; m++) {
            System.out.println(m + " " + ile[m]);
        }
        System.out.println("min: " + Collections.min(moc1) + " max: " + Collections.max(moc1));
    }
}
