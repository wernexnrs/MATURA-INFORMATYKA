def ostatnia_cyfra(k):
    # Dane: k >= 0, Wynik: reszta z dzielenia 2^k przez 10 (stała liczba operacji)
    if k == 0:
        return 1
    reszta = k % 4
    if reszta == 0:
        return 6
    if reszta == 1:
        return 2
    if reszta == 2:
        return 4
    return 8


print(ostatnia_cyfra(15))
print(all(ostatnia_cyfra(k) == 2 ** k % 10 for k in range(1000)))

Odpowiedź: k = 0 → 1; w przeciwnym razie zależnie od k mod 4: 0 → 6, 1 → 2, 2 → 4, 3 → 8 (dla k = 15 wynik 8)

def potega(a, n):
    # Dane: a - liczba całkowita, n = 2^k, Wynik: a^n (log2(n) mnożeń)
    p = a
    while n > 1:
        p = p * p
        n = n // 2
    return p


print(potega(3, 8), 3 ** 8)
print(potega(2, 1024) == 2 ** 1024)

Odpowiedź: p := a; dopóki n > 1: p := p·p, n := n div 2; wynikiem jest p

def reg2(w):
    print(w)
    if len(w) == 1:
        return "TAK"
    if len(w) % 2 == 1:
        return "NIE"
    polowa = len(w) // 2
    w1 = w[:polowa]
    w2 = w[polowa:]
    if w1 != w2:
        return "NIE"
    return reg2(w1)


for napis in ["aabbaabb", "aaaaaaaa", "bbbbbbbbbbbbbbbbbbbb"]:
    print(reg2(napis))
    print()

Odpowiedź: i. aabbaabb → aabb → NIE; ii. aaaaaaaa → aaaa → aa → a → TAK; iii. bbbbbbbbbbbbbbbbbbbb → bbbbbbbbbb → bbbbb → NIE

def reg3(w):
    # Dane: napis w nad alfabetem {a, b}, Wynik: TAK, jeśli w jest 3-regularny
    n = len(w)
    if n == 1:
        return "TAK"
    if n % 3 != 0:
        return "NIE"
    d = n // 3
    w1 = w[:d]
    w2 = w[d:2 * d]
    w3 = w[2 * d:]
    if w1 == w2 or w1 == w3:
        return reg3(w1)
    return "NIE"


for napis in ["a", "aba", "abaabaaaa", "aaaabaaba"]:
    print(napis, reg3(napis))

Odpowiedź: dzielimy w na trzy równe części w1 w2 w3; jeśli w1 = w2 lub w1 = w3, to wynikiem jest 3REG(w1), w przeciwnym razie NIE (a, aba, abaabaaaa — TAK; aaaabaaba — NIE)

def podziel_mandaty(glosy, mandaty):
    # uproszczona metoda d'Hondta: X = v / (s + 1)
    s = [0] * len(glosy)
    for _ in range(mandaty):
        najlepszy = 0
        for i in range(len(glosy)):
            if glosy[i] / (s[i] + 1) > glosy[najlepszy] / (s[najlepszy] + 1):
                najlepszy = i
        s[najlepszy] += 1
    return s


print("1")
for i in range(6):
    print(komitety[i], sum(okreg[i] for okreg in okregi))

print("2")
sumy = [sum(okreg[:6]) for okreg in okregi]
print("najwięcej: okręg nr", sumy.index(max(sumy)) + 1)
print("najmniej: okręg nr", sumy.index(min(sumy)) + 1)

def utworz_haslo(w):
    # najdłuższy palindrom będący początkiem słowa w
    dlugosc = len(w)
    while w[:dlugosc] != w[:dlugosc][::-1]:
        dlugosc -= 1
    w2 = w[dlugosc:]
    return w2[::-1] + w


hasla = []
with open("slowa.txt") as file:
    for linia in file:
        hasla.append(utworz_haslo(linia.strip()))

with open("hasla_b.txt", "w") as wyjscie:
    for haslo in hasla:
        wyjscie.write(haslo + "\n")

print("1")
for haslo in hasla:
    if len(haslo) == 12:
        print(haslo)

print("2")
print(min(hasla, key=len))
print(max(hasla, key=len))

print("3")
print(sum(len(haslo) for haslo in hasla))
