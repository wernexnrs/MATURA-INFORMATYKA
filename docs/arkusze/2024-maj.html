def algorytm(A):
    n = len(A)
    m = len(A[0])
    P = [[False] * m for _ in range(n)]
    P[0][0] = True
    for i in range(n):
        for j in range(m):
            if A[i][j] == 0:
                P[i][j] = False
            elif i == 0 and j != 0:
                P[i][j] = P[i][j - 1]
            elif i != 0 and j == 0:
                P[i][j] = P[i - 1][j]
            elif i != 0 and j != 0:
                P[i][j] = P[i][j - 1] or P[i - 1][j]
    return P[n - 1][m - 1]

for n in range(2, 5):
    najwiecej = 0
    for maska in range(2 ** (n * n)):
        A = [[0 if maska >> (i * n + j) & 1 else 1 for j in range(n)] for i in range(n)]
        if A[0][0] == 1 and A[n - 1][n - 1] == 1 and algorytm(A):
            czarne = sum(row.count(0) for row in A)
            najwiecej = max(najwiecej, czarne)

def algorytm(n):
    b = 1
    c = 0
    wykonania = 0
    while n > 0:
        a = n % 10
        n = n // 10
        if a % 2 == 0:
            c = c + b * (a // 2)
        else:
            c = c + b
            wykonania += 1
        b = b * 10
    return c, wykonania

    print(n, *algorytm(n))

# 4.3
wystapienia = {}
for p in pierwsze:
    wystapienia[p] = wystapienia.get(p, 0) + 1

wynik = []
for x in liczby:
    reszta = x
    dobra = True
    for p in wystapienia:
        ile_razy = 0
        while reszta % p == 0:
            reszta //= p
            ile_razy += 1
        if ile_razy > wystapienia[p]:
            dobra = False
    if dobra and reszta == 1:
        wynik.append(x)
print("4.3", *wynik)

# 4.4
n = len(pierwsze)
prefiks = [0] * (n + 1)
for i in range(n):
    prefiks[i + 1] = prefiks[i] + pierwsze[i]

najl_suma, najl_dl, najl_start = 0, 1, -1
for start in range(n):
    for koniec in range(start + 50, n + 1):
        suma = prefiks[koniec] - prefiks[start]
        dl = koniec - start
        # suma / dl > najl_suma / najl_dl, bez ułamków
        if najl_start == -1 or suma * najl_dl > najl_suma * dl:
            najl_suma, najl_dl, najl_start = suma, dl, start
print("4.4", najl_suma / najl_dl, najl_dl, pierwsze[najl_start])
