def dobre_przyjaciolki(a, b):
    ostatnia_a = a % 10
    ostatnia_b = b % 10
    suma_a = 0
    while a > 0:
        if a < 10:
            pierwsza_a = a
        suma_a += a % 10
        a //= 10
    suma_b = 0
    while b > 0:
        if b < 10:
            pierwsza_b = b
        suma_b += b % 10
        b //= 10
    if suma_a != suma_b:
        return False
    return pierwsza_a == ostatnia_b or pierwsza_b == ostatnia_a

def iloczyn(x, y):
    global dodawania
    if y == 1:
        return x
    k = y // 2
    z = iloczyn(x, k)
    if y % 2 == 0:
        dodawania += 1
        wynik = z + z
    else:
        dodawania += 2
        wynik = x + z + z
    print(x, y, k, z, wynik)        # 2.1 - wiersz tabeli (wypisywane od najgłębszego wywołania)
    return wynik

def iloczyn_iter(x, y):
    z = 0
    while y >= 1:
        if y % 2 == 1:
            z = z + x
        x = x + x
        y = y // 2
    return z

# 4.2 - pierwsza cyfra musi być 1, pozostałe jedynki rozmieszczamy na 7 miejscach: C(7, jedynki - 1)
def silnia(n):
    s = 1
    for i in range(2, n + 1):
        s *= i
    return s


def anagramy(liczba):
    reszta = len(liczba) - 1
    jedynki = liczba.count("1") - 1
    return silnia(reszta) // (silnia(jedynki) * silnia(reszta - jedynki))
