def minuty(godzina):
    h, m = godzina.split(":")
    return int(h) * 60 + int(m)


strategie = {
    "A": lambda f: (-(f[2] - f[1]), f[2]),  # najdłuższy, potem najwcześniej się kończy
    "B": lambda f: (f[2] - f[1], f[2]),     # najkrótszy, potem najwcześniej się kończy
    "C": lambda f: (f[1], f[2]),            # najwcześniej się zaczyna, potem najwcześniej się kończy
    "D": lambda f: (f[2], -f[1]),           # najwcześniej się kończy, potem najpóźniej się zaczyna
}


def telewidz(program, strategia):
    T = [(nazwa, minuty(od), minuty(do)) for nazwa, od, do in program]
    P = []
    while T:
        x = min(T, key=strategie[strategia])
        T.remove(x)
        P.append(x[0])
        # usuwamy filmy kolidujące z x (koniec jednego = początek drugiego to nie kolizja)
        T = [f for f in T if f[2] <= x[1] or f[1] >= x[2]]
    return P

def rozszerzony_euklides(a, b, poziom=1):
    if b == 0:
        print(poziom, a, b, 1, 0)
        return 1, 0
    r = a % b
    x, y = rozszerzony_euklides(b, r, poziom + 1)  # krok 3: (b, r)
    x, y = y, x - (a // b) * y                     # krok 4: (y', x' - (a div b) * y')
    print(poziom, a, b, x, y)
    return x, y


# 3.1 - tabela dla a = 188, b = 12 (wiersze wypisywane przy powrocie z rekurencji)
x, y = rozszerzony_euklides(188, 12)
print("NWD =", 188 * x + 12 * y)

Odpowiedź: 3.1: (188, 12, -1, 16), (12, 8, 1, -1), (8, 4, 0, 1), (4, 0, 1, 0). 3.2: krok 3: (b, r), krok 4: (y', x' - (a div b) · y').

suma_2025 = 0
najwiecej = 0
najwieksze = ""
przeludnione = 0

with open("kraina.txt") as file:
    for line in file:
        nazwa, k13, m13, k14, m14 = line.strip().split(";")
        l2013 = int(k13) + int(m13)
        l2014 = int(k14) + int(m14)
        # tempo zaokrąglone w dół do 4 miejsc - liczymy w dziesięciotysięcznych, żeby uniknąć błędów float
        tempo = l2014 * 10000 // l2013

        ludnosc = l2014
        przeludnienie = ludnosc > 2 * l2013
        for rok in range(2015, 2026):
            if not przeludnienie:
                ludnosc = ludnosc * tempo // 10000
                if ludnosc > 2 * l2013:
                    przeludnienie = True

        if przeludnienie:
            przeludnione += 1
        suma_2025 += ludnosc
        if ludnosc > najwiecej:
            najwiecej = ludnosc
            najwieksze = nazwa

print(suma_2025)
print(najwieksze)
print(przeludnione)
