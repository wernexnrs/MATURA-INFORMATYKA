Treść zadania

Problem telewidza. Program telewizyjny to lista filmów emitowanych jednego dnia w różnych stacjach. Telewidz chce obejrzeć w całości jak najwięcej filmów, ale może oglądać tylko jeden film naraz (zmiana stacji nie zajmuje czasu, więc film kończący się o 12:00 nie koliduje z filmem zaczynającym się o 12:00). Przykładowy program:

  • TV1: film 1 od 9:00 do 12:00, film 2 od 15:00 do 17:00,
  • TV2: film 3 od 11:00 do 16:00,
  • TV3: film 4 od 12:00 do 14:00,
  • TV4: film 5 od 11:30 do 12:30.

Algorytm (dane: zbiór filmów T i strategia S; wynik: zbiór P filmów do obejrzenia): dopóki T nie jest pusty, wybierz z T film x według strategii S, przenieś go do P i usuń z T wszystkie filmy kolidujące z x.

  • Strategia A — najdłuższy film (remis: kończący się najwcześniej),
  • Strategia B — najkrótszy film (remis: kończący się najwcześniej),
  • Strategia C — film zaczynający się najwcześniej (remis: kończący się najwcześniej),
  • Strategia D — film kończący się najwcześniej (remis: zaczynający się najpóźniej).

Dla podanego programu strategia A daje P = {film 3}.

  • 1.1 — Podaj zawartość zbioru P po wykonaniu algorytmu dla powyższego programu przy strategiach B, C i D.
  • 1.2 — Strategia jest optymalna, jeśli dla każdego programu daje największą możliwą liczbę filmów. Dla każdej ze strategii A, B i C podaj kontrprzykład: program z czterema filmami w dwóch stacjach (godziny emisji i czasy trwania), wynik algorytmu dla tej strategii oraz liczniejszy zbiór filmów, które telewidz może obejrzeć.
Python
def minuty(godzina):
    h, m = godzina.split(":")
    return int(h) * 60 + int(m)


strategie = {
    "A": lambda f: (-(f[2] - f[1]), f[2]),  # najdłuższy, potem najwcześniej się kończy
    "B": lambda f: (f[2] - f[1], f[2]),     # najkrótszy, potem najwcześniej się kończy
    "C": lambda f: (f[1], f[2]),            # najwcześniej się zaczyna, potem najwcześniej się kończy
    "D": lambda f: (f[2], -f[1]),           # najwcześniej się kończy, potem najpóźniej się zaczyna
}


def telewidz(program, strategia):
    T = [(nazwa, minuty(od), minuty(do)) for nazwa, od, do in program]
    P = []
    while T:
        x = min(T, key=strategie[strategia])
        T.remove(x)
        P.append(x[0])
        # usuwamy filmy kolidujące z x (koniec jednego = początek drugiego to nie kolizja)
        T = [f for f in T if f[2] <= x[1] or f[1] >= x[2]]
    return P

#include <iostream>
#include <map>
#include <string>
#include <utility>
#include <vector>

struct Film {
    std::string nazwa;
    int od, koniec;
};

struct Pozycja {  // film z programu: nazwa, godzina rozpoczęcia, godzina zakończenia
    std::string nazwa, od, koniec;
};

int minuty(const std::string& godzina) {
    auto dwukropek = godzina.find(':');
    return std::stoi(godzina.substr(0, dwukropek)) * 60 + std::stoi(godzina.substr(dwukropek + 1));
}

// klucz wyboru filmu dla danej strategii (porównywany jak krotka w Pythonie)
std::pair<int, int> klucz(const Film& f, char strategia) {
    if (strategia == 'A') return {-(f.koniec - f.od), f.koniec};  // najdłuższy, potem najwcześniej się kończy
    if (strategia == 'B') return {f.koniec - f.od, f.koniec};     // najkrótszy, potem najwcześniej się kończy
    if (strategia == 'C') return {f.od, f.koniec};                // najwcześniej się zaczyna, potem najwcześniej się kończy
    return {f.koniec, -f.od};                                     // D: najwcześniej się kończy, potem najpóźniej się zaczyna
}

std::vector<std::string> telewidz(const std::vector<Pozycja>& program, char strategia) {
    std::vector<Film> T;
    for (const auto& p : program) T.push_back({p.nazwa, minuty(p.od), minuty(p.koniec)});
    std::vector<std::string> P;
    while (!T.empty()) {
        size_t najlepszy = 0;
        for (size_t i = 1; i < T.size(); i++)
            if (klucz(T[i], strategia) < klucz(T[najlepszy], strategia)) najlepszy = i;
        Film x = T[najlepszy];
        T.erase(T.begin() + najlepszy);
        P.push_back(x.nazwa);
        // usuwamy filmy kolidujące z x (koniec jednego = początek drugiego to nie kolizja)
        std::vector<Film> reszta;
        for (const auto& f : T)
            if (f.koniec <= x.od || f.od >= x.koniec) reszta.push_back(f);
        T = reszta;
    }
    return P;
}

std::string jako_lista(const std::vector<std::string>& P) {  // zapis jak lista w Pythonie
    std::string s = "[";
    for (size_t i = 0; i < P.size(); i++) {
        if (i > 0) s += ", ";
        s += "'" + P[i] + "'";
    }
    return s + "]";
}

import java.util.*;

public class Zadanie {
    record Film(String nazwa, int od, int koniec) {}

    static int minuty(String godzina) {
        var czesci = godzina.split(":");
        return Integer.parseInt(czesci[0]) * 60 + Integer.parseInt(czesci[1]);
    }

    // porównanie filmów dla danej strategii (jak klucz-krotka w Pythonie)
    static Comparator<Film> strategia(char s) {
        return switch (s) {
            case 'A' -> Comparator.comparingInt((Film f) -> -(f.koniec() - f.od())).thenComparingInt(Film::koniec);
            case 'B' -> Comparator.comparingInt((Film f) -> f.koniec() - f.od()).thenComparingInt(Film::koniec);
            case 'C' -> Comparator.comparingInt(Film::od).thenComparingInt(Film::koniec);
            default -> Comparator.comparingInt(Film::koniec).thenComparingInt((Film f) -> -f.od());
        };
    }

    static List<String> telewidz(String[][] program, char s) {
        var T = new ArrayList<Film>();
        for (var p : program) T.add(new Film(p[0], minuty(p[1]), minuty(p[2])));
        var P = new ArrayList<String>();
        while (!T.isEmpty()) {
            Film x = T.get(0);
            for (var f : T) if (strategia(s).compare(f, x) < 0) x = f;  // pierwszy najmniejszy
            T.remove(x);
            P.add(x.nazwa());
            // usuwamy filmy kolidujące z x (koniec jednego = początek drugiego to nie kolizja)
            final Film wybrany = x;
            T.removeIf(f -> !(f.koniec() <= wybrany.od() || f.od() >= wybrany.koniec()));
        }
        return P;
    }

    static String jakoLista(List<String> P) {  // zapis jak lista w Pythonie
        var s = new StringJoiner(", ", "[", "]");
        for (var nazwa : P) s.add("'" + nazwa + "'");
        return s.toString();
    }

    public static void main(String[] args) {
        // 1.1 - program z treści zadania
        String[][] program = {
            {"film 1", "9:00", "12:00"},
            {"film 2", "15:00", "17:00"},
            {"film 3", "11:00", "16:00"},
            {"film 4", "12:00", "14:00"},
            {"film 5", "11:30", "12:30"},
        };
        for (char s : "ABCD".toCharArray())
            System.out.println(s + " " + jakoLista(telewidz(program, s)));

        // 1.2 - kontrprzykłady (strategia D jest optymalna, więc służy jako wzorzec)
        var kontrprzyklady = new LinkedHashMap<Character, String[][]>();
        kontrprzyklady.put('A', new String[][] {{"film 1", "10:00", "12:00"}, {"film 2", "12:00", "14:00"},
                                               {"film 3", "10:00", "11:00"}, {"film 4", "11:00", "12:00"}});
        kontrprzyklady.put('B', new String[][] {{"film 1", "11:30", "12:30"}, {"film 2", "15:00", "16:00"},
                                               {"film 3", "10:00", "12:00"}, {"film 4", "12:00", "14:00"}});
        kontrprzyklady.put('C', new String[][] {{"film 1", "9:00", "14:00"}, {"film 2", "15:00", "16:00"},
                                               {"film 3", "10:00", "12:00"}, {"film 4", "12:00", "14:00"}});
        for (var e : kontrprzyklady.entrySet())
            System.out.println(e.getKey() + " " + jakoLista(telewidz(e.getValue(), e.getKey()))
                    + " optymalnie: " + jakoLista(telewidz(e.getValue(), 'D')));
    }
}

Rozszerzony algorytm Euklidesa. Dla liczb całkowitych a > 0 i b ≥ 0 należy wyznaczyć liczby całkowite x, y takie, że NWD(a, b) = a·x + b·y. Jeśli b ≠ 0 i dla pary (b, a mod b) znamy x', y' spełniające NWD(b, a mod b) = b·x' + (a mod b)·y', to x = y', y = x' − (a div b)·y'. Dla b = 0 wynikiem jest para (1, 0). Przykład: NWD(231, 30) = 3 = 3·231 + (−23)·30.

  • 3.1 — Uzupełnij tabelę kolejnych wywołań funkcji RozszerzonyEuklides(a, b) dla a = 188, b = 12: dla każdego wywołania podaj wartości a i b oraz zwrócone x i y.
  • 3.2 — Uzupełnij rekurencyjną funkcję:
Kod
RozszerzonyEuklides(a, b):
  Krok 1. Jeśli b = 0, zwróć (1, 0) i zakończ.
  Krok 2. r ← a mod b
  Krok 3. (x, y) ← RozszerzonyEuklides(____, ____)
  Krok 4. Zwróć parę (____, ____).
Python
def rozszerzony_euklides(a, b, poziom=1):
    if b == 0:
        print(poziom, a, b, 1, 0)
        return 1, 0
    r = a % b
    x, y = rozszerzony_euklides(b, r, poziom + 1)  # krok 3: (b, r)
    x, y = y, x - (a // b) * y                     # krok 4: (y', x' - (a div b) * y')
    print(poziom, a, b, x, y)
    return x, y


# 3.1 - tabela dla a = 188, b = 12 (wiersze wypisywane przy powrocie z rekurencji)
x, y = rozszerzony_euklides(188, 12)
print("NWD =", 188 * x + 12 * y)

#include <iostream>
#include <utility>

std::pair<long long, long long> rozszerzony_euklides(long long a, long long b, int poziom = 1) {
    if (b == 0) {
        std::cout << poziom << ' ' << a << ' ' << b << ' ' << 1 << ' ' << 0 << '\n';
        return {1, 0};
    }
    long long r = a % b;
    auto [x1, y1] = rozszerzony_euklides(b, r, poziom + 1);  // krok 3: (b, r)
    long long x = y1, y = x1 - (a / b) * y1;                  // krok 4: (y', x' - (a div b) * y')
    std::cout << poziom << ' ' << a << ' ' << b << ' ' << x << ' ' << y << '\n';
    return {x, y};
}

int main() {
    // 3.1 - tabela dla a = 188, b = 12 (wiersze wypisywane przy powrocie z rekurencji)
    auto [x, y] = rozszerzony_euklides(188, 12);
    std::cout << "NWD = " << 188 * x + 12 * y << '\n';
}

public class Zadanie {
    static long[] rozszerzonyEuklides(long a, long b, int poziom) {
        if (b == 0) {
            System.out.println(poziom + " " + a + " " + b + " " + 1 + " " + 0);
            return new long[] {1, 0};
        }
        long r = a % b;
        long[] xy = rozszerzonyEuklides(b, r, poziom + 1);  // krok 3: (b, r)
        long x = xy[1], y = xy[0] - (a / b) * xy[1];        // krok 4: (y', x' - (a div b) * y')
        System.out.println(poziom + " " + a + " " + b + " " + x + " " + y);
        return new long[] {x, y};
    }

    public static void main(String[] args) {
        // 3.1 - tabela dla a = 188, b = 12 (wiersze wypisywane przy powrocie z rekurencji)
        long[] xy = rozszerzonyEuklides(188, 12, 1);
        System.out.println("NWD = " + (188 * xy[0] + 12 * xy[1]));
    }
}

#include <fstream>
#include <iostream>
#include <string>

// liczby mają do 200 cyfr, więc nie mieszczą się w long long - porównujemy napisy:
// dłuższa liczba (bez zer wiodących) jest większa, przy równej długości decyduje porządek leksykograficzny
bool mniejsza(const std::string& a, const std::string& b) {
    if (a.size() != b.size()) return a.size() < b.size();
    return a < b;
}

int main() {
    std::vector<std::string> lista;
    std::ifstream plik("liczby.txt");
    std::string liczba;
    while (plik >> liczba) {
        while (liczba.size() > 1 && liczba[0] == '0') liczba.erase(0, 1);  // usuwamy zera wiodące
        lista.push_back(liczba);
    }
    size_t najmniejsza = 0, najwieksza = 0;  // pierwsze wystąpienie, jak index() w Pythonie
    for (size_t i = 1; i < lista.size(); i++) {
        if (mniejsza(lista[i], lista[najmniejsza])) najmniejsza = i;
        if (mniejsza(lista[najwieksza], lista[i])) najwieksza = i;
    }
    // numer wiersza najmniejszej i największej liczby
    std::cout << najmniejsza + 1 << ' ' << najwieksza + 1 << '\n';
}
