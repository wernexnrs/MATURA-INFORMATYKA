# biblioteczka jako słownik: (półka, przegródka) -> numer książki
def wstaw(B, ksiazka):
    i, j = 0, 1
    while (i, j) in B:
        if ksiazka < B[(i, j)]:
            i, j = i + 1, 2 * j - 1
        else:
            i, j = i + 1, 2 * j
    B[(i, j)] = ksiazka


def A(B, i, j, wynik):
    wynik.append(B[(i, j)])
    if (i + 1, 2 * j - 1) in B:
        A(B, i + 1, 2 * j - 1, wynik)
    if (i + 1, 2 * j) in B:
        A(B, i + 1, 2 * j, wynik)
    return wynik

# 2.1 - liczba bloków bez funkcji wbudowanych
def bloki(n):
    b = 1
    poprzednia = n % 2
    n = n // 2
    while n > 0:
        cyfra = n % 2
        if cyfra != poprzednia:
            b += 1
            poprzednia = cyfra
        n = n // 2
    return b


def na_dziesietny(napis):
    liczba = 0
    for c in napis:
        liczba = liczba * 2 + int(c)
    return liczba


def na_binarny(n):
    if n == 0:
        return "0"
    wynik = ""
    while n > 0:
        wynik = str(n % 2) + wynik
        n = n // 2
    return wynik


print("2.1", bloki(67), bloki(245))

with open("bin.txt") as file:

def rosnaco_malejacy(c):
    n = len(c)
    if n < 4:
        return False
    for k in range(2, n - 1):  # a1..ak rosnący, a(k+1)..an malejący
        rosnie = all(c[i] < c[i + 1] for i in range(k - 1))
        maleje = all(c[i] > c[i + 1] for i in range(k, n - 1))
        if rosnie and maleje:
            return True
    return False

def na_dziesietny(napis, podstawa):
    liczba = 0
    for c in napis:
        liczba = liczba * podstawa + int(c)
    return liczba
