Zadanie 2 (punkty kratowe). Punkt kratowy ma obie współrzędne całkowite. K(R) to koło o środku (0, 0) i promieniu R (z brzegiem), a N(R) — liczba punktów kratowych w K(R). Np. N(0) = 1, N(1) = 5, N(1,5) = 9.

  • 2a — podaj N(R) dla R = 2,01 oraz R = 4,50;
  • 2b — zaproponuj algorytm, który dla danej nieujemnej liczby całkowitej R oblicza N(R).
Python
from math import floor


def punkty_kratowe(R):
    # liczba punktów (x, y) o współrzędnych całkowitych, dla których x^2 + y^2 <= R^2
    r = floor(R)
    N = 0
    for x in range(-r, r + 1):
        for y in range(-r, r + 1):
            if x * x + y * y <= R * R:
                N += 1
    return N


print(punkty_kratowe(0), punkty_kratowe(1), punkty_kratowe(1.5))  # przykłady z treści: 1 5 9
print(punkty_kratowe(2.01))
print(punkty_kratowe(4.50))

def nwd(a, b):
    while b > 0:
        reszta = a % b
        a = b

    return a

Zadanie 4 (iteracje). Dwa przekształcenia punktu (x, y), stosowane wielokrotnie w losowej kolejności (każde z prawdopodobieństwem 1/2), dają obraz smoka Heighwaya:

  • x' = −0,4x − 1, y' = −0,4y + 0,1;
  • x' = 0,76x − 0,4y, y' = 0,4x + 0,76y.

Wyniki podpunktów a, c, d zapisz w pliku zad_4.txt, poprzedzając je literą podpunktu:

  • a) zaczynając od x = 1, y = 1, oblicz pierwsze 5000 wartości (x, y) kolejnych iteracji;
  • b) sporządź obraz smoka (pomiń 100 pierwszych iteracji) i zapisz go w pliku smok.*;
  • c) oblicz środek masy smoka — średnie wartości x i y (zaokrąglone do 0,1), pomijając 100 pierwszych iteracji;
  • d) podaj minimalne i maksymalne wartości x oraz y (zaokrąglone do 0,1), pomijając 100 pierwszych iteracji.

x, y = 1, 1

std::string jak_python(double x) {
    if (x == 0) return std::signbit(x) ? "-0.0" : "0.0";
    char bufor[64];
    auto koniec = std::to_chars(bufor, bufor + sizeof bufor, std::fabs(x), std::chars_format::scientific).ptr;
    std::string s(bufor, koniec); // np. "1.2345e-05"
    size_t e = s.find('e');
    int wykladnik = std::stoi(s.substr(e + 1));
    std::string cyfry = s.substr(0, e);
    std::erase(cyfry, '.'); // np. "12345"
    std::string znak = x < 0 ? "-" : "";
    if (wykladnik < -4 || wykladnik >= 16) {
        std::string mantysa = cyfry.substr(0, 1) + (cyfry.size() > 1 ? "." + cyfry.substr(1) : "");
        return std::format("{}{}e{}{:02}", znak, mantysa, wykladnik < 0 ? '-' : '+', std::abs(wykladnik));
    }
    if (wykladnik < 0) return znak + "0." + std::string(-wykladnik - 1, '0') + cyfry;
    if ((int)cyfry.size() <= wykladnik + 1)
        return znak + cyfry + std::string(wykladnik + 1 - cyfry.size(), '0') + ".0";
    return znak + cyfry.substr(0, wykladnik + 1) + "." + cyfry.substr(wykladnik + 1);
}

import java.io.PrintWriter;
import java.math.*;
import java.util.*;

public class Zadanie {
    // liczba rzeczywista zapisana tak jak w Pythonie: np. 2.0, -0.123, 1.5e-05
    // (Java pisze małe liczby inaczej, np. 1.5E-5)
    static String jakPython(double x) {
        if (x == 0) return (1 / x < 0) ? "-0.0" : "0.0";
        var bd = new BigDecimal(Double.toString(Math.abs(x))).stripTrailingZeros();
        String cyfry = bd.unscaledValue().toString();
        int wykladnik = cyfry.length() - 1 - bd.scale();
        String znak = x < 0 ? "-" : "";
        if (wykladnik < -4 || wykladnik >= 16) {
            String mantysa = cyfry.length() > 1 ? cyfry.charAt(0) + "." + cyfry.substring(1) : cyfry;
            return znak + mantysa + "e" + (wykladnik < 0 ? "-" : "+") + String.format("%02d", Math.abs(wykladnik));
        }
        String s = bd.toPlainString();
        return znak + (s.contains(".") ? s : s + ".0");
    }

    // f"{v:.1f}" z Pythona (dokładne zaokrąglenie wartości binarnej)
    static String jedenMiejsce(double v) {
        String s = new BigDecimal(v).setScale(1, RoundingMode.HALF_EVEN).toPlainString();
        return (v < 0 && !s.startsWith("-")) ? "-" + s : s;
    }

    // round(v, 1) z Pythona
    static double zaokraglij1(double v) {
        return Double.parseDouble(jedenMiejsce(v));
    }

    public static void main(String[] args) throws Exception {
        var los = new Random();
        double x = 1, y = 1;
        var punkty = new ArrayList<double[]>();

        // a) 5000 iteracji, za każdym razem losowy układ równań
        for (int i = 0; i < 5000; i++) {
            double nowyX, nowyY;
            if (los.nextInt(2) == 0) {
                nowyX = -0.4 * x - 1;
                nowyY = -0.4 * y + 0.1;
            } else {
                nowyX = 0.76 * x - 0.4 * y;
                nowyY = 0.4 * x + 0.76 * y;
            }
            x = nowyX;
            y = nowyY;
            punkty.add(new double[]{x, y});
        }

        var smok = punkty.subList(100, punkty.size()); // pomijamy 100 pierwszych iteracji
        double sumaX = 0, sumaY = 0;
        double minX = Double.MAX_VALUE, maxX = -Double.MAX_VALUE, minY = Double.MAX_VALUE, maxY = -Double.MAX_VALUE;
        for (double[] p : smok) {
            sumaX += p[0];
            sumaY += p[1];
            minX = Math.min(minX, p[0]);
            maxX = Math.max(maxX, p[0]);
            minY = Math.min(minY, p[1]);
            maxY = Math.max(maxY, p[1]);
        }
        String srodekX = jakPython(zaokraglij1(sumaX / smok.size()));
        String srodekY = jakPython(zaokraglij1(sumaY / smok.size()));
        String xmin = jakPython(zaokraglij1(minX)), xmax = jakPython(zaokraglij1(maxX));

def slowo_c(a, b):
    if b in a:
        return a

    k2 = nakladka(b, a)  # najpierw B, potem A
    if k1 >= k2:

import java.io.PrintWriter;
import java.nio.file.*;

public class Zadanie {
    static int nakladka(String a, String b) {
        // długość najdłuższego sufiksu słowa a, który jest prefiksem słowa b
        for (int k = Math.min(a.length(), b.length()); k > 0; k--)
            if (a.substring(a.length() - k).equals(b.substring(0, k))) return k;
        return 0;
    }

    static String slowoC(String a, String b) {
        if (a.contains(b)) return a;
        int k1 = nakladka(a, b); // najpierw A, potem B
        int k2 = nakladka(b, a); // najpierw B, potem A
        if (k1 >= k2) return a + b.substring(k1);
        return b + a.substring(k2);
    }

    public static void main(String[] args) throws Exception {
        try (var wynik = new PrintWriter("slowa.txt")) {
            for (String linia : Files.readAllLines(Path.of("dane.txt"))) {
                if (linia.isBlank()) continue;
                String[] slowa = linia.trim().split("\\s+");
                wynik.println(slowoC(slowa[0], slowa[1]));
            }
        }
    }
}
