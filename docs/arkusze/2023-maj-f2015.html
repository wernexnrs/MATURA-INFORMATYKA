def opis(A):
    B = []
    i = 0
    while i < len(A):
        p = 1
        while i + p < len(A) and A[i + p] == A[i]:
            p += 1
        B += [p, A[i]]
        i += p
    return B


# 1.2 - długość opisu: każda zmiana wartości rozpoczyna nowy fragment (2 liczby)
def dlugosc_opisu(A, n):
    w = 2
    for i in range(n - 1):
        if A[i] != A[i + 1]:
            w += 2
    return w


# 1.1
for A in [[3, 3, 3, 2, 2, 1, 1, 1, 6], [2, 2, 2, 2, 2, 2]]:

with open("slowa.txt") as file:
    slowa = [linia.strip() for linia in file if linia.strip()]

wzor = "wakacje"
wyniki = []
for slowo in slowa:
    j = 0           # która litera słowa "wakacje" jest teraz szukana
    pelne = 0       # liczba ułożonych pełnych słów "wakacje"
    for litera in slowo:
        if litera == wzor[j]:
            j += 1
            if j == len(wzor):
                pelne += 1
                j = 0
    wyniki.append(len(slowo) - len(wzor) * pelne)

with open("wyniki4_3.txt", "w") as wynik:
    print(*wyniki, file=wynik)
