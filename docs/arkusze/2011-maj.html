# drzewo wywołań - każde wywołanie wypisane z wcięciem równym głębokości rekurencji
def drzewo(n, glebokosc=0):
    print("    " * glebokosc + f"sklej({n})")
    if n == 1:
        return
    if n % 2 == 0:
        drzewo(n // 2, glebokosc + 1)
    else:
        drzewo((n - 1) // 2, glebokosc + 1)
        drzewo((n + 1) // 2, glebokosc + 1)


drzewo(7)

Odpowiedź: sklej(7) → sklej(3), sklej(4); sklej(3) → sklej(1), sklej(2); sklej(4) → sklej(2); każde sklej(2) → sklej(1)

def sklej(n):
    if n == 1:
        return 0
    if n % 2 == 0:
        return n - 1 + 2 * sklej(n // 2)
    return n - 1 + sklej((n - 1) // 2) + sklej((n + 1) // 2)


for n in range(1, 7):
    print(n, sklej(n))

Odpowiedź: sklej(3) = 3, sklej(4) = 5, sklej(5) = 8, sklej(6) = 11

def wypelnij(n):
    # s[1..n], s[i] = sklej(i), bez rekurencji
    s = [0] * (n + 1)
    s[1] = 0
    for i in range(2, n + 1):
        if i % 2 == 0:
            s[i] = i - 1 + 2 * s[i // 2]
        else:
            s[i] = i - 1 + s[(i - 1) // 2] + s[(i + 1) // 2]
    return s


s = wypelnij(10)
print(s[1:])

# sprawdzenie: s[n] to długość sklejonych zapisów binarnych liczb od 1 do n-1
for n in range(1, 11):
    napis = ""
    for i in range(1, n):
        napis += bin(i)[2:]
    print(n, s[n], len(napis))

Odpowiedź: s[1] := 0; dla i = 2..n: jeśli i parzyste, to s[i] := i − 1 + 2·s[i div 2], w przeciwnym razie s[i] := i − 1 + s[(i − 1) div 2] + s[(i + 1) div 2]

        wp = wp + wb  # p := p * b
    wb = wb + wb      # b := b * b
    n = n // 2

def liczba_mnozen(n):
    mnozenia = 0
    while n > 0:
        if n % 2 != 0:
            mnozenia += 1  # p := p * b
        mnozenia += 1      # b := b * b
        n = n // 2
    return mnozenia


# b)
for n in range(2, 8):
    print(n, liczba_mnozen(n))

# c) dla n będącego potęgą dwójki
for k in range(6):
    print(2 ** k, liczba_mnozen(2 ** k), 2 + k)

Odpowiedź: b) 2 → 3, 3 → 4, 4 → 4, 5 → 5, 6 → 5, 7 → 6; c) f(n) = 2 + log₂n

from datetime import date, timedelta


def poranki(trawa, samochody, dni):
    # objętość trawy rano w kolejnych dniach, indeks 0 to 1 kwietnia
    rano = [trawa]
    for i in range(dni):
        trawa = max(0, trawa - samochody * 15)
        trawa += 600
        trawa -= trawa * 3 // 100
        rano.append(trawa)
    return rano


start = date(2011, 4, 1)
rano = poranki(10000, 30, 213)  # do 31 października

# 1.
zdanie1 = False
for i in range(1, len(rano)):
    if rano[i] > rano[i - 1]:
        zdanie1 = True
        print("1.", i, start + timedelta(days=i))
        break
if not zdanie1:
    print("1. fałsz")

# 2.
for i in range(1, len(rano)):
    if rano[i] == rano[i - 1]:
        print("2.", i, start + timedelta(days=i))
        break

# 3. 12 kwietnia ma indeks 11
samochody = 1
while True:
    rano = poranki(10000, samochody, 11)
    pierwszy = None
    for i in range(len(rano)):
        if rano[i] <= samochody * 15:
            pierwszy = i
            break
    if pierwszy == 11:
        print("3.", samochody)
        break
    samochody += 1

Odpowiedź: 1. fałsz; 2. prawda – po 185 dniach, 2011-10-03; 3. prawda – 83 samochody
