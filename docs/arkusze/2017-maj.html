def najwieksze_pole(A, p):
    # pole a*b nie dzieli się przez liczbę pierwszą p <=> ani a, ani b nie dzieli się przez p,
    # więc wystarczy znaleźć dwie największe (różne) liczby z A niepodzielne przez p
    max1 = max2 = 0
    for a in A:
        if a % p != 0:
            if a > max1:
                max2 = max1
                max1 = a
            elif a > max2:
                max2 = a
    return max1 * max2  # 0, gdy nie ma dwóch takich liczb


# 1.1
print(najwieksze_pole([15, 12, 10, 6, 5, 1], 5))
print(najwieksze_pole([6, 28, 7, 12, 10, 14, 5, 9, 4, 8, 18], 7))
print(najwieksze_pole([4, 34, 16, 8, 6, 22, 14, 12, 2, 7], 2))

Odpowiedź: 1.1: 72 (12 · 6), 216 (18 · 12), 0. 1.2: algorytm liniowy jak w funkcji najwieksze_pole.

wywolania = 0


def licz(x):
    global wywolania
    wywolania += 1
    if x == 1:
        return 1
    w = licz(x // 2)
    if x % 2 == 1:
        return w + 1
    return w - 1


# 2.1
for x in [11, 13, 21, 32]:
    print(x, licz(x))

# 2.2 - najmniejsze x wymagające dokładnie k wywołań
for k in range(1, 8):
    x = 1
    while True:
        wywolania = 0
        licz(x)
        if wywolania == k:
            break
        x += 1
    print(k, x, 2 ** (k - 1))

# 2.3
x = 101
while licz(x) != 0:
    x += 1
print(x)

Odpowiedź: 2.1: licz(11) = 2, licz(13) = 2, licz(21) = 1, licz(32) = -4; 2.2: B) x = 2^(k-1); 2.3: 135

obraz = []
with open("dane.txt") as file:
    for line in file:
        obraz.append(list(map(int, line.split())))

wiersze = len(obraz)
kolumny = len(obraz[0])
licznik = 0

for i in range(wiersze):
    for j in range(kolumny):
        kontrastujacy = False
        for di, dj in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            ni, nj = i + di, j + dj
            if 0 <= ni < wiersze and 0 <= nj < kolumny:
                if abs(obraz[i][j] - obraz[ni][nj]) > 128:
                    kontrastujacy = True
        if kontrastujacy:
            licznik += 1

print(licznik)
