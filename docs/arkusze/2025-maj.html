Treść zadania

Dana jest rekurencyjna funkcja przestaw dla nieujemnej liczby całkowitej n (mod — reszta, div — część całkowita z dzielenia):

Kod
przestaw(n):
    r ← n mod 100
    a ← r div 10
    b ← r mod 10
    n ← n div 100
    jeżeli n > 0
        w ← a + 10 * b + 100 * przestaw(n)
    w przeciwnym razie
        jeżeli a > 0
            w ← a + 10 * b
        w przeciwnym razie
            w ← b
    wynikiem jest w
  • 1.1 — Uzupełnij tabelę: dla n = 43657688, 154005710 i 998877665544321 podaj wynik przestaw(n) oraz łączną liczbę wywołań funkcji (z pierwszym włącznie). Przykład: przestaw(316498) = 134689, 3 wywołania.
  • 1.2 — Oceń (P/F), czy liczba wywołań funkcji dla liczby k-cyfrowej (k > 0) jest równa: 1) k/2; 2) (k + 1) div 2; 3) k/2 dla parzystego k oraz (k + 1)/2 dla nieparzystego k; 4) k/2 + 1.
Python
wywolania = 0


def przestaw(n):
    global wywolania
    wywolania += 1
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        w = a + 10 * b + 100 * przestaw(n)
    else:
        if a > 0:
            w = a + 10 * b
        else:
            w = b
    return w

          wzor3 == wywolania, wzor4 == wywolania)

def przestaw2(n):
    w = 0
    p = 1
    while n > 0:
        r = n % 100
        a = r // 10
        b = r % 10
        if n > 9:
            w = p * a + 10 * p * b + w
        else:
            w = p * b + w
        n = n // 100
        p = p * 100
    return w

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    static final String SYMBOL = "o+*";

    static long naLiczbe(String napis) {
        long liczba = 0;
        for (char znak : napis.toCharArray()) {
            liczba = liczba * 3 + SYMBOL.indexOf(znak);  // "o" -> 0, "+" -> 1, "*" -> 2
        }
        return liczba;
    }

    static String naNapis(long liczba) {
        String napis = "";
        while (liczba > 0) {
            napis = SYMBOL.charAt((int) (liczba % 3)) + napis;
            liczba /= 3;
        }
        return napis;
    }

    public static void main(String[] args) throws Exception {
        List<String> napisy = new ArrayList<>();
        for (String wiersz : Files.readAllLines(Path.of("symbole.txt"))) {
            napisy.add(wiersz.strip());
        }

        // 2.1
        System.out.println("2.1");
        for (String napis : napisy) {
            if (napis.equals(new StringBuilder(napis).reverse().toString())) {
                System.out.println(napis);
            }
        }

        // 2.2
        List<int[]> srodki = new ArrayList<>();
        for (int i = 1; i < napisy.size() - 1; i++) {

            }

Dron leci po łamanej od punktu (0, 0) do punktu (20000, 0), poza startem i lądowaniem zawsze na dodatniej wysokości. Plik dron.txt zawiera 100 wierszy; w każdym są dwie liczby całkowite A i B oddzielone spacją: A (zawsze dodatnie) to przesunięcie w poziomie, a B — w pionie (dodatnie w górę, ujemne w dół, 0 bez zmiany wysokości) względem poprzedniej pozycji. Odpowiedzi zapisz w pliku wyniki3.txt, poprzedzając każdą numerem zadania. Plik dron_przyklad.txt służy do testów.

  • 3.1 — Podaj liczbę par [A, B], dla których NWD(|A|, |B|) jest większy od 1 (przyjmujemy NWD(A, 0) = A).
  • 3.2 — Rozważ punkty, w których dron znajduje się po kolejnych ruchach (np. dla ruchów [3000, 2000], [2000, 9000], [5000, −7000] są to (3000, 2000), (5000, 11000), (10000, 4000)…). a) Podaj, ile z nich leży wewnątrz kwadratu o wierzchołkach (0, 0), (0, 5000), (5000, 5000), (5000, 0) (bez punktów na brzegu). b) Znajdź trzy różne punkty takie, że jeden jest środkiem odcinka o końcach w dwóch pozostałych (jest jedna taka trójka; punkty nie muszą być kolejne). Dla pliku przykładowego: a) 2, b) (14000, 3014), (16000, 2010), (18000, 1006).

public class Zadanie {
    // punkt jako rekord - equals i hashCode działają po wartościach
    record Punkt(int x, int y) {
        @Override
        public String toString() {
            return "(" + x + ", " + y + ")";
        }
    }

    static int gcd(int a, int b) {
        while (b != 0) {
            int t = a % b;
            a = b;
            b = t;
        }
        return a;
    }

    public static void main(String[] args) throws Exception {
        List<int[]> ruchy = new ArrayList<>();
        for (String wiersz : Files.readAllLines(Path.of("dron.txt"))) {
            String[] t = wiersz.trim().split("\\s+");
            ruchy.add(new int[]{Integer.parseInt(t[0]), Integer.parseInt(t[1])});
        }

        // 3.1
        int ile = 0;
        for (int[] r : ruchy) {
            if (gcd(Math.abs(r[0]), Math.abs(r[1])) > 1) {
                ile += 1;
            }
        }
        System.out.println("3.1 " + ile);

        // 3.2
        List<Punkt> punkty = new ArrayList<>();
        int x = 0, y = 0;
        for (int[] r : ruchy) {
            x += r[0];
            y += r[1];
            punkty.add(new Punkt(x, y));
        }

        int wewnatrz = 0;
        for (Punkt p : punkty) {
            if (0 < p.x() && p.x() < 5000 && 0 < p.y() && p.y() < 5000) {
                wewnatrz += 1;
            }
        }
        System.out.println("3.2 a) " + wewnatrz);

        Set<Punkt> zbior = new HashSet<>(punkty);
        for (int i = 0; i < punkty.size(); i++) {
            for (int j = i + 1; j < punkty.size(); j++) {
                Punkt p1 = punkty.get(i), p2 = punkty.get(j);
                if ((p1.x() + p2.x()) % 2 == 0 && (p1.y() + p2.y()) % 2 == 0) {
                    Punkt srodek = new Punkt((p1.x() + p2.x()) / 2, (p1.y() + p2.y()) / 2);
                    if (zbior.contains(srodek) && !srodek.equals(p1)) {
                        System.out.println("3.2 b) " + p1 + " " + srodek + " " + p2);
                    }
                }
            }
        }
    }
