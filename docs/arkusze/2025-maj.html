wywolania = 0


def przestaw(n):
    global wywolania
    wywolania += 1
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        w = a + 10 * b + 100 * przestaw(n)
    else:
        if a > 0:
            w = a + 10 * b
        else:
            w = b
    return w


# 1.1

# 1.2 - liczba wywołań dla liczb k-cyfrowych porównana ze wzorami
for k in range(1, 9):
    wywolania = 0
    przestaw(10 ** (k - 1))   # dowolna liczba k-cyfrowa
    wzor1 = k / 2
    wzor2 = (k + 1) // 2
    wzor3 = k / 2 if k % 2 == 0 else (k + 1) / 2
    wzor4 = (k + 1) / 2
    print(k, wywolania, wzor1 == wywolania, wzor2 == wywolania,
          wzor3 == wywolania, wzor4 == wywolania)

def przestaw2(n):
    w = 0
    p = 1
    while n > 0:
        r = n % 100
        a = r // 10
        b = r % 10
        if n > 9:
            w = p * a + 10 * p * b + w
        else:
            w = p * b + w
        n = n // 100
        p = p * 100
    return w


for n in [316498, 43657688, 154005710, 998877665544321]:
    print(n, przestaw2(n))

def przestaw(n):
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        return a + 10 * b + 100 * przestaw(n)
    if a > 0:
        return a + 10 * b
    return b

cyfra = {"o": 0, "+": 1, "*": 2}
symbol = "o+*"


def na_liczbe(napis):
    liczba = 0
    for znak in napis:
        liczba = liczba * 3 + cyfra[znak]
    return liczba


def na_napis(liczba):
    napis = ""
    while liczba > 0:
        napis = symbol[liczba % 3] + napis
        liczba //= 3
    return napis


najwiekszy = max(napisy, key=na_liczbe)
print("2.3", na_liczbe(najwiekszy), najwiekszy)

suma = sum(na_liczbe(napis) for napis in napisy)
print("2.4", suma, na_napis(suma))

from math import gcd

with open("dron.txt") as file:
    ruchy = [tuple(map(int, wiersz.split())) for wiersz in file]

# 3.1
ile = 0
for a, b in ruchy:
    if gcd(abs(a), abs(b)) > 1:
        ile += 1
print("3.1", ile)

# 3.2
punkty = []
x, y = 0, 0
for a, b in ruchy:
    x += a
    y += b
    punkty.append((x, y))

wewnatrz = 0
for x, y in punkty:
    if 0 < x < 5000 and 0 < y < 5000:
        wewnatrz += 1
print("3.2 a)", wewnatrz)

zbior = set(punkty)
for i in range(len(punkty)):
    for j in range(i + 1, len(punkty)):
        (x1, y1), (x2, y2) = punkty[i], punkty[j]
        if (x1 + x2) % 2 == 0 and (y1 + y2) % 2 == 0:
            srodek = ((x1 + x2) // 2, (y1 + y2) // 2)
            if srodek in zbior and srodek != punkty[i]:
                print("3.2 b)", punkty[i], srodek, punkty[j])
