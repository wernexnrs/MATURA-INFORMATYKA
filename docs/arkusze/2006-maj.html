def ss_wprost(n):
    # każdą silnię liczymy osobno i liczymy mnożenia
    mnozenia = 0
    suma = 0
    for k in range(1, n + 1):
        silnia = 1
        for i in range(2, k + 1):
            silnia *= i
            mnozenia += 1
        suma += silnia
    return suma, mnozenia


for n in [3, 4, 10]:
    print(n, *ss_wprost(n), n * (n - 1) // 2)

Odpowiedź: ss(3): 0+1+2 = 3 mnożenia, ss(4): 0+1+2+3 = 6 mnożeń, ss(n): n(n-1)/2 mnożeń

def ss(n):
    # Dane: n - liczba naturalna > 0
    # Wynik: ss = 1! + 2! + ... + n!
    # ss(n) = 1+2*(1+3*(1+...(n-2)*(1+(n-1)*(1+n))...))
    if n == 1:
        return 1
    wynik = 1 + n
    i = n - 1
    while i > 1:
        wynik = 1 + i * wynik
        i -= 1
    return wynik


for n in range(1, 8):
    print(n, ss(n))

Odpowiedź: ss(5) = 1+2*(1+3*(1+4*(1+5))) (3 mnożenia); algorytm: ss := 1 + n, i := n − 1; dopóki i > 1: ss := 1 + i·ss, i := i − 1

def niepodzielne(N, A):
    # Dane: N > 1, tablica A liczb z przedziału [2, N]
    # Wynik: T[i] = 0 wtedy i tylko wtedy, gdy i nie dzieli się przez żadną z liczb A
    T = [0] * (N + 1)
    for liczba in A:
        j = liczba
        while j <= N:
            T[j] = 1
            j += liczba
    return [i for i in range(2, N + 1) if T[i] == 0]


print(niepodzielne(30, [2, 3]))

Odpowiedź: dla każdej liczby A[k] wykreślamy (T[j] := 1) wszystkie jej wielokrotności j = A[k], 2A[k], … ≤ N; liczby z T[i] = 0 nie są podzielne przez żadną z A[1..M].

def czy_pierwsza(L):
    j = 2
    pierwsza = True
    while j * j <= L and pierwsza:
        pierwsza = L % j != 0
        j += 1
    return "Tak" if pierwsza else "Nie"


for L in [2, 9, 13, 97, 100]:
    print(L, czy_pierwsza(L))

Odpowiedź: sprawdzamy dzielniki j = 2, 3, … dopóki j·j ≤ L; złożoność O(√L).

import math


def f(x):
    return -x * x / 50


def g(x):
    return 1 + x * x / 100 - x / 200


# dla x >= 0 funkcja f maleje, a g rośnie dla x >= 0.25,
# więc na przedziale [x0, x0+100] największe f jest w x0, a najmniejsze g w max(x0, 0.25)
znalezione = False
C = 100
while not znalezione:
    for x0 in range(0, C - 100 + 1):
        dol = math.ceil(f(x0))
        gora = math.floor(g(max(x0, 0.25)))
        if gora - dol >= 26:
            znalezione = True
            print("C =", C)
            for y0 in range(dol, gora - 26 + 1):
                print((x0, y0), (x0 + 100, y0), (x0, y0 + 26), (x0 + 100, y0 + 26))
            break
    if not znalezione:
        C += 1

Odpowiedź: C = 130, np. prostokąt (30, −17), (130, −17), (30, 9), (130, 9)

def czy_palindrom(slowo):
    i = 0
    j = len(slowo) - 1
    while i < j:
        if slowo[i] != slowo[j]:

        i += 1
        j -= 1
    return True

def oprocentowanie(kwota):
    if kwota < 10000:
        return 6
    if kwota < 20000:
        return 7
    if kwota < 30000:
        return 8
    if kwota < 40000:
        return 9
    if kwota < 50000:
        return 10
    return 11


suma_poczatek = 0
suma_koniec = 0
najwieksza = 0
with open("lokaty.txt") as file:
    for linia in file:
        kwota = float(linia.strip().replace(",", "."))
        koniec = round(kwota * (100 + oprocentowanie(kwota)) / 100, 2)
        suma_poczatek += kwota
        suma_koniec += koniec
        najwieksza = max(najwieksza, koniec)

print("%.2f" % suma_poczatek)
print("%.2f" % suma_koniec)
print("%.2f" % najwieksza)

import datetime

kursy = []
with open("kursy.txt") as file:
    for linia in file:
        kursy.append(float(linia.strip().replace(",", ".")))

# sposób I - zmiana waluty pierwszego dnia każdego miesiąca
pierwsze_dni = []
for miesiac in range(1, 13):
    dzien = datetime.date(2005, miesiac, 1) - datetime.date(2005, 1, 1)
    pierwsze_dni.append(dzien.days)

kwota = 20000.00
w_euro = False
for dzien in pierwsze_dni:
    if w_euro:
        kwota = round(kwota * kursy[dzien], 2)
    else:
        kwota = round(kwota / kursy[dzien], 2)
    w_euro = not w_euro
if w_euro:
    kwota = round(kwota * kursy[364], 2)
print("Sposób I:", "%.2f" % kwota)

# sposób II
kwota = round(20000.00 / kursy[0], 2)
w_euro = True
for dzien in range(1, 365):
    if w_euro:
        if kursy[dzien] < kursy[dzien - 1]:
            kwota = round(kwota * kursy[dzien], 2)
            w_euro = False
    else:
        kwota = round(kwota / kursy[dzien], 2)
        w_euro = True
if w_euro:
    kwota = round(kwota * kursy[364], 2)
print("Sposób II:", "%.2f" % kwota)
