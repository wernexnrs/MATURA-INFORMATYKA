Treść zadania

Dla dodatnich liczb całkowitych m, n funkcja A jest zdefiniowana rekurencyjnie:

  • A(m, n) = m, gdy n = 1;
  • A(m, n) = A(2·m, n/2), gdy n > 1 i n jest parzyste;
  • A(m, n) = 2·A(m, (n − 1)/2) + m, gdy n > 1 i n jest nieparzyste.

Przykład: obliczenie A(3, 9) wymaga trzech wywołań rekurencyjnych A(3, 4), A(6, 2), A(12, 1), bo A(3, 9) = 2·A(3, 4) + 3 = 2·A(6, 2) + 3 = 2·A(12, 1) + 3 = 27.

  • 1.1 — Dla par (m, n) = (32, 17), (2⁵, 2⁵) i (10, 15) podaj liczbę wywołań rekurencyjnych funkcji A i wypisz te wywołania z argumentami; dla (1, 2¹⁰⁰ + 1) podaj tylko liczbę wywołań rekurencyjnych.
  • 1.2 — Podaj wartości A(1, 777) oraz A(2·10⁹, 256·10⁹) (wynik można zapisać z użyciem mnożenia, potęgowania i dodawania).
  • 1.3 — Dla n = 2ᵏ oraz n = 2ᵏ − 1 (k — liczba całkowita większa od 2) podaj liczbę wywołań rekurencyjnych funkcji A oraz wyrażenie opisujące drugi argument w i-tym wywołaniu rekurencyjnym (pierwszy argument nie ma znaczenia). Przykład dla n = 8: 3 wywołania, drugi argument w i-tym to 8/2ⁱ, czyli 2³⁻ⁱ.

import java.math.BigInteger;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    // n = 2^100 + 1 oraz iloczyn 512 * 10^18 nie mieszczą się w long - używamy BigInteger
    static final BigInteger DWA = BigInteger.TWO;
    static List<BigInteger[]> wywolania = new ArrayList<>();   // argumenty kolejnych wywołań rekurencyjnych

    static BigInteger A(BigInteger m, BigInteger n, boolean pierwsze) {
        if (!pierwsze) {
            wywolania.add(new BigInteger[]{m, n});
        }
        if (n.equals(BigInteger.ONE)) {
            return m;
        }
        if (n.mod(DWA).signum() == 0) {
            return A(m.multiply(DWA), n.divide(DWA), false);
        }
        return A(m, n.subtract(BigInteger.ONE).divide(DWA), false).multiply(DWA).add(m);
    }

    static BigInteger A(long m, long n) {
        return A(BigInteger.valueOf(m), BigInteger.valueOf(n), true);
    }

    public static void main(String[] args) throws Exception {
        // 1.1
        BigInteger[][] dane = {
            {BigInteger.valueOf(32), BigInteger.valueOf(17)},
            {BigInteger.valueOf(32), BigInteger.valueOf(32)},
            {BigInteger.valueOf(10), BigInteger.valueOf(15)},
            {BigInteger.ONE, DWA.pow(100).add(BigInteger.ONE)}
        };
        for (BigInteger[] para : dane) {
            wywolania.clear();
            A(para[0], para[1], true);
            // lista par wypisana jak w Pythonie: [(32, 8), (64, 4)]
            String lista = "";
            if (para[1].compareTo(BigInteger.valueOf(1000)) < 0) {
                List<String> elementy = new ArrayList<>();
                for (BigInteger[] w : wywolania) {
                    elementy.add("(" + w[0] + ", " + w[1] + ")");
                }
                lista = "[" + String.join(", ", elementy) + "]";
            }
            System.out.println(para[0] + " " + para[1] + " " + wywolania.size() + " " + lista);
        }

        // 1.2 - funkcja oblicza iloczyn m * n
        System.out.println(A(1, 777) + " " + A(2_000_000_000L, 256_000_000_000L));

        // 1.3 - drugie argumenty kolejnych wywołań dla n = 2^k oraz n = 2^k - 1
        for (int k = 3; k < 7; k++) {
            for (int n : new int[]{1 << k, (1 << k) - 1}) {
                wywolania.clear();
                A(1, n);
                List<BigInteger> drugie = new ArrayList<>();
                for (BigInteger[] w : wywolania) {
                    drugie.add(w[1]);
                }
                System.out.println(k + " " + n + " " + wywolania.size() + " " + drugie);
            }
        }
    }
}

Wolno używać tylko zmiennych przechowujących pojedyncze liczby całkowite, operatorów arytmetycznych (w tym dzielenia całkowitego i reszty), logicznych, porównań, przypisań i własnych funkcji — bez funkcji wbudowanych, tablic/list i konwersji liczba ↔ napis.

def przeniesienia(a, b):
    w_pamieci = 0
    p = 0
    while a > 0:
        suma = a % 10 + b % 10 + w_pamieci
        if suma >= 10:
            w_pamieci = 1
            p += 1
        else:
            w_pamieci = 0
        a //= 10
        b //= 10
    return p

int f(const std::string& s) {
    int suma = 0;
    for (char c : s) {
        suma += c;  // kod ASCII znaku
    }
    return suma;
}

int main() {
    std::vector<std::pair<std::string, std::string>> pary;
    std::ifstream plik("pary.txt");
    std::string s1, s2;
    while (plik >> s1 >> s2) {
        pary.push_back({s1, s2});

    // pierwsza para z najmniejszą sumą i pierwsza z największą różnicą (jak min/max w Pythonie)
    auto a = pary[0];
    auto b = pary[0];
    for (const auto& para : pary) {
        if (f(para.first) + f(para.second) < f(a.first) + f(a.second)) {
            a = para;
        }
        if (std::abs(f(para.first) - f(para.second)) > std::abs(f(b.first) - f(b.second))) {
            b = para;
        }

Wspólna liczba wystąpień litery x w słowach s1, s2 to W(x, s1, s2) = min(d(x, s1), d(x, s2)), gdzie d(x, s) to liczba wystąpień x w s (np. dla adabbcdd i aadabbbccdc: a: 2, b: 2, c: 1, d: 2, suma 7).

  • a) Podaj parę (lub wszystkie pary, jeśli jest ich więcej) z jednego wiersza, w której litera b ma największą wspólną liczbę wystąpień (dla pliku przykładowego: bbabac bbac).
  • b) Podaj parę z jednego wiersza, dla której suma wspólnych wystąpień wszystkich liter jest największa (jest jedna taka; dla pliku przykładowego: ccccdcedddddecde ccdcdddecc).

def W(x, s1, s2):
    return min(s1.count(x), s2.count(x))


def suma_wspolnych(para):
    s1, s2 = para
    return sum(W(x, s1, s2) for x in "abcdefghijklmnopqrstuvwxyz")

#include <algorithm>
#include <fstream>

using Para = std::pair<std::string, std::string>;

int W(char x, const std::string& s1, const std::string& s2) {
    return std::min(std::ranges::count(s1, x), std::ranges::count(s2, x));
}

int suma_wspolnych(const Para& para) {
    int suma = 0;
    for (char x = 'a'; x <= 'z'; x++) {
        suma += W(x, para.first, para.second);
    }
    return suma;
}

int main() {
    std::vector<Para> pary;
    std::ifstream plik("pary.txt");
    std::string s1, s2;
    while (plik >> s1 >> s2) {
        pary.push_back({s1, s2});
    }

    int najwiecej_b = 0;
    for (const auto& [s1, s2] : pary) {
        najwiecej_b = std::max(najwiecej_b, W('b', s1, s2));
    }
    Para b = pary[0];  // pierwsza para z największą sumą (jak max w Pythonie)
    for (const Para& para : pary) {
        if (suma_wspolnych(para) > suma_wspolnych(b)) {
            b = para;
        }
    }

    std::ofstream wynik("wyniki4.txt", std::ios::app);  // dopisywanie na koniec pliku
    std::println(wynik, "4.2");
    std::println(wynik, "a)");

        if (W('b', s1, s2) == najwiecej_b) {
            std::println(wynik, "{} {}", s1, s2);
        }

}

import java.io.FileWriter;
import java.io.PrintWriter;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    static int ile(String s, char x) {
        int wynik = 0;
        for (char c : s.toCharArray()) {
            if (c == x) {
                wynik++;
            }
        }
        return wynik;
    }

    static int W(char x, String s1, String s2) {
        return Math.min(ile(s1, x), ile(s2, x));
    }

    static int sumaWspolnych(String[] para) {
        int suma = 0;
        for (char x = 'a'; x <= 'z'; x++) {
            suma += W(x, para[0], para[1]);
        }
        return suma;
    }

    public static void main(String[] args) throws Exception {
        List<String[]> pary = new ArrayList<>();
        for (String linia : Files.readAllLines(Path.of("pary.txt"))) {
            if (!linia.isBlank()) {
                pary.add(linia.trim().split("\\s+"));
            }
        }

        int najwiecejB = 0;
        for (String[] para : pary) {
            najwiecejB = Math.max(najwiecejB, W('b', para[0], para[1]));
        }
        String[] b = pary.get(0);  // pierwsza para z największą sumą (jak max w Pythonie)
        for (String[] para : pary) {
            if (sumaWspolnych(para) > sumaWspolnych(b)) {
                b = para;
            }
        }

        // FileWriter(..., true) - dopisywanie na koniec pliku
        try (PrintWriter wynik = new PrintWriter(new FileWriter("wyniki4.txt", true))) {
            wynik.println("4.2");
            wynik.println("a)");
            for (String[] para : pary) {
                if (W('b', para[0], para[1]) == najwiecejB) {
                    wynik.println(para[0] + " " + para[1]);
                }
            }
            wynik.println("b) " + b[0] + " " + b[1]);
        }
    }
}

Prefiksosufiks pary słów s1, s2 to słowo, które jest początkiem s1 i końcem s2 albo początkiem s2 i końcem s1. Wypisz wszystkie pary z pliku, których najdłuższy prefiksosufiks ma co najmniej 5 liter, a przy każdej podaj długość tego najdłuższego prefiksosufiksu. Przykład: dla aabbbca caacaab najdłuższy prefiksosufiks to aab (3 litery) — para nie spełnia warunku; dla abbaabaa baabaabba jest to baabaa (6) — spełnia; dla pliku przykładowego: ececdddeed dddeedd 6.

Python
pary = []
with open("pary.txt") as file:
    for linia in file:
        if linia.strip():
            pary.append(linia.split())


def prefiksosufiks(s1, s2):
    najdluzszy = 0
    for d in range(1, min(len(s1), len(s2)) + 1):
        if s1[:d] == s2[-d:] or s2[:d] == s1[-d:]:
            najdluzszy = d
    return najdluzszy


with open("wyniki4.txt", "a") as wynik:
    print("4.3", file=wynik)
    for s1, s2 in pary:
        d = prefiksosufiks(s1, s2)
        if d >= 5:
            print(s1, s2, d, file=wynik)

#include <algorithm>
#include <fstream>
#include <print>
#include <string>

int prefiksosufiks(const std::string& s1, const std::string& s2) {
    int najdluzszy = 0;
    for (size_t d = 1; d <= std::min(s1.size(), s2.size()); d++) {
        // prefiks jednego słowa równy sufiksowi drugiego
        if (s1.substr(0, d) == s2.substr(s2.size() - d) || s2.substr(0, d) == s1.substr(s1.size() - d)) {
            najdluzszy = d;
        }
    }
    return najdluzszy;
}

int main() {
    std::ifstream plik("pary.txt");
    std::ofstream wynik("wyniki4.txt", std::ios::app);  // dopisywanie na koniec pliku
    std::println(wynik, "4.3");
    std::string s1, s2;
    while (plik >> s1 >> s2) {
        int d = prefiksosufiks(s1, s2);
        if (d >= 5) {
            std::println(wynik, "{} {} {}", s1, s2, d);
        }
    }
}
Java 25
import java.io.FileWriter;
import java.io.PrintWriter;
import java.nio.file.Files;
import java.nio.file.Path;

public class Zadanie {
    static int prefiksosufiks(String s1, String s2) {
        int najdluzszy = 0;
        for (int d = 1; d <= Math.min(s1.length(), s2.length()); d++) {
            // prefiks jednego słowa równy sufiksowi drugiego
            if (s2.endsWith(s1.substring(0, d)) || s1.endsWith(s2.substring(0, d))) {
                najdluzszy = d;
            }
        }
        return najdluzszy;
    }

    public static void main(String[] args) throws Exception {
        // FileWriter(..., true) - dopisywanie na koniec pliku
        try (PrintWriter wynik = new PrintWriter(new FileWriter("wyniki4.txt", true))) {
            wynik.println("4.3");
            for (String linia : Files.readAllLines(Path.of("pary.txt"))) {
                if (linia.isBlank()) {
                    continue;
                }
                String[] s = linia.trim().split("\\s+");
                int d = prefiksosufiks(s[0], s[1]);
                if (d >= 5) {
                    wynik.println(s[0] + " " + s[1] + " " + d);
                }
            }
        }
    }
}
