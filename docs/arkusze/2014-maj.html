# C - czarny koralik, B - bialy koralik
def korale(n):
    if n == 1:
        return "C"
    if n % 2 == 0:
        return korale(n // 2) + "B"
    return korale((n - 1) // 2) + "C"


for n in [1, 2, 3, 4, 7, 8, 15, 16]:
    print(n, korale(n))

Odpowiedź: 1: C, 2: CB, 3: CC, 4: CBB, 7: CCC, 8: CBBB, 15: CCCC, 16: CBBBB (czarny = 1, biały = 0 — zapis binarny liczby n)

def korale_bis(n):
    sznurek = ""
    while n > 0:
        if n % 2 == 0:
            sznurek = "B" + sznurek
        else:
            sznurek = "C" + sznurek
        n = n // 2
    return sznurek


for n in [1, 2, 3, 4, 7, 8, 15, 16]:
    print(n, korale_bis(n))

def bisekcja(f, a, b, d):
    x = (a + b) / 2
    while b - a >= d:
        if f(x) == 0:
            return x
        if f(a) * f(x) < 0:
            b = x
        else:
            a = x
        x = (a + b) / 2
    return x


print(bisekcja(lambda x: x ** 3 - x - 2, 0, 2, 0.0001))

def algorytm(n):
    if n == 1:
        return 1
    suma = 1 + n
    i = n - 1
    while i > 1:
        suma = 1 + i * suma
        i -= 1
    return suma


print(algorytm(4), algorytm(6))

Odpowiedź: n = 4: 33, n = 6: 873

dzien_200 = None
dzien_1500 = None

for dzien in range(1, 181):
    # rano - co 30 dni zakup kur
    koszt_kur = 0
    if dzien % 30 == 0:
        nowe = int(kury * 0.2)
        kury += nowe
        koszt_kur = nowe * 18
        if kury == 200 and dzien_200 is None:
            dzien_200 = (dzien, "rano")
    # posilek
    koszt_paszy = kury * 0.2 * 1.9
    pasza += koszt_paszy
    # poludnie - sprzedaz jajek (bez niedziel)
    przychod = 0
    if dzien % 7 != 0:
        przychod = kury * 0.9
    realny_zysk += przychod - koszt_paszy - koszt_kur
    if realny_zysk > 1500 and dzien_1500 is None:
        dzien_1500 = dzien
    # po zmroku - lis
    if dzien % 2 == 1:
        kury -= 2
        if kury == 200 and dzien_200 is None:
            dzien_200 = (dzien, "po zmroku")

print("a)", dzien_200)
print("b)", round(pasza, 2))
print("c)", dzien_1500, round(realny_zysk, 2))

# dane do wykresu liniowego: dzien, przychod ze sprzedazy jaj, koszty (pasza + zakup kur)
kury = 200

with open("wykres_4d.txt", "w") as wynik:
    wynik.write("dzien;przychod;koszty\n")
    for dzien in range(1, 181):
        koszt_kur = 0
        if dzien % 30 == 0:
            nowe = int(kury * 0.2)
            kury += nowe
            koszt_kur = nowe * 18
        koszty = kury * 0.2 * 1.9 + koszt_kur
        przychod = kury * 0.9 if dzien % 7 != 0 else 0
        wynik.write(f"{dzien};{przychod:.2f};{koszty:.2f}\n")
        if dzien % 2 == 1:
            kury -= 2

def czy_pierwsza(x):
    if x < 2:
        return False
    i = 2
    while i * i <= x:
        if x % i == 0:
            return False
        i += 1
    return True


licznik = 0
with open("NAPIS.TXT") as file:
    for line in file:
        napis = line.strip()
        suma = 0
        for znak in napis:
            suma += ord(znak)
        if czy_pierwsza(suma):
            licznik += 1

print(licznik)
