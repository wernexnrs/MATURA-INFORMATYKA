wywolania = 0


def przestaw(n):
    global wywolania
    wywolania += 1
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        w = a + 10 * b + 100 * przestaw(n)
    elif a > 0:
        w = a + 10 * b
    else:
        w = b
    return w


# 1.1
for n in [43657688, 154005710, 998877665544321]:
    wywolania = 0
    print(n, przestaw(n), wywolania)

# 1.2 - liczba wywołań dla liczby k-cyfrowej porównana z podanymi wzorami
for k in range(1, 9):
    wywolania = 0
    przestaw(10 ** (k - 1))
    print(k, wywolania, k / 2, (k + 1) // 2, k / 2 if k % 2 == 0 else (k + 1) / 2, (k + 1) / 2)

def falista(n):
    baza = n % 100
    f = 0
    pot = 1
    while n > 0:
        if n >= 10:                 # zostały co najmniej dwie cyfry - wstawiamy całą bazę
            f = f + baza * pot
        else:                       # została jedna cyfra - wstawiamy cyfrę jedności bazy
            f = f + (baza % 10) * pot
        n = n // 100
        pot = pot * 100
    return f


# 2.1
for n in [326, 414141, 7732, 21289]:
    print(n, falista(n))

for napis in napisy:
    rozne = 0
    for i in range(len(napis) // 2):
        if napis[i] != napis[len(napis) - 1 - i]:
            rozne += 1
    if rozne == 1:
        ile += 1

srodki = []
for w in range(1, len(napisy) - 1):
    for k in range(1, len(napisy[w]) - 1):
        znak = napisy[w][k]
        kwadrat = True
        for dw in (-1, 0, 1):
            for dk in (-1, 0, 1):
                if napisy[w + dw][k + dk] != znak:
                    kwadrat = False
        if kwadrat:
            srodki.append((w + 1, k + 1))
print("4.3", len(srodki), file=wynik)
for w, k in srodki:
    print(w, k, file=wynik)

# 4.4 i 4.5
cyfry = {"o": 0, "+": 1, "*": 2}
symbole = "o+*"


def na_liczbe(napis):
    liczba = 0
    for znak in napis:
        liczba = liczba * 3 + cyfry[znak]
    return liczba


def na_napis(liczba):
    napis = ""
    while liczba > 0:
        napis = symbole[liczba % 3] + napis
        liczba //= 3
    return napis


najwiekszy = max(napisy, key=na_liczbe)
print("4.4", na_liczbe(najwiekszy), najwiekszy, file=wynik)

suma = 0
for napis in napisy:
    suma += na_liczbe(napis)
print("4.5", suma, na_napis(suma), file=wynik)

wynik.close()
