def pierwsza_parzysta(A):
    # wyszukiwanie binarne - złożoność logarytmiczna
    p = 0
    k = len(A) - 1
    while p < k:
        s = (p + k) // 2
        if A[s] % 2 == 1:
            p = s + 1
        else:
            k = s
    return A[p]


A = [5, 99, 3, 7, 111, 13, 4, 24, 4, 8]
print(pierwsza_parzysta(A))

Odpowiedź: dla przykładu w = 4; złożoność logarytmiczna (wyszukiwanie binarne)

wywolania = 0


def pisz(s, n, k):
    global wywolania
    wywolania += 1
    if len(s) == n:
        print(s)
    else:
        for i in range(k):
            pisz(s + str(i), n, k)


for n, k in [(2, 2), (3, 2), (2, 3)]:
    wywolania = 0
    print(f'pisz("", {n}, {k}):')
    pisz("", n, k)
    # 2.3 - wzór: 1 + k + k^2 + ... + k^n = (k^(n+1) - 1) / (k - 1)
    print("liczba wywołań:", wywolania, "wzór:", (k ** (n + 1) - 1) // (k - 1))

Odpowiedź: 2.1: pisz("10",2,2), pisz("11",2,2) (kolejność wywołań 1, 2, 3, 4, 5, 6, 7 – w głąb); 2.2: pisz("",3,2) wypisuje 000 001 010 011 100 101 110 111, 15 wywołań; pisz("",2,3) wypisuje 00 01 02 10 11 12 20 21 22, 13 wywołań; 2.3: (k^(n+1) − 1)/(k − 1)

from math import gcd

with open("liczby.txt") as file:
    liczby = [int(line) for line in file]

najdluzszy = 0

for i in range(len(liczby)):
    nwd = liczby[i]
    j = i
    # rozszerzamy ciąg zaczynający się od i, dopóki NWD wszystkich liczb jest większy od 1
    while j + 1 < len(liczby) and gcd(nwd, liczby[j + 1]) > 1:
        j += 1
        nwd = gcd(nwd, liczby[j])
    if j - i + 1 > najdluzszy:
        najdluzszy = j - i + 1
        pierwsza = liczby[i]
        dzielnik = nwd

print(pierwsza, najdluzszy, dzielnik)

with open("pogoda.txt") as file:
    file.readline()
    dane = [line.strip().split(";") for line in file]

wielkosc = 0      # dzień pierwszy jest bezchmurny
kategoria = "0"
dni_tej_wielkosci = 1

ile_wielkosci = [0] * 6
zgodne_wielkosci = 0
zgodne_kategorie = 0

for i in range(len(dane)):
    dzien, temperatura, opad, kat_kamera, wlk_kamera = dane[i]
    temperatura = float(temperatura.replace(",", "."))

    if i > 0:
        poprzedni_opad = int(dane[i - 1][2])
        if wielkosc == 0:
            # pojawiają się chmury wielkości 1, kategoria zależy od temperatury w dniu ich powstania
            wielkosc = 1
            kategoria = "C" if temperatura >= 10 else "S"
            dni_tej_wielkosci = 1
        elif wielkosc == 5:
            if poprzedni_opad >= 20:
                wielkosc = 0
                kategoria = "0"
        elif dni_tej_wielkosci == 3:
            wielkosc += 1
            dni_tej_wielkosci = 1
        else:
            dni_tej_wielkosci += 1

    ile_wielkosci[wielkosc] += 1
    if int(dzien) <= 300:
        if int(wlk_kamera) == wielkosc:
            zgodne_wielkosci += 1
        if kat_kamera == kategoria:
            zgodne_kategorie += 1

print("a)")
for w in range(6):
    print(w, ile_wielkosci[w])
print("b)", zgodne_wielkosci)
print("c)", zgodne_kategorie)

def wczytaj(nazwa):
    with open(nazwa, encoding="utf-8") as file:

        return [line.rstrip("\n").split("\t") for line in file if line.strip()]


marki = wczytaj("marki.txt")

marka_perfum = {}
for id_perfum, nazwa, id_marki, rodzina, cena in perfumy:
    marka_perfum[id_perfum] = id_marki

marki_z_paczula = set()
for id_perfum, skladnik in sklad:
    if "paczula" in skladnik:
        marki_z_paczula.add(marka_perfum[id_perfum])

wynik = [nazwa for id_marki, nazwa in marki if id_marki not in marki_z_paczula]
for nazwa in sorted(wynik):
    print(nazwa)

def wczytaj(nazwa):
    with open(nazwa, encoding="utf-8") as file:
        file.readline()
        return [line.rstrip("\n").split("\t") for line in file if line.strip()]

for id_marki, nazwa in marki:
    if nazwa == "Mou De Rosine":
        szukana_marka = id_marki

wynik = []
for id_perfum, nazwa, id_marki, rodzina, cena in perfumy:
    if id_marki == szukana_marka and rodzina == "orientalno-drzewna":

for cena, nazwa in sorted(wynik):
    print(nazwa, cena)
