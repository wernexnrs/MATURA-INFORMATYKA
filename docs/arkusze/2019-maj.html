Zadanie 1 (ulubione liczby). Małgosia lubi liczby nieparzyste, a Jaś parzyste. Na wspólnej kartce najpierw Małgosia zapisała kilka swoich liczb, a potem Jaś dopisał swoje (każde co najmniej jedną). Dane: n > 1 i tablica A[1..n] — najpierw same liczby nieparzyste, potem same parzyste. Wynik: w — pierwsza od lewej liczba parzysta w A.

Przykład: dla A = 5, 99, 3, 7, 111, 13, 4, 24, 4, 8 (n = 10) wynikiem jest w = 4.

  • 1.1 — napisz algorytm znajdujący pierwszą liczbę zapisaną przez Jasia. Maksymalną liczbę punktów daje algorytm o złożoności lepszej niż liniowa. Wolno używać tylko operacji arytmetycznych (w tym div i mod), porównań, instrukcji sterujących, przypisań i własnych funkcji.
  • 1.2 — podaj złożoność czasową swojego algorytmu (kwadratowa, liniowa, logarytmiczna lub inna).
Python
def pierwsza_parzysta(A):
    # wyszukiwanie binarne - złożoność logarytmiczna
    p = 0
    k = len(A) - 1
    while p < k:
        s = (p + k) // 2
        if A[s] % 2 == 1:
            p = s + 1
        else:
            k = s
    return A[p]


A = [5, 99, 3, 7, 111, 13, 4, 24, 4, 8]
print(pierwsza_parzysta(A))

Kod
funkcja pisz(s, n, k)
    jeżeli dł(s) = n

    w przeciwnym razie
        dla i = 0, 1, …, k-1 wykonuj

  • 2.1 — uzupełnij drzewo wywołań funkcji dla pisz("", 2, 2) (korzeń ma dzieci pisz("0",2,2) i pisz("1",2,2)) i ponumeruj węzły w kolejności wywołań (pierwsze wywołanie ma numer 1).

Python
wywolania = 0


def pisz(s, n, k):
    global wywolania
    wywolania += 1
    if len(s) == n:
        print(s)
    else:
        for i in range(k):
            pisz(s + str(i), n, k)


for n, k in [(2, 2), (3, 2), (2, 3)]:
    wywolania = 0
    print(f'pisz("", {n}, {k}):')
    pisz("", n, k)
    # 2.3 - wzór: 1 + k + k^2 + ... + k^n = (k^(n+1) - 1) / (k - 1)
    print("liczba wywołań:", wywolania, "wzór:", (k ** (n + 1) - 1) // (k - 1))

Dane z pliku liczby.txt (500 liczb z zakresu [1, 100 000]).

Silnia: k! = 1·2·…·k, przy czym 0! = 1. Dla liczby można obliczyć sumę silni jej cyfr, np. dla 343: 3! + 4! + 3! = 36. Wypisz, w kolejności z pliku, wszystkie liczby równe sumie silni swoich cyfr.

W pliku przykładowym jest jedna taka liczba: 145 (1! + 4! + 5! = 145).

Python
from math import factorial

with open("liczby.txt") as file:
    for line in file:
        liczba = line.strip()
        suma = 0
        for cyfra in liczba:
            suma += factorial(int(cyfra))
        if suma == int(liczba):
            print(liczba)

Treść zadania

Dane z pliku liczby.txt (500 liczb z zakresu [1, 100 000]).

Znajdź najdłuższy spójny fragment ciągu liczb, których największy wspólny dzielnik jest większy od 1 (jest dokładnie jeden taki fragment). Podaj pierwszą liczbę fragmentu, jego długość i NWD wszystkich jego liczb. Można korzystać z NWD(a, b, c) = NWD(NWD(a, b), c).

Przykłady: dla 3, 7, 4, 6, 10, 2, 5 odpowiedź to 4, 4, 2; dla 5, 70, 28, 42, 98, 1 — 70, 4, 14. Dla pliku przykładowego: 90, 5, 10.

Python
from math import gcd

with open("liczby.txt") as file:
    liczby = [int(line) for line in file]

najdluzszy = 0

for i in range(len(liczby)):
    nwd = liczby[i]
    j = i
    # rozszerzamy ciąg zaczynający się od i, dopóki NWD wszystkich liczb jest większy od 1
    while j + 1 < len(liczby) and gcd(nwd, liczby[j + 1]) > 1:
        j += 1
        nwd = gcd(nwd, liczby[j])
    if j - i + 1 > najdluzszy:
        najdluzszy = j - i + 1
        pierwsza = liczby[i]
        dzielnik = nwd

print(pierwsza, najdluzszy, dzielnik)

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;

public class Zadanie {
    static long gcd(long a, long b) {
        while (b != 0) {
            long r = a % b;
            a = b;
            b = r;
        }
        return a;
    }

    public static void main(String[] args) throws Exception {
        var liczby = new ArrayList<Long>();
        for (String line : Files.readAllLines(Path.of("liczby.txt")))
            if (!line.isBlank())
                liczby.add(Long.parseLong(line.trim()));

        int n = liczby.size();
        int najdluzszy = 0;
        long pierwsza = 0, dzielnik = 0;

        for (int i = 0; i < n; i++) {
            long nwd = liczby.get(i);
            int j = i;
            // rozszerzamy ciąg zaczynający się od i, dopóki NWD wszystkich liczb jest większy od 1
            while (j + 1 < n && gcd(nwd, liczby.get(j + 1)) > 1) {
                j++;
                nwd = gcd(nwd, liczby.get(j));
            }
            if (j - i + 1 > najdluzszy) {
                najdluzszy = j - i + 1;
                pierwsza = liczby.get(i);
                dzielnik = nwd;
            }
        }

        System.out.println(pierwsza + " " + najdluzszy + " " + dzielnik);
    }
}

// dzieli wiersz po średnikach
std::vector<std::string> podziel(const std::string& wiersz) {
    std::vector<std::string> czesci;
    std::stringstream ss(wiersz);
    std::string czesc;
    while (std::getline(ss, czesc, ';'))
        czesci.push_back(czesc);
    return czesci;
}

// liczba z przecinkiem dziesiętnym, np. "12,5" -> 12.5
double na_liczbe(std::string tekst) {
    std::ranges::replace(tekst, ',', '.');
    return std::stod(tekst);
}

int main() {
    int ile = 0;
    std::ifstream plik("pogoda.txt");
    std::string linia;
    std::getline(plik, linia);  // nagłówek
    while (std::getline(plik, linia)) {
        if (!linia.empty() && linia.back() == '\r')
            linia.pop_back();
        if (linia.empty())
            continue;
        auto czesci = podziel(linia);  // dzien, temperatura, opad, kategoria, wielkosc
        double temperatura = na_liczbe(czesci[1]);
        if (temperatura >= 20 && std::stoi(czesci[2]) <= 5)
            ile++;
    }

}

// liczba z przecinkiem dziesiętnym, np. "12,5" -> 12.5
double na_liczbe(std::string tekst) {
    std::ranges::replace(tekst, ',', '.');
    return std::stod(tekst);
}

int main() {
    std::ifstream plik("pogoda.txt");
    std::string linia;
    std::getline(plik, linia);  // nagłówek

    int dlugosc = 0;  // liczba kolejnych dni cieplejszych od dnia poprzedniego

            dlugosc++;

            dlugosc = 0;
