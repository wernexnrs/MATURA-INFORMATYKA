def czy_k_podobne(n, A, B, k):
    # A[1..k] = B[n-k+1..n] (tutaj tablice są indeksowane od 0)
    for i in range(k):
        if A[i] != B[n - k + i]:
            return False
    # A[k+1..n] = B[1..n-k]
    for i in range(n - k):
        if B[i] != A[k + i]:
            return False
    return True


def czy_podobne(n, A, B):
    for k in range(n):
        if czy_k_podobne(n, A, B, k):
            return True
    return False


# 1.1 - wiersze 5. i 7. tabeli
print(czy_k_podobne(5, [1, 2, 3, 4, 5], [3, 4, 5, 1, 2], 2))

def czy_pierwsza(x):
    if x < 2:
        return False
    i = 2
    while i * i <= x:
        if x % i == 0:
            return False
        i += 1
    return True


with open("pary.txt") as file:

        if liczba > 4 and liczba % 2 == 0:
            # najmniejszy pierwszy składnik daje największą różnicę składników
            for p in range(3, liczba // 2 + 1, 2):
                if czy_pierwsza(p) and czy_pierwsza(liczba - p):
                    print(liczba, p, liczba - p)
                    break

with open("statek.txt") as file:
    file.readline()
    dane = [line.rstrip("\n").split("\t") for line in file]

kasa = 500000
zmiana = 0  # zmiana kasy względem stanu początkowego
najmniejsza_zmiana = 0
najwiecej = 0

for i in range(len(dane)):
    data, port, towar, zw, tony, cena = dane[i]
    wartosc = int(tony) * int(cena)
    if zw == "Z":
        kasa -= wartosc  # kapitan płaci za załadunek
        zmiana -= wartosc
    else:
        kasa += wartosc  # kapitan dostaje zapłatę za wyładunek
        zmiana += wartosc
    najmniejsza_zmiana = min(najmniejsza_zmiana, zmiana)
    # stan kasy po wypłynięciu z portu = po ostatniej transakcji danego dnia
    if i == len(dane) - 1 or dane[i + 1][0] != data:
        if kasa > najwiecej:
            najwiecej = kasa
            dzien = data

print("a)", kasa, dzien, najwiecej)
print("b)", -najmniejsza_zmiana)
