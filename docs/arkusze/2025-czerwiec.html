def skroc(x):
    return x // 10


def dopisz(x):
    return x * 10


def ostatnia(x):
    return x % 10


wywolania = 0
dodawania = 0


def f(a, b):
    global wywolania, dodawania
    wywolania += 1
    if b == 0:
        return 0
    k = ostatnia(b)
    w = f(a, skroc(b))
    w = dopisz(w)
    while k > 0:
        w = w + a
        dodawania += 1
        k = k - 1
    return w


# 1.1
for a, b in [(42, 2), (4, 125), (103, 104)]:
    print(a, b, f(a, b))

# 1.2
wywolania = 0
f(987654321, 123456789)
print("1.2", wywolania)

# 1.3
for a, b in [(2024, 1000), (2024, 1234)]:
    dodawania = 0
    f(a, b)
    print("1.3", a, b, dodawania)

def polowy(k):
    dlugosc = 0
    x = k
    while x > 0:
        x = x // 10
        dlugosc = dlugosc + 1
    potega = 1
    for i in range(dlugosc // 2):
        potega = potega * 10
    a = k // potega
    b = k % potega
    return a, b

from math import gcd


def polowy(k):
    dlugosc = 0
    x = k
    while x > 0:
        x = x // 10
        dlugosc = dlugosc + 1
    potega = 10 ** (dlugosc // 2)
    return k // potega, k % potega


ile = 0
with open("liczby1.txt") as file:
    for wiersz in file:
        a, b = polowy(int(wiersz))
        if gcd(a, b) == 1:
            ile += 1
print(ile)

def stopien_kaprekara(n):
    kwadrat = str(n * n)
    stopien = 0
    for i in range(1, len(kwadrat)):
        a = int(kwadrat[:i])
        b = int(kwadrat[i:])
        if a + b <= n:
            stopien += 1
    return stopien


najlepszy_stopien = -1
najlepsza_liczba = 0
with open("liczby2.txt") as file:
    for wiersz in file:
        n = int(wiersz)
        s = stopien_kaprekara(n)
        if s > najlepszy_stopien:
            najlepszy_stopien = s
            najlepsza_liczba = n
print(najlepszy_stopien)
print(najlepsza_liczba)
