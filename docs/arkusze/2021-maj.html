def dopelnienie(n):
    # 1.2 - tylko operacje na liczbach całkowitych
    d = 0
    potega = 1
    while n > 0:
        cyfra = n % 10
        n = n // 10
        d = d + (9 - cyfra) * potega
        potega = potega * 10
    return d


print(dopelnienie(2021))

# 1.1 - sprawdzamy wszystkie czterocyfrowe n z pierwszą cyfrą od 1 do 8
najmniejsza = najwieksza = 1000
for n in range(1000, 9000):
    roznica = abs(n - dopelnienie(n))
    if roznica < abs(najmniejsza - dopelnienie(najmniejsza)):
        najmniejsza = n
    if roznica > abs(najwieksza - dopelnienie(najwieksza)):
        najwieksza = n
print(najmniejsza, najwieksza)

def d(T, x):
    # T[0] nieużywane - tablica indeksowana od 1, jak w treści
    # zwraca liczbę sprawdzeń warunku pętli dopóki
    T.append(x)
    s = len(T) - 1
    sprawdzenia = 0
    while True:
        sprawdzenia += 1
        if not (s // 2 >= 1 and T[s] > T[s // 2]):
            break
        T[s], T[s // 2] = T[s // 2], T[s]
        s = s // 2
    return sprawdzenia

instrukcje = []
with open("instrukcje.txt") as file:
    for line in file:
        polecenie, znak = line.split()
        instrukcje.append((polecenie, znak))

# 4.1 i 4.4 - wykonujemy wszystkie instrukcje
napis = []
for polecenie, znak in instrukcje:
    if polecenie == "DOPISZ":
        napis.append(znak)
    elif polecenie == "ZMIEN":
        napis[-1] = znak
    elif polecenie == "USUN":
        napis.pop()
    elif polecenie == "PRZESUN":
        if znak in napis:
            i = napis.index(znak)  # pierwsze wystąpienie od lewej
            napis[i] = chr((ord(znak) - ord("A") + 1) % 26 + ord("A"))

# 4.2 - najdłuższy ciąg kolejnych instrukcji tego samego rodzaju
najdluzszy = 0
rodzaj = ""
dlugosc = 0
for i in range(len(instrukcje)):
    if i > 0 and instrukcje[i][0] == instrukcje[i - 1][0]:
        dlugosc += 1
    else:
        dlugosc = 1
    if dlugosc > najdluzszy:
        najdluzszy = dlugosc
        rodzaj = instrukcje[i][0]

# 4.3 - najczęściej dopisywana litera
ile = {}
for polecenie, znak in instrukcje:
    if polecenie == "DOPISZ":
        ile[znak] = ile.get(znak, 0) + 1
litera = max(ile, key=ile.get)

with open("wyniki4.txt", "w") as wynik:
    print("4.1.", len(napis), file=wynik)
    print("4.2.", rodzaj, najdluzszy, file=wynik)
    print("4.3.", litera, ile[litera], file=wynik)
    print("4.4.", "".join(napis), file=wynik)
