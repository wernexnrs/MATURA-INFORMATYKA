def na_u1(x):
    binarnie = bin(abs(x))[2:]
    if x >= 0:
        return "0" + binarnie
    # liczba ujemna: negacja bitów zapisu wartości bezwzględnej i bit znaku 1
    negacja = ""
    for bit in binarnie:
        if bit == "0":
            negacja += "1"
        else:
            negacja += "0"
    return "1" + negacja


print(na_u1(9), na_u1(-9))  # przykład z treści: 01001 10110
print(na_u1(46), na_u1(-46))

Odpowiedź: 46 = 0101110 (U1), −46 = 1010001 (U1)

def z_u1(bin):
    # bin[0] to bit znaku (w zadaniu bin[1])
    d = len(bin)
    bity = list(bin)
    if bity[0] == "1":  # liczba ujemna - negujemy bity wartości
        for i in range(1, d):
            if bity[i] == "1":
                bity[i] = "0"
            else:
                bity[i] = "1"
    x = 0
    for i in range(1, d):
        x = x * 2 + int(bity[i])
    if bity[0] == "1":
        x = -x
    return x


print(z_u1("10110"))  # przykład z treści: -9
print(z_u1("0100111"), z_u1("1001101"))

Odpowiedź: 0100111 = 39, 1001101 = −50

from datetime import date, timedelta

zubry = 90
siano, zoledzie = 100000, 5000  # w kg, 1 grudnia rano
dzien = date(2012, 12, 1)
dostawy_siana, dostawy_zoledzi = 0, 0
karmienia_sianem, karmienia_zoledziami = 0, 0
pierwszy_dzien_zoledzi = None

while dzien <= date(2013, 2, 28):
    if siano >= 50000:
        siano -= zubry * 40
        karmienia_sianem += 1
    else:
        zoledzie -= zubry * 20
        karmienia_zoledziami += 1
        if pierwszy_dzien_zoledzi is None:
            pierwszy_dzien_zoledzi = dzien
    if dzien.weekday() == 4:  # piątek wieczorem
        siano += 15000
        dostawy_siana += 1
    if dzien.weekday() == 1:  # wtorek wieczorem
        zoledzie += 4000
        dostawy_zoledzi += 1
    dzien += timedelta(days=1)

print("a)", dostawy_siana, dostawy_zoledzi)
print("b)", pierwszy_dzien_zoledzi)
print("c)", karmienia_sianem, karmienia_zoledziami)

Odpowiedź: a) 12 dostaw siana, 13 dostaw żołędzi; b) 28.12.2012; c) sianem 64 razy, żołędziami 26 razy

from datetime import date, timedelta


def czy_wystarczy(zubry):
    siano, zoledzie = 100000, 5000
    dzien = date(2012, 12, 1)
    while dzien <= date(2013, 2, 28):
        if siano >= 50000:
            siano -= zubry * 40
        else:
            zoledzie -= zubry * 20
        if siano < 0 or zoledzie < 0:
            return False
        if dzien.weekday() == 4:
            siano += 15000
        if dzien.weekday() == 1:
            zoledzie += 4000
        dzien += timedelta(days=1)
    return True


zubry = 90
while czy_wystarczy(zubry + 1):
    zubry += 1
print(zubry - 90)
