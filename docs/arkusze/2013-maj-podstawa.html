def sposob1(a, x):
    mnozenia = 0
    dodawania = 0
    W = 0
    i = 0
    while i <= 4:
        sk = a[i]
        j = 4 - i
        while j > 0:
            sk = sk * x
            mnozenia += 1
            j = j - 1
        W = W + sk
        dodawania += 1
        i = i + 1
    return W, mnozenia, dodawania


print(sposob1([1, 2, 3, 4, 5], 2))

Odpowiedź: liczba mnożeń – 10, liczba dodawań – 5

def sposob2(a, x):  # schemat Hornera
    mnozenia = 0
    dodawania = 0
    W = a[0]
    i = 0
    while i < 4:
        i = i + 1
        W = W * x + a[i]
        mnozenia += 1
        dodawania += 1
    return W, mnozenia, dodawania


print(sposob2([1, 2, 3, 4, 5], 2))

def algorytm(N):
    wypisane = []
    d = 2
    while N >= d:
        if N % d == 0:
            wypisane.append(d)
            N = N // d
        else:
            d = d + 1
    return wypisane


for N in [36, 120, 675]:
    print(N, algorytm(N))

print(len(algorytm(2 ** 100)))

Odpowiedź: a) 36 → 2, 2, 3, 3; 120 → 2, 2, 2, 3, 5; 675 → 3, 3, 3, 5, 5; b) czynniki pierwsze liczby N; c) dla N = 2100 zostanie wypisanych 100 liczb

def czynniki_pary(N):
    a = 2
    b = 0
    while N >= a:
        if N % a == 0:
            b = b + 1
            N = N // a
        else:
            if b > 0:
                print(a, b)
            a = a + 1
            b = 0
    print(a, b)

czynniki_pary(784)
