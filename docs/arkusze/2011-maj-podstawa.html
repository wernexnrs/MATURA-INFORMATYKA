def liczba_jedynek(s):
    l = 0
    while s > 0:
        if s % 2 == 1:
            l += 1
        s = s // 2
    return l


print(liczba_jedynek(3600), liczba_jedynek(5400))

def algorytm(n):
    i = 2
    while True:
        if i >= n:
            return "TAK"
        if n % i == 0:
            return "NIE"
        i += 1


for n in [25, 37, 41, 49]:

def rozklad(n):
    czynniki = []
    i = 2
    while n > 1:
        while n % i == 0:
            czynniki.append(i)
            n = n // i
        i += 1
    return czynniki


for n in [42, 17, 36]:
    print(n, rozklad(n))

Odpowiedź: 42 → 2, 3, 7; 17 → 17; 36 → 2, 2, 3, 3
