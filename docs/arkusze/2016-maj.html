Treść zadania

Liczby skojarzone. Dwie różne liczby całkowite a, b > 1 są skojarzone, jeśli suma wszystkich dodatnich dzielników a mniejszych od a wynosi b + 1, a suma wszystkich dodatnich dzielników b mniejszych od b wynosi a + 1. Np. 140 i 195 są skojarzone: dzielniki 140 sumują się do 196 = 195 + 1, a dzielniki 195 — do 141 = 140 + 1.

  • 1.1 — Dla par (78, 64), (20, 21), (75, 48) podaj dzielniki a i b (mniejsze od tych liczb), ich sumy oraz rozstrzygnij, czy liczby są skojarzone (dla pary 78, 64: sumy 90 i 63, odpowiedź NIE).
  • 1.2 — Zapisz algorytm, który dla danej liczby całkowitej a > 1 znajdzie i wypisze liczbę b skojarzoną z a albo komunikat „NIE”, jeśli taka liczba nie istnieje. Wolno używać tylko dodawania, odejmowania, mnożenia, dzielenia całkowitego i reszty z dzielenia; oceniana jest też liczba wykonywanych operacji arytmetycznych.
Python
def suma_dzielnikow(n):
    # suma dzielników n mniejszych od n - sprawdzamy tylko i <= pierwiastek(n)
    suma = 1
    i = 2
    while i * i <= n:
        if n % i == 0:
            suma += i
            if n // i != i:
                suma += n // i
        i += 1
    return suma


def czy_skojarzone(a, b):
    return a != b and suma_dzielnikow(a) == b + 1 and suma_dzielnikow(b) == a + 1


def skojarzona(a):
    b = suma_dzielnikow(a) - 1
    if b > 1 and b != a and suma_dzielnikow(b) == a + 1:
        return b
    return "NIE"


# 1.1
for a, b in [(78, 64), (20, 21), (75, 48)]:
    print(a, b, suma_dzielnikow(a), suma_dzielnikow(b), "TAK" if czy_skojarzone(a, b) else "NIE")

# 1.2
print(skojarzona(140), skojarzona(75), skojarzona(20))

Przestawienia w tablicy. Dana jest dodatnia liczba całkowita n i tablica A[1..n] liczb całkowitych. Funkcja przestawia elementy tak, by na lewo znalazły się elementy mniejsze od klucza (pierwszego elementu tablicy), a na prawo — większe lub równe:

Kod
funkcja przestaw(A)
  klucz ← A[1]
  w ← 1
  dla k = 2, 3, ..., n wykonaj
      jeśli A[k] < klucz
          zamień(A[w], A[k])
          w ← w + 1
  • 2.1 — Podaj kolejność elementów tablicy A = [4, 6, 3, 5, 2, 1] (n = 6) po wykonaniu przestaw(A).
  • 2.2 — Podaj przykład siedmioelementowej tablicy A, dla której funkcja wykona zamień dokładnie 5 razy.
  • 2.3 — Tablica A[1..100] zawiera liczby 1–100 w kolejności: najpierw rosnąco liczby kończące się na 0 (10, 20, …, 100), potem rosnąco kończące się na 9 (9, 19, …, 99), potem na 8 itd., aż do 1, 11, …, 91. Podaj wartość zmiennej w oraz A[1], A[2], A[3] po wykonaniu przestaw(A).
Python
def przestaw(A):
    # A indeksowana od 0 (w treści od 1); zwraca w (indeksowane od 1) i liczbę zamian
    klucz = A[0]
    w = 0
    zamiany = 0
    for k in range(1, len(A)):
        if A[k] < klucz:
            A[w], A[k] = A[k], A[w]
            w += 1
            zamiany += 1
    return w + 1, zamiany

#include <iostream>
#include <utility>

std::pair<int, int> przestaw(std::vector<int>& A) {
    // A indeksowana od 0 (w treści od 1); zwraca w (indeksowane od 1) i liczbę zamian
    int klucz = A[0];
    int w = 0;
    int zamiany = 0;
    for (size_t k = 1; k < A.size(); k++) {
        if (A[k] < klucz) {
            std::swap(A[w], A[k]);
            w++;
            zamiany++;
        }
    }
    return {w + 1, zamiany};
}

import java.util.*;

public class Zadanie {
    // A indeksowana od 0 (w treści od 1); zwraca {w (indeksowane od 1), liczba zamian}
    static int[] przestaw(List<Integer> A) {
        int klucz = A.get(0);
        int w = 0;
        int zamiany = 0;
        for (int k = 1; k < A.size(); k++) {
            if (A.get(k) < klucz) {
                Collections.swap(A, w, k);
                w++;
                zamiany++;
            }
        }
        return new int[] {w + 1, zamiany};
    }

    public static void main(String[] args) {
        // 2.1
        var A = new ArrayList<>(List.of(4, 6, 3, 5, 2, 1));
        przestaw(A);
        System.out.println(A);

        // 2.2 - zamiana wykona się tyle razy, ile elementów A[2..n] jest mniejszych od A[1]
        A = new ArrayList<>(List.of(8, 1, 2, 3, 4, 5, 9));
        System.out.println(przestaw(A)[1]);

        // 2.3
        A = new ArrayList<>();
        for (int koncowka : new int[] {0, 9, 8, 7, 6, 5, 4, 3, 2, 1})
            for (int liczba = 1; liczba < 101; liczba++)
                if (liczba % 10 == koncowka)
                    A.add(liczba);
        int[] wynik = przestaw(A);
        System.out.println(wynik[0] + " " + A.get(0) + " " + A.get(1) + " " + A.get(2));
    }
}

Punkty z pliku punkty.txt (kwadrat o boku 400, środek (200, 200), koło wpisane). Przy równomiernym rozkładzie punktów stosunek liczby punktów w kole n_k do liczby wszystkich punktów n jest w przybliżeniu równy stosunkowi pola koła (πr²) do pola kwadratu. Wyznacz przybliżoną wartość π na podstawie: pierwszych 1000 punktów, pierwszych 5000 punktów oraz wszystkich punktów. Wyniki zaokrąglij do 4 miejsc po przecinku (dla pierwszych 100 punktów wychodzi 3,2000).

Python
punkty = []
with open("punkty.txt") as file:
    for line in file:
        x, y = map(int, line.split())
        punkty.append((x, y))


def przyblizenie_pi(n):
    # punkty należące do koła (razem z brzegiem) spośród n pierwszych
    w_kole = 0
    for x, y in punkty[:n]:
        if (x - 200) ** 2 + (y - 200) ** 2 <= 200 ** 2:
            w_kole += 1
    # nk / n = Pk / P = pi * 200^2 / 400^2 = pi / 4
    return 4 * w_kole / n


for n in [1000, 5000, len(punkty)]:
    print(f"{przyblizenie_pi(n):.4f}")

#include <fstream>
#include <print>
#include <utility>

std::vector<std::pair<int, int>> punkty;

double przyblizenie_pi(int n) {
    // punkty należące do koła (razem z brzegiem) spośród n pierwszych
    int w_kole = 0;
    for (int i = 0; i < n; i++) {
        auto [x, y] = punkty[i];
        if ((x - 200) * (x - 200) + (y - 200) * (y - 200) <= 200 * 200)
            w_kole++;
    }
    // nk / n = Pk / P = pi * 200^2 / 400^2 = pi / 4
    return 4.0 * w_kole / n;
}

int main() {
    std::ifstream plik("punkty.txt");
    int x, y;
    while (plik >> x >> y)
        punkty.push_back({x, y});

    for (int n : {1000, 5000, static_cast<int>(punkty.size())})
        std::println("{:.4f}", przyblizenie_pi(n));
}
Java 25
import java.nio.file.*;
import java.util.*;

public class Zadanie {
    static List<int[]> punkty = new ArrayList<>();

    static double przyblizeniePi(int n) {
        // punkty należące do koła (razem z brzegiem) spośród n pierwszych
        int wKole = 0;
        for (var p : punkty.subList(0, n)) {
            int x = p[0], y = p[1];
            if ((x - 200) * (x - 200) + (y - 200) * (y - 200) <= 200 * 200)
                wKole++;
        }
        // nk / n = Pk / P = pi * 200^2 / 400^2 = pi / 4
        return 4.0 * wKole / n;
    }

    public static void main(String[] args) throws Exception {
        for (var line : Files.readAllLines(Path.of("punkty.txt"))) {
            var p = line.strip().split("\\s+");
            punkty.add(new int[] {Integer.parseInt(p[0]), Integer.parseInt(p[1])});
        }
        for (int n : new int[] {1000, 5000, punkty.size()})
            System.out.println(String.format(Locale.ROOT, "%.4f", przyblizeniePi(n)));  // Locale.ROOT - kropka dziesiętna
    }
}

Punkty z pliku punkty.txt; π_n to przybliżenie liczby π wyznaczone (jak w 4.2) z n pierwszych punktów pliku. Błąd bezwzględny definiujemy jako ε_n = |π − π_n|, gdzie π to wartość ze standardowej funkcji używanego narzędzia. Oblicz ε_n dla n = 1, 2, …, 1700 i utwórz czytelny wykres liniowy zmian tego błędu. Wartości ε_1000 i ε_1700, zaokrąglone do 4 miejsc po przecinku, zapisz w wyniki_4.txt.

Python
import math

w_kole = 0
bledy = []  # bledy[n - 1] = |pi - pi_n|

with open("punkty.txt") as file:
    for n in range(1, 1701):
        x, y = map(int, file.readline().split())
        if (x - 200) ** 2 + (y - 200) ** 2 <= 200 ** 2:
            w_kole += 1
        bledy.append(abs(math.pi - 4 * w_kole / n))

# dane do wykresu liniowego: n -> bledy[n - 1]
print(f"{bledy[1000 - 1]:.4f}")
print(f"{bledy[1700 - 1]:.4f}")

Szyfr Cezara z kluczem k zastępuje każdą literę literą położoną o k pozycji dalej w alfabecie, cyklicznie (po Z następuje A), np. dla k = 3: B → E, X → A, Z → C. Rozważamy słowa z wielkich liter alfabetu angielskiego (A–Z) o długości do 30 znaków.
