## 1a

```py
from math import ceil, sqrt


def wymiar(d):
    # najmniejsze n, dla którego n * n >= d, czyli n = sufit(pierwiastek(d))
    n = ceil(sqrt(d))
    while n * n < d:  # zabezpieczenie przed błędem zaokrąglenia
        n += 1
    return n


print(wymiar(24), wymiar(25), wymiar(26))
```

**Odpowiedź:** n = ⌈√d⌉ – najmniejsza liczba naturalna większa lub równa √d

```py
from math import ceil, sqrt


def szyfruj(tekst):
    d = len(tekst)
    n = ceil(sqrt(d))
    while n * n < d:
        n += 1
    s = n * n
    tekst = tekst + "_" * (s - d)  # puste pola uzupełniamy znakami odstępu
    szyfr = ""
    for kolumna in range(n):  # odczyt kolumnami
        for wiersz in range(n):
            szyfr += tekst[wiersz * n + kolumna]
    return szyfr


# c) przykład z treści
print(szyfruj("ALGORYTM_PRZESTAWIENIOWY"))

# b) szyfr ma 64 znaki (kwadrat 8 x 8), a zamiana wierszy z kolumnami
#    wykonana drugi raz przywraca tekst - wystarczy "zaszyfrować" szyfr
szyfr = "BTLLTU_ĘL_EOYPM_ĄPJZLCYNDREOKYLI_ZMFO_ĄGJY_Ó_N_DEWFWGISYSII_ŁEI_"
print(szyfruj(szyfr))
```

**Odpowiedź:** b) BŁĄD JEST PRZYWILEJEM FILOZOFÓW, TYLKO GŁUPCY NIE MYLĄ SIĘ NIGDY; c) przykład: AYRAILTZWOGMEIWO_SEYRPTN_

```py
def liczba_zer(a):
    # a[1..1023], a[0] nieużywane - indeksy jak w zadaniu
    przypisania_s = 0
    zera = 0
    l, p = 1, len(a) - 1
    while l <= p:
        s = (l + p) // 2
        przypisania_s += 1
        if a[s] == 1:
            p = s - 1
        else:
            zera = zera + (s - l + 1)  # zera na pozycjach l..s
            l = s + 1
    return zera, przypisania_s


# sprawdzenie dla wszystkich możliwych tablic (k zer, potem jedynki)
liczby_przypisan = set()
for k in range(1024):
    a = [None] + [0] * k + [1] * (1023 - k)
    zera, przypisania_s = liczba_zer(a)
    if zera != k:
        print("błąd dla", k)
    liczby_przypisan.add(przypisania_s)
print(liczby_przypisan)
```

**Odpowiedź:** a) liczba_zer + (s − l + 1), l ← s + 1; b) zawsze 10 = log₂1024 – w każdym przebiegu pętli przedział do sprawdzenia zmniejsza się o połowę

```py
def czy_anagram(a, b):
    return sorted(a) == sorted(b)


with open("anagram.txt") as file, open("odp_4b.txt", "w") as wynik:
    for line in file:
        slowa = line.split()
        if all(czy_anagram(slowa[0], slowo) for slowo in slowa[1:]):
            wynik.write(" ".join(slowa) + "\n")
            print(" ".join(slowa))
```

```py
dekady = {"50": 0, "60": 0, "70": 0, "80": 0, "90": 0}

with open("pesel.txt") as file:
```
