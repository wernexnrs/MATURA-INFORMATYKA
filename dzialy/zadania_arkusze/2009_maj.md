```py
from math import floor


def punkty_kratowe(R):
    # liczba punktów (x, y) o współrzędnych całkowitych, dla których x^2 + y^2 <= R^2
    r = floor(R)
    N = 0
    for x in range(-r, r + 1):
        for y in range(-r, r + 1):
            if x * x + y * y <= R * R:
                N += 1
    return N


print(punkty_kratowe(0), punkty_kratowe(1), punkty_kratowe(1.5))  # przykłady z treści: 1 5 9
print(punkty_kratowe(2.01))
print(punkty_kratowe(4.50))
```

**Odpowiedź:** N(2,01) = 13, N(4,50) = 69

## 3b

```py
# Dane: liczby naturalne a i b, z których co najmniej jedna jest większa od 0
# Wynik: największy wspólny dzielnik liczb a i b
def nwd(a, b):
    while b > 0:
        reszta = a % b
        a = b
        b = reszta
    return a


print(nwd(16, 12), nwd(56, 72))
```

**Odpowiedź:** dopóki b > 0: reszta := a mod b, a := b, b := reszta; wynikiem jest a (np. nwd(16, 12) = 4)

```py
from random import randint

x, y = 1, 1
punkty = []

# a) 5000 iteracji, za każdym razem losowy układ równań
for i in range(5000):
    if randint(0, 1) == 0:
        x, y = -0.4 * x - 1, -0.4 * y + 0.1
    else:
        x, y = 0.76 * x - 0.4 * y, 0.4 * x + 0.76 * y
    punkty.append((x, y))

smok = punkty[100:]  # pomijamy 100 pierwszych iteracji
xs = [p[0] for p in smok]
ys = [p[1] for p in smok]

with open("zad_4.txt", "w") as wynik:
    wynik.write("a)\n")
    for x, y in punkty:
        wynik.write(f"{x} {y}\n")
    # c) środek masy
    wynik.write(f"c) {round(sum(xs) / len(xs), 1)} {round(sum(ys) / len(ys), 1)}\n")
    # d) rozmiary smoka
    wynik.write(f"d) xmin={round(min(xs), 1)} xmax={round(max(xs), 1)} "
                f"ymin={round(min(ys), 1)} ymax={round(max(ys), 1)}\n")

print("c)", round(sum(xs) / len(xs), 1), round(sum(ys) / len(ys), 1))
print("d)", round(min(xs), 1), round(max(xs), 1), round(min(ys), 1), round(max(ys), 1))

# b) obraz smoka jako plik SVG
skala = 300
with open("smok.svg", "w") as obraz:
    obraz.write('<svg xmlns="http://www.w3.org/2000/svg" width="600" height="400" '
                'style="background:white">\n')
    for x, y in smok:
        px = (x + 1.2) * skala
        py = 400 - (y + 0.5) * skala
        obraz.write(f'<circle cx="{px:.1f}" cy="{py:.1f}" r="0.8" fill="green"/>\n')
    obraz.write("</svg>\n")
```

**Odpowiedź:** wyniki są losowe, klucz nie podaje wartości; przykładowo: c) środek masy ≈ (−0,6; −0,1), d) x od −1,2 do 0,4, y od −0,7 do 0,4

```py
def nakladka(a, b):
    # długość najdłuższego sufiksu słowa a, który jest prefiksem słowa b
    for k in range(min(len(a), len(b)), 0, -1):
        if a[-k:] == b[:k]:
            return k
    return 0


def slowo_c(a, b):
    if b in a:
        return a
    k1 = nakladka(a, b)  # najpierw A, potem B
    k2 = nakladka(b, a)  # najpierw B, potem A
    if k1 >= k2:
        return a + b[k1:]
    return b + a[k2:]


with open("dane.txt") as file, open("slowa.txt", "w") as wynik:
    for line in file:
        a, b = line.split()
        wynik.write(slowo_c(a, b) + "\n")
```

**Odpowiedź:** plik slowa.txt: 0001100111101010111, 1100001101, 11000100, …
