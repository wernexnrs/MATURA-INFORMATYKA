## 1.1 i 1.2

```py
wywolania = 0


def przestaw(n):
    global wywolania
    wywolania += 1
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        w = a + 10 * b + 100 * przestaw(n)
    else:
        if a > 0:
            w = a + 10 * b
        else:
            w = b
    return w


# 1.1
for n in [316498, 43657688, 154005710, 998877665544321]:
    wywolania = 0
    print(n, przestaw(n), wywolania)

# 1.2 - liczba wywołań dla liczb k-cyfrowych porównana ze wzorami
for k in range(1, 9):
    wywolania = 0
    przestaw(10 ** (k - 1))   # dowolna liczba k-cyfrowa
    wzor1 = k / 2
    wzor2 = (k + 1) // 2
    wzor3 = k / 2 if k % 2 == 0 else (k + 1) / 2
    wzor4 = (k + 1) / 2
    print(k, wywolania, wzor1 == wywolania, wzor2 == wywolania,
          wzor3 == wywolania, wzor4 == wywolania)
```

**Odpowiedź:** 1.1: 316498 → 134689, 3 wywołania; 43657688 → 34566788, 4; 154005710 → 145007501, 5; 998877665544321 → 989786756453412, 8. 1.2: 1. F, 2. P, 3. P, 4. F (liczba wywołań to (k + 1) div 2).

```py
def przestaw2(n):
    w = 0
    p = 1
    while n > 0:
        r = n % 100
        a = r // 10
        b = r % 10
        if n > 9:
            w = p * a + 10 * p * b + w
        else:
            w = p * b + w
        n = n // 100
        p = p * 100
    return w


for n in [316498, 43657688, 154005710, 998877665544321]:
    print(n, przestaw2(n))


# sprawdzenie z wersją rekurencyjną
def przestaw(n):
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        return a + 10 * b + 100 * przestaw(n)
    if a > 0:
        return a + 10 * b
    return b


for n in range(0, 200000):
    assert przestaw(n) == przestaw2(n)
```

```py
with open("symbole.txt") as file:
    napisy = [wiersz.strip() for wiersz in file]

print("2.1")

srodki = []
for i in range(1, len(napisy) - 1):
    for j in range(1, 11):
        znak = napisy[i][j]
        kwadrat = True
        for di in [-1, 0, 1]:
            for dj in [-1, 0, 1]:
                if napisy[i + di][j + dj] != znak:
                    kwadrat = False
        if kwadrat:
            srodki.append((i + 1, j + 1))   # numeracja od 1
print("2.2", len(srodki), end=" ")
for wiersz, pozycja in srodki:
    print(wiersz, pozycja, end=" ")
print()

# 2.3 i 2.4
cyfra = {"o": 0, "+": 1, "*": 2}
symbol = "o+*"


def na_liczbe(napis):
    liczba = 0
    for znak in napis:
        liczba = liczba * 3 + cyfra[znak]
    return liczba


def na_napis(liczba):
    napis = ""
    while liczba > 0:
        napis = symbol[liczba % 3] + napis
        liczba //= 3
    return napis


najwiekszy = max(napisy, key=na_liczbe)
print("2.3", na_liczbe(najwiekszy), najwiekszy)

suma = sum(na_liczbe(napis) for napis in napisy)
print("2.4", suma, na_napis(suma))
```

## 3.1 i 3.2

```py
from math import gcd

with open("dron.txt") as file:
    ruchy = [tuple(map(int, wiersz.split())) for wiersz in file]

# 3.1
ile = 0
for a, b in ruchy:
    if gcd(abs(a), abs(b)) > 1:
        ile += 1
print("3.1", ile)

# 3.2
punkty = []
x, y = 0, 0
for a, b in ruchy:
    x += a
    y += b
    punkty.append((x, y))

wewnatrz = 0
for x, y in punkty:
    if 0 < x < 5000 and 0 < y < 5000:
        wewnatrz += 1
print("3.2 a)", wewnatrz)

zbior = set(punkty)
for i in range(len(punkty)):
    for j in range(i + 1, len(punkty)):
        (x1, y1), (x2, y2) = punkty[i], punkty[j]
        if (x1 + x2) % 2 == 0 and (y1 + y2) % 2 == 0:
            srodek = ((x1 + x2) // 2, (y1 + y2) // 2)
            if srodek in zbior and srodek != punkty[i]:
                print("3.2 b)", punkty[i], srodek, punkty[j])
```

**Odpowiedź:** 3.1: 40. 3.2: a) 24; b) (5832, 1801), (7410, 1990), (8988, 2179) — środkiem jest (7410, 1990).
