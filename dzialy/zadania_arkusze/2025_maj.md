## 1.1 i 1.2

> **Treść zadania:** Dana jest rekurencyjna funkcja przestaw dla nieujemnej liczby całkowitej n (mod — reszta, div — część całkowita z dzielenia):
>
> ```
> przestaw(n):
>     r ← n mod 100
>     a ← r div 10
>     b ← r mod 10
>     n ← n div 100
>     jeżeli n > 0
>         w ← a + 10 * b + 100 * przestaw(n)
>     w przeciwnym razie
>         jeżeli a > 0
>             w ← a + 10 * b
>         w przeciwnym razie
>             w ← b
>     wynikiem jest w
> ```
>
> - 1.1 — Uzupełnij tabelę: dla n = 43657688, 154005710 i 998877665544321 podaj wynik przestaw(n) oraz łączną liczbę wywołań funkcji (z pierwszym włącznie). Przykład: przestaw(316498) = 134689, 3 wywołania.
> - 1.2 — Oceń (P/F), czy liczba wywołań funkcji dla liczby k-cyfrowej (k > 0) jest równa: 1) k/2; 2) (k + 1) div 2; 3) k/2 dla parzystego k oraz (k + 1)/2 dla nieparzystego k; 4) k/2 + 1.

```py
wywolania = 0


def przestaw(n):
    global wywolania
    wywolania += 1
    r = n % 100
    a = r // 10
    b = r % 10
    n = n // 100
    if n > 0:
        w = a + 10 * b + 100 * przestaw(n)
    else:
        if a > 0:
            w = a + 10 * b
        else:
            w = b
    return w
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    static final String SYMBOL = "o+*";

    static long naLiczbe(String napis) {
        long liczba = 0;
        for (char znak : napis.toCharArray()) {
            liczba = liczba * 3 + SYMBOL.indexOf(znak);  // "o" -> 0, "+" -> 1, "*" -> 2
        }
        return liczba;
    }

    static String naNapis(long liczba) {
        String napis = "";
        while (liczba > 0) {
            napis = SYMBOL.charAt((int) (liczba % 3)) + napis;
            liczba /= 3;
        }
        return napis;
    }

    public static void main(String[] args) throws Exception {
        List<String> napisy = new ArrayList<>();
        for (String wiersz : Files.readAllLines(Path.of("symbole.txt"))) {
            napisy.add(wiersz.strip());
        }

        // 2.1
        System.out.println("2.1");
        for (String napis : napisy) {
            if (napis.equals(new StringBuilder(napis).reverse().toString())) {
                System.out.println(napis);
            }
        }

        // 2.2
        List<int[]> srodki = new ArrayList<>();
        for (int i = 1; i < napisy.size() - 1; i++) {
            for (int j = 1; j < 11; j++) {
                char znak = napisy.get(i).charAt(j);
                boolean kwadrat = true;
                for (int di = -1; di <= 1; di++) {
                    for (int dj = -1; dj <= 1; dj++) {
                        if (napisy.get(i + di).charAt(j + dj) != znak) {
                            kwadrat = false;
                        }
                    }
                }
                if (kwadrat) {
                    srodki.add(new int[]{i + 1, j + 1});   // numeracja od 1
                }
            }
        }
        StringBuilder linia = new StringBuilder("2.2 " + srodki.size() + " ");
        for (int[] s : srodki) {
            linia.append(s[0]).append(" ").append(s[1]).append(" ");
        }
        System.out.println(linia);

        // 2.3 i 2.4
        String najwiekszy = napisy.get(0);

            suma += naLiczbe(napis);
        }
        System.out.println("2.4 " + suma + " " + naNapis(suma));
    }
}
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class Zadanie {
    // punkt jako rekord - equals i hashCode działają po wartościach
    record Punkt(int x, int y) {
        @Override
        public String toString() {
            return "(" + x + ", " + y + ")";
        }
    }

    static int gcd(int a, int b) {
        while (b != 0) {
            int t = a % b;
            a = b;
            b = t;
        }
        return a;
    }

    public static void main(String[] args) throws Exception {
        List<int[]> ruchy = new ArrayList<>();
        for (String wiersz : Files.readAllLines(Path.of("dron.txt"))) {
            String[] t = wiersz.trim().split("\\s+");
            ruchy.add(new int[]{Integer.parseInt(t[0]), Integer.parseInt(t[1])});
        }

        // 3.1
        int ile = 0;
        for (int[] r : ruchy) {
            if (gcd(Math.abs(r[0]), Math.abs(r[1])) > 1) {
                ile += 1;
            }
        }
        System.out.println("3.1 " + ile);

        // 3.2
        List<Punkt> punkty = new ArrayList<>();
        int x = 0, y = 0;
        for (int[] r : ruchy) {
            x += r[0];
            y += r[1];
            punkty.add(new Punkt(x, y));
        }

        int wewnatrz = 0;
        for (Punkt p : punkty) {
            if (0 < p.x() && p.x() < 5000 && 0 < p.y() && p.y() < 5000) {
                wewnatrz += 1;
            }
        }
        System.out.println("3.2 a) " + wewnatrz);

        Set<Punkt> zbior = new HashSet<>(punkty);
        for (int i = 0; i < punkty.size(); i++) {
            for (int j = i + 1; j < punkty.size(); j++) {
                Punkt p1 = punkty.get(i), p2 = punkty.get(j);
                if ((p1.x() + p2.x()) % 2 == 0 && (p1.y() + p2.y()) % 2 == 0) {
                    Punkt srodek = new Punkt((p1.x() + p2.x()) / 2, (p1.y() + p2.y()) / 2);
                    if (zbior.contains(srodek) && !srodek.equals(p1)) {
                        System.out.println("3.2 b) " + p1 + " " + srodek + " " + p2);
                    }
                }
            }
        }
    }
}
```
