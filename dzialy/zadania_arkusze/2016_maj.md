def suma_dzielnikow(n):
    # suma dzielników n mniejszych od n - sprawdzamy tylko i <= pierwiastek(n)
    suma = 1
    i = 2
    while i * i <= n:
        if n % i == 0:
            suma += i
            if n // i != i:
                suma += n // i
        i += 1
    return suma


def czy_skojarzone(a, b):
    return a != b and suma_dzielnikow(a) == b + 1 and suma_dzielnikow(b) == a + 1


def skojarzona(a):
    b = suma_dzielnikow(a) - 1
    if b > 1 and b != a and suma_dzielnikow(b) == a + 1:
        return b
    return "NIE"


# 1.1
for a, b in [(78, 64), (20, 21), (75, 48)]:
    print(a, b, suma_dzielnikow(a), suma_dzielnikow(b), "TAK" if czy_skojarzone(a, b) else "NIE")

# 1.2
print(skojarzona(140), skojarzona(75), skojarzona(20))
```
**Odpowiedź:** 1.1: (78, 64) NIE, (20, 21) sumy 22 i 11 - NIE, (75, 48) sumy 49 i 76 - TAK. 1.2: b = suma_dzielnikow(a) - 1, sprawdzamy, czy suma_dzielnikow(b) = a + 1.

def przestaw(A):
    # A indeksowana od 0 (w treści od 1); zwraca w (indeksowane od 1) i liczbę zamian
    klucz = A[0]
    w = 0
    zamiany = 0
    for k in range(1, len(A)):
        if A[k] < klucz:
            A[w], A[k] = A[k], A[w]
            w += 1
            zamiany += 1
    return w + 1, zamiany


# 2.1
A = [4, 6, 3, 5, 2, 1]
przestaw(A)
print(A)

# 2.2 - zamiana wykona się tyle razy, ile elementów A[2..n] jest mniejszych od A[1]
A = [8, 1, 2, 3, 4, 5, 9]
print(przestaw(A)[1])

# 2.3
A = []
for koncowka in [0, 9, 8, 7, 6, 5, 4, 3, 2, 1]:
    for liczba in range(1, 101):
        if liczba % 10 == koncowka:
            A.append(liczba)
w, zamiany = przestaw(A)
print(w, A[0], A[1], A[2])
```
**Odpowiedź:** 2.1: A = [3, 2, 1, 5, 6, 4]; 2.2: np. [8, 1, 2, 3, 4, 5, 9]; 2.3: w = 10, A[1] = 9, A[2] = 8, A[3] = 7

```py
punkty = []
with open("punkty.txt") as file:
    for line in file:
        x, y = map(int, line.split())
        punkty.append((x, y))


def przyblizenie_pi(n):
    # punkty należące do koła (razem z brzegiem) spośród n pierwszych
    w_kole = 0
    for x, y in punkty[:n]:
        if (x - 200) ** 2 + (y - 200) ** 2 <= 200 ** 2:
            w_kole += 1
    # nk / n = Pk / P = pi * 200^2 / 400^2 = pi / 4
    return 4 * w_kole / n


for n in [1000, 5000, len(punkty)]:
    print(f"{przyblizenie_pi(n):.4f}")
```
**Odpowiedź:** 3,2280; 3,1656; 3,1416

## 4.3

```py
import math

w_kole = 0
bledy = []  # bledy[n - 1] = |pi - pi_n|

with open("punkty.txt") as file:
    for n in range(1, 1701):
        x, y = map(int, file.readline().split())
        if (x - 200) ** 2 + (y - 200) ** 2 <= 200 ** 2:
            w_kole += 1
        bledy.append(abs(math.pi - 4 * w_kole / n))

# dane do wykresu liniowego: n -> bledy[n - 1]
print(f"{bledy[1000 - 1]:.4f}")
print(f"{bledy[1700 - 1]:.4f}")
```
**Odpowiedź:** ε1000 = 0,0864; ε1700 = 0,0184 (+ wykres liniowy)

def szyfruj(slowo, klucz):
    wynik = ""
    for znak in slowo:
        wynik += chr((ord(znak) - 65 + klucz) % 26 + 65)
    return wynik


with open("dane_6_1.txt") as file, open("wyniki_6_1.txt", "w") as wynik:

def odszyfruj(slowo, klucz):
    wynik = ""
    for znak in slowo:
        wynik += chr((ord(znak) - 65 - klucz) % 26 + 65)
    return wynik


with open("dane_6_2.txt") as file, open("wyniki_6_2.txt", "w") as wynik:
    for line in file:
        dane = line.split()
        slowo = dane[0]
        if len(dane) > 1:
            klucz = int(dane[1])
        else:
            klucz = 0  # w kilku wierszach brakuje klucza - słowo zostaje bez zmian
