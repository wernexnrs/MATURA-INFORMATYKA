```py
def A(m, n, wywolania):
    # wywolania - lista, do której dopisujemy argumenty kolejnych wywołań rekurencyjnych
    if n == 1:
        return m
    if n % 2 == 0:
        wywolania.append((2 * m, n // 2))
        return A(2 * m, n // 2, wywolania)
    wywolania.append((m, (n - 1) // 2))
    return 2 * A(m, (n - 1) // 2, wywolania) + m


for m, n in [(3, 9), (2 ** 5, 2 ** 5), (10, 15), (1, 2 ** 100 + 1)]:
    wywolania = []
    A(m, n, wywolania)
    print(m, n, len(wywolania), wywolania if len(wywolania) < 10 else "")
```

**Odpowiedź:** (2⁵, 2⁵): 5 wywołań — A(2⁶, 2⁴), A(2⁷, 2³), A(2⁸, 2²), A(2⁹, 2¹), A(2¹⁰, 2⁰); (10, 15): 3 wywołania — A(10, 7), A(10, 3), A(10, 1); (1, 2¹⁰⁰ + 1): 100 wywołań.

```py
def drugie_argumenty(n):
    # wartości drugiego argumentu w kolejnych wywołaniach rekurencyjnych
    wynik = []
    while n > 1:
        if n % 2 == 0:
            n = n // 2
        else:
            n = (n - 1) // 2
        wynik.append(n)
    return wynik


print(8, drugie_argumenty(8))
for k in range(3, 7):
    print("n = 2^%d:" % k, drugie_argumenty(2 ** k))
    print("n = 2^%d - 1:" % k, drugie_argumenty(2 ** k - 1))
```

**Odpowiedź:** n = 8: 3 wywołania, i-te ma drugi argument 8/2ⁱ = 2³⁻ⁱ; n = 2ᵏ: k wywołań, drugi argument 2ᵏ⁻ⁱ; n = 2ᵏ − 1: k − 1 wywołań, drugi argument 2ᵏ⁻ⁱ − 1.

```py
def przeniesienia(a, b):
    # a i b mają tyle samo cyfr; operujemy tylko na liczbach całkowitych
    w_pamieci = 0
    p = 0
    while a > 0:
        suma = a % 10 + b % 10 + w_pamieci
        if suma >= 10:
            w_pamieci = 1
            p += 1
        else:
            w_pamieci = 0
        a = a // 10
        b = b // 10
    return p


print(przeniesienia(27732, 72619))  # przykład z treści: 4
for a, b in [(37932, 12528), (88765, 11111), (456789, 222222)]:
    print(a, b, przeniesienia(a, b))
```

**Odpowiedź:** 2.1: 37932 + 12528 → 3, 88765 + 11111 → 0, 456789 + 222222 → 3. 2.2: funkcja `przeniesienia` powyżej.

```py
def najdluzszy_prefiksosufiks(s1, s2):
    for dlugosc in range(min(len(s1), len(s2)), 0, -1):
```

```py
# przykładowa hierarchia z Rysunku 1. (zapis jak w pliku: przełożony pracownika i)
szef = [0, 1, 2, 1, 2, 2, 3, 4]
n = len(szef)
bezposredni = [0] * (n + 1)
wszyscy = [0] * (n + 1)
for i in range(2, n + 1):
    bezposredni[szef[i - 1]] += 1
    p = szef[i - 1]
    while p != 0:  # każdy przełożony pracownika i ma go wśród podwładnych
        wszyscy[p] += 1
        p = szef[p - 1]

for i in range(1, n + 1):
    print(i, bezposredni[i], wszyscy[i])
```

**Odpowiedź:** 1: 2 i 7; 2: 3 i 4; 3: 1 i 1; 4: 1 i 1; 5, 6, 7, 8: 0 i 0.

```py
szef = [0]  # szef[i] - bezpośredni przełożony pracownika i (indeks 0 nieużywany)

podwladni = [0] * (n + 1)

    podwladni[szef[i]] += 1

# 4.3
numer = 1
for i in range(2, n + 1):
    if podwladni[i] > podwladni[numer]:
        numer = i
print("4.3", numer, podwladni[numer])

# 4.4 - liczba przełożonych = głębokość pracownika w drzewie
glebokosc = [-1] * (n + 1)
glebokosc[1] = 0
for i in range(2, n + 1):
    sciezka = []
    p = i
    while glebokosc[p] == -1:
        sciezka.append(p)
        p = szef[p]
    for q in reversed(sciezka):
        glebokosc[q] = glebokosc[szef[q]] + 1

najwiecej = max(glebokosc)
print("4.4", najwiecej, glebokosc.count(najwiecej))
```

```py
def na_system(liczba, podstawa):
    cyfry = ""
    while liczba > 0:
        cyfry = str(liczba % podstawa) + cyfry
        liczba //= podstawa
    return cyfry


# 1440(5) + x(5) = 427(10) = y(3) - 110002(3)
x = 427 - int("1440", 5)
y = 427 + int("110002", 3)
print(na_system(x, 5), na_system(y, 3))
```
