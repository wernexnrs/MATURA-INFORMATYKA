## 1.1

> **Treść zadania:** Dla dodatnich liczb całkowitych m, n funkcja A jest zdefiniowana rekurencyjnie:
>
> - A(m, n) = m, gdy n = 1;
> - A(m, n) = A(2·m, n/2), gdy n > 1 i n jest parzyste;
> - A(m, n) = 2·A(m, (n − 1)/2) + m, gdy n > 1 i n jest nieparzyste.
>
> Przykład: obliczenie A(3, 9) wymaga trzech wywołań rekurencyjnych A(3, 4), A(6, 2), A(12, 1), bo A(3, 9) = 2·A(3, 4) + 3 = 2·A(6, 2) + 3 = 2·A(12, 1) + 3 = 27.
>
> Uzupełnij tabelę: dla par (m, n) = (2⁵, 2⁵) oraz (10, 15) podaj liczbę wywołań rekurencyjnych funkcji A i wypisz te wywołania z argumentami, a dla (1, 2¹⁰⁰ + 1) podaj tylko liczbę wywołań rekurencyjnych.

```py
def A(m, n, wywolania):
    # wywolania - lista, do której dopisujemy argumenty kolejnych wywołań rekurencyjnych
    if n == 1:
        return m
    if n % 2 == 0:
        wywolania.append((2 * m, n // 2))
        return A(2 * m, n // 2, wywolania)
    wywolania.append((m, (n - 1) // 2))
    return 2 * A(m, (n - 1) // 2, wywolania) + m


for m, n in [(3, 9), (2 ** 5, 2 ** 5), (10, 15), (1, 2 ** 100 + 1)]:
    wywolania = []
    A(m, n, wywolania)
    print(m, n, len(wywolania), wywolania if len(wywolania) < 10 else "")
```

```cpp
// n = 2^100 + 1 nie mieści się w long long - używamy 128-bitowego typu GCC (__int128)
using Liczba = __int128;

std::string na_napis(Liczba x) {
    if (x == 0) {
        return "0";
    }
    std::string s = "";
    while (x > 0) {
        s = char('0' + int(x % 10)) + s;
        x /= 10;
    }
    return s;
}

// wywolania - lista, do której dopisujemy argumenty kolejnych wywołań rekurencyjnych
Liczba A(Liczba m, Liczba n, std::vector<std::pair<Liczba, Liczba>>& wywolania) {
    if (n == 1) {
        return m;
    }
    if (n % 2 == 0) {
        wywolania.push_back({2 * m, n / 2});
        return A(2 * m, n / 2, wywolania);
    }
    wywolania.push_back({m, (n - 1) / 2});
    return 2 * A(m, (n - 1) / 2, wywolania) + m;
}

int main() {
    Liczba jeden = 1;
    std::vector<std::pair<Liczba, Liczba>> dane = {{3, 9}, {32, 32}, {10, 15}, {1, (jeden << 100) + 1}};
    for (auto [m, n] : dane) {

                if (i > 0) {
                    lista += ", ";
                }
                lista += "(" + na_napis(wywolania[i].first) + ", " + na_napis(wywolania[i].second) + ")";
            }
            lista += "]";
        }
        std::println("{} {} {} {}", na_napis(m), na_napis(n), wywolania.size(), lista);
    }
}
```

```java
// Java 25
import java.math.BigInteger;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    static final BigInteger DWA = BigInteger.TWO;

    // wywolania - lista, do której dopisujemy argumenty kolejnych wywołań rekurencyjnych
    // (n = 2^100 + 1 nie mieści się w long - używamy BigInteger)
    static BigInteger A(BigInteger m, BigInteger n, List<String> wywolania) {
        if (n.equals(BigInteger.ONE)) {
            return m;
        }
        if (n.mod(DWA).signum() == 0) {
            wywolania.add("(" + m.multiply(DWA) + ", " + n.divide(DWA) + ")");
            return A(m.multiply(DWA), n.divide(DWA), wywolania);
        }
        BigInteger polowa = n.subtract(BigInteger.ONE).divide(DWA);
        wywolania.add("(" + m + ", " + polowa + ")");
        return A(m, polowa, wywolania).multiply(DWA).add(m);
    }

    public static void main(String[] args) throws Exception {
        BigInteger[][] dane = {
            {BigInteger.valueOf(3), BigInteger.valueOf(9)},
            {BigInteger.valueOf(32), BigInteger.valueOf(32)},
            {BigInteger.valueOf(10), BigInteger.valueOf(15)},
            {BigInteger.ONE, DWA.pow(100).add(BigInteger.ONE)}
        };
        for (BigInteger[] para : dane) {
            List<String> wywolania = new ArrayList<>();
            A(para[0], para[1], wywolania);
            // lista wypisana jak w Pythonie: [(3, 4), (6, 2)]
            String lista = wywolania.size() < 10 ? "[" + String.join(", ", wywolania) + "]" : "";
            System.out.println(para[0] + " " + para[1] + " " + wywolania.size() + " " + lista);
        }
    }
}
```

> **Treść zadania:** Rozważamy dodawanie pisemne dwóch liczb w systemie dziesiętnym i liczymy przeniesienia, np. przy dodawaniu 27732 + 72619 = 100351 występują 4 przeniesienia.

> - 2.2 — Zapisz (w pseudokodzie lub wybranym języku) algorytm, który dla dwóch dodatnich liczb całkowitych a i b o tej samej liczbie cyfr oblicza liczbę p przeniesień w trakcie ich dodawania pisemnego (dla 27732 i 72619 wynik to 4).

```py
def przeniesienia(a, b):
    # a i b mają tyle samo cyfr; operujemy tylko na liczbach całkowitych
    w_pamieci = 0
    p = 0
    while a > 0:
        suma = a % 10 + b % 10 + w_pamieci
        if suma >= 10:
            w_pamieci = 1
            p += 1
        else:
            w_pamieci = 0
        a = a // 10
        b = b // 10
    return p


print(przeniesienia(27732, 72619))  # przykład z treści: 4
for a, b in [(37932, 12528), (88765, 11111), (456789, 222222)]:
    print(a, b, przeniesienia(a, b))
```

> Wspólna liczba wystąpień litery x w słowach s1, s2 to W(x, s1, s2) = min(d(x, s1), d(x, s2)), gdzie d(x, s) to liczba wystąpień x w s. najwieksza = -1
```py
with open("pary.txt") as file:
    for linia in file:
        s1, s2 = linia.split()
        suma = 0
```

## 3.3

> **Treść zadania:** Dane jak w zadaniu 3.1.
>
> Prefiksosufiks pary słów s1, s2 to słowo, które jest początkiem s1 i końcem s2 albo początkiem s2 i końcem s1. Wypisz wszystkie pary z pliku, których najdłuższy prefiksosufiks ma co najmniej 5 liter, a przy każdej podaj długość tego najdłuższego prefiksosufiksu. Przykład: dla `aabbbca caacaab` najdłuższy prefiksosufiks to `aab` (3 litery) — para nie spełnia warunku; dla `abbaabaa baabaabba` jest to `baabaa` (6) — spełnia; dla pliku przykładowego: `ececdddeed dddeedd 6`.

```py
def najdluzszy_prefiksosufiks(s1, s2):
    for dlugosc in range(min(len(s1), len(s2)), 0, -1):
        if s1[:dlugosc] == s2[-dlugosc:] or s2[:dlugosc] == s1[-dlugosc:]:
            return dlugosc
    return 0


with open("pary.txt") as file:
    for linia in file:
        s1, s2 = linia.split()
        dlugosc = najdluzszy_prefiksosufiks(s1, s2)
        if dlugosc >= 5:
            print(s1, s2, dlugosc)
```

```cpp
// C++23
#include <algorithm>
#include <fstream>
#include <print>
#include <string>

int najdluzszy_prefiksosufiks(const std::string& s1, const std::string& s2) {
    for (int dlugosc = std::min(s1.size(), s2.size()); dlugosc > 0; dlugosc--) {
        // prefiks jednego słowa równy sufiksowi drugiego
        if (s1.substr(0, dlugosc) == s2.substr(s2.size() - dlugosc)
                || s2.substr(0, dlugosc) == s1.substr(s1.size() - dlugosc)) {
            return dlugosc;
        }
    }
    return 0;
}

int main() {
    std::ifstream plik("pary.txt");
    std::string s1, s2;
    while (plik >> s1 >> s2) {
        int dlugosc = najdluzszy_prefiksosufiks(s1, s2);
        if (dlugosc >= 5) {
            std::println("{} {} {}", s1, s2, dlugosc);
        }
    }
}
```
