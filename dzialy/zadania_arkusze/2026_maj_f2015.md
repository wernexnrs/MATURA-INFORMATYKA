## 1.1, 1.2 i 1.3

```py
wywolania = []                      # argumenty kolejnych wywołań rekurencyjnych


def A(m, n, pierwsze=True):
    if not pierwsze:
        wywolania.append((m, n))
    if n == 1:
        return m
    if n % 2 == 0:
        return A(2 * m, n // 2, False)
    return 2 * A(m, (n - 1) // 2, False) + m


# 1.1
for m, n in [(32, 17), (2 ** 5, 2 ** 5), (10, 15), (1, 2 ** 100 + 1)]:
    wywolania = []
    A(m, n)
    print(m, n, len(wywolania), wywolania if n < 1000 else "")

# 1.2 - funkcja oblicza iloczyn m * n
print(A(1, 777), A(2 * 10 ** 9, 256 * 10 ** 9))

# 1.3 - drugie argumenty kolejnych wywołań dla n = 2^k oraz n = 2^k - 1
for k in range(3, 7):
    for n in [2 ** k, 2 ** k - 1]:
        wywolania = []
        A(1, n)
        print(k, n, len(wywolania), [drugi for pierwszy, drugi in wywolania])
```

**Odpowiedź:** 1.1: (32, 17): 4 wywołania – A(32, 8), A(64, 4), A(128, 2), A(256, 1); (2⁵, 2⁵): 5 – A(2⁶, 2⁴), A(2⁷, 2³), A(2⁸, 2²), A(2⁹, 2¹), A(2¹⁰, 2⁰); (10, 15): 3 – A(10, 7), A(10, 3), A(10, 1); (1, 2¹⁰⁰ + 1): 100. 1.2: 777; 512 · 10¹⁸. 1.3: n = 2ᵏ: k wywołań, i-te ma drugi argument 2ᵏ⁻ⁱ; n = 2ᵏ − 1: k − 1 wywołań, i-te ma drugi argument 2ᵏ⁻ⁱ − 1.

## 2.1 i 2.2

```py
# 2.2
def przeniesienia(a, b):
    w_pamieci = 0
    p = 0
    while a > 0:
        suma = a % 10 + b % 10 + w_pamieci
        if suma >= 10:
            w_pamieci = 1
            p += 1
        else:
            w_pamieci = 0
        a //= 10
        b //= 10
    return p


# 2.1
for a, b in [(88765, 11111), (456789, 222222)]:
    print(a, b, przeniesienia(a, b))
```

**Odpowiedź:** 2.1: 88765 i 11111 – 0; 456789 i 222222 – 3. 2.2: funkcja `przeniesienia`.

pary = []
with open("pary.txt") as file:
    for linia in file:
        if linia.strip():

    return sum(ord(c) for c in s)

## 4.2

```py
pary = []
with open("pary.txt") as file:
    for linia in file:
        if linia.strip():
            pary.append(linia.split())


def W(x, s1, s2):
    return min(s1.count(x), s2.count(x))


def suma_wspolnych(para):
    s1, s2 = para
    return sum(W(x, s1, s2) for x in "abcdefghijklmnopqrstuvwxyz")


najwiecej_b = max(W("b", s1, s2) for s1, s2 in pary)
b = max(pary, key=suma_wspolnych)

with open("wyniki4.txt", "a") as wynik:
    print("4.2", file=wynik)
    print("a)", file=wynik)
    for s1, s2 in pary:
        if W("b", s1, s2) == najwiecej_b:
            print(s1, s2, file=wynik)
    print("b)", b[0], b[1], file=wynik)
```

**Odpowiedź:** a) abbbbaabbbabbba abbbbbbbbbabbabbaba oraz aababbbababbbbbbaab bbbbaabbababababa; b) aacbcccaacacbcabac cccccaaaacaccbabcba.

## 4.3

```py
pary = []
with open("pary.txt") as file:
    for linia in file:
        if linia.strip():
            pary.append(linia.split())


def prefiksosufiks(s1, s2):
    najdluzszy = 0
    for d in range(1, min(len(s1), len(s2)) + 1):
        if s1[:d] == s2[-d:] or s2[:d] == s1[-d:]:
            najdluzszy = d
    return najdluzszy


with open("wyniki4.txt", "a") as wynik:
    print("4.3", file=wynik)
    for s1, s2 in pary:
        d = prefiksosufiks(s1, s2)
        if d >= 5:
            print(s1, s2, d, file=wynik)
```
