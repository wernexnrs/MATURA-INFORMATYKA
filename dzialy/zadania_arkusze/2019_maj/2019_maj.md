```py
def pierwsza_parzysta(A):
    # wyszukiwanie binarne - złożoność logarytmiczna

        if A[s] % 2 == 1:
            p = s + 1
        else:
            k = s
    return A[p]
```

>
> Przykłady: dla 3, 7, 4, 6, 10, 2, 5 odpowiedź to 4, 4, 2; dla 5, 70, 28, 42, 98, 1 — 70, 4, 14. Dla pliku przykładowego: 90, 5, 10.

```py
from math import gcd

with open("liczby.txt") as file:
    liczby = [int(line) for line in file]

najdluzszy = 0

for i in range(len(liczby)):
    nwd = liczby[i]
    j = i
    # rozszerzamy ciąg zaczynający się od i, dopóki NWD wszystkich liczb jest większy od 1
    while j + 1 < len(liczby) and gcd(nwd, liczby[j + 1]) > 1:
        j += 1
        nwd = gcd(nwd, liczby[j])
    if j - i + 1 > najdluzszy:
        najdluzszy = j - i + 1
        pierwsza = liczby[i]
        dzielnik = nwd

print(pierwsza, najdluzszy, dzielnik)
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;

public class Zadanie {
    static long gcd(long a, long b) {
        while (b != 0) {
            long r = a % b;
            a = b;
            b = r;
        }
        return a;
    }

    public static void main(String[] args) throws Exception {
        var liczby = new ArrayList<Long>();
        for (String line : Files.readAllLines(Path.of("liczby.txt")))
            if (!line.isBlank())
                liczby.add(Long.parseLong(line.trim()));

        int n = liczby.size();
        int najdluzszy = 0;
        long pierwsza = 0, dzielnik = 0;

        for (int i = 0; i < n; i++) {
            long nwd = liczby.get(i);
            int j = i;
            // rozszerzamy ciąg zaczynający się od i, dopóki NWD wszystkich liczb jest większy od 1
            while (j + 1 < n && gcd(nwd, liczby.get(j + 1)) > 1) {
                j++;
                nwd = gcd(nwd, liczby.get(j));
            }
            if (j - i + 1 > najdluzszy) {
                najdluzszy = j - i + 1;
                pierwsza = liczby.get(i);
                dzielnik = nwd;
            }
        }

        System.out.println(pierwsza + " " + najdluzszy + " " + dzielnik);
    }
}
```

```cpp
// C++23
#include <algorithm>
#include <fstream>
#include <print>
#include <string>
#include <vector>

// liczba z przecinkiem dziesiętnym, np. "12,5" -> 12.5
double na_liczbe(std::string tekst) {
    std::ranges::replace(tekst, ',', '.');
    return std::stod(tekst);
}

int main() {
    std::ifstream plik("pogoda.txt");
    std::string linia;
    std::getline(plik, linia);  // nagłówek
    std::vector<double> temperatury;
    while (std::getline(plik, linia)) {
        if (linia.empty() || linia == "\r")
            continue;
        // temperatura jest między pierwszym a drugim średnikiem
        size_t a = linia.find(';'), b = linia.find(';', a + 1);
        temperatury.push_back(na_liczbe(linia.substr(a + 1, b - a - 1)));
    }

    int najdluzszy = 0;
    int dlugosc = 0;  // liczba kolejnych dni cieplejszych od dnia poprzedniego
    int koniec = 0;

    for (size_t i = 1; i < temperatury.size(); i++) {
        if (temperatury[i] > temperatury[i - 1])
            dlugosc++;
        else
            dlugosc = 0;
        if (dlugosc > najdluzszy) {
            najdluzszy = dlugosc;
            koniec = i + 1;  // dni numerujemy od 1
        }
    }

    std::println("{} {}", koniec - najdluzszy + 1, koniec);
}
```

```java
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.TreeMap;

public class Zadanie {
    // round(x, 2) z Pythona: zaokrąglenie do 2 miejsc po przecinku
    static double zaokraglij(double x) {
        return new BigDecimal(x).setScale(2, RoundingMode.HALF_EVEN).doubleValue();
    }

    public static void main(String[] args) throws Exception {
        var suma = new TreeMap<String, Long>();  // TreeMap trzyma klucze posortowane
        var ile = new TreeMap<String, Long>();

        var linie = Files.readAllLines(Path.of("pogoda.txt"));
        for (String line : linie.subList(1, linie.size())) {  // pomijamy nagłówek
            if (line.isBlank())
                continue;
            String[] czesci = line.strip().split(";");  // dzien, temperatura, opad, kategoria, wielkosc
            String kategoria = czesci[3], wielkosc = czesci[4];
            if (Integer.parseInt(czesci[0]) <= 300 && !kategoria.equals("0")) {
                String rodzaj = kategoria + wielkosc;
                suma.merge(rodzaj, Long.parseLong(czesci[2]), Long::sum);
                ile.merge(rodzaj, 1L, Long::sum);
            }
        }

        for (var e : suma.entrySet())
            System.out.println(e.getKey() + " " + zaokraglij((double) e.getValue() / ile.get(e.getKey())));
        // na podstawie zestawienia należy jeszcze wykonać wykres kolumnowy
    }
}
```
