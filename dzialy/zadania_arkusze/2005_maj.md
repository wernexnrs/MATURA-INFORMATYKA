> **Treść zadania:** Zadanie 1 (szeregi nieskończone). Wartość ln 2 można przybliżać sumą nieskończoną:
> ln 2 = 2/3 · (1 + 1/3·1/9 + 1/5·1/9² + 1/7·1/9³ + 1/9·1/9⁴ + ...).
> Oznaczmy l₀ = 2/3 oraz dla n ≥ 1: lₙ = 2/3 · (1 + 1/3·1/9 + ... + 1/(2n+1)·1/9ⁿ).
>
> Wypełnij tabelę wartości lₙ dla n = 0, 1, 2, 3. Podaj zależność między lₙ a lₙ₋₁ dla n ≥ 1 oraz wzór rekurencyjny na różnicę rₙ = lₙ − lₙ₋₁ (n > 0).

```py
import math


def ln2(eps):
    # Dane: eps > 0, Wynik: przybliżona wartość ln 2
    n = 1
    r = 2 / 81           # różnica l1 - l0
    l = 2 / 3 + r        # l1
    while r >= eps:
        n += 1
        r = r * (2 * n - 1) / ((2 * n + 1) * 9)
        l = l + r
    return l


for eps in [0.1, 0.001, 0.000001, 0.000000001]:
```

```java
// Java 25
public class Zadanie {
    static double ln2(double eps) {
        // Dane: eps > 0, Wynik: przybliżona wartość ln 2
        int n = 1;
        double r = 2.0 / 81;    // różnica l1 - l0
        double l = 2.0 / 3 + r; // l1
        while (r >= eps) {
            n++;
            r = r * (2 * n - 1) / ((2 * n + 1) * 9);
            l = l + r;
        }
        return l;
    }

    // liczba w zapisie jak w Pythonie (1e-06 zamiast javowego 1.0E-6)
    static String zapis(double x) {
        String s = Double.toString(x);
        if (!s.contains("E")) return s;
        String[] czesci = s.split("E");
        String mantysa = czesci[0].endsWith(".0") ? czesci[0].substring(0, czesci[0].length() - 2) : czesci[0];
        int wykladnik = Integer.parseInt(czesci[1]);
        return mantysa + "e" + (wykladnik < 0 ? "-" : "+") + String.format("%02d", Math.abs(wykladnik));
    }

    public static void main(String[] args) {
        for (double eps : new double[]{0.1, 0.001, 0.000001, 0.000000001})
            System.out.println(zapis(eps) + " " + ln2(eps) + " " + Math.log(2));
    }
}
```

> **Treść zadania:** Dla zbioru reguł `L₁ → F₁S₁, ..., Lₚ → FₚSₚ` z zadania 2 dana jest rekurencyjna funkcja `sprawdź(napis, start)`:
>
> ```
> jeśli długość napisu nie jest potęgą 2 → NIE
> jeśli napis = start → TAK
> jeśli długość napisu = 1 → NIE
> podziel napis na równe połowy napis1, napis2
> dla i = 1..p: jeśli Li = start oraz sprawdź(napis1, Fi) = TAK
>               i sprawdź(napis2, Si) = TAK → TAK
> NIE
> ```
>
> Uzupełnij specyfikację funkcji (czym są dane `napis` i `start`). Dla reguł `A → BC`, `A → CD`, `B → AD`, `C → BA`, `D → AA`, `D → BB` oraz `napis = BCAAADCD`, `start = A` podaj parametry wszystkich wywołań rekurencyjnych i odpowiedź, jaką zwróci funkcja.

```py
reguly = [("A", "BC"), ("A", "CD"), ("B", "AD"), ("C", "BA"), ("D", "AA"), ("D", "BB")]


def potega_dwojki(n):
    while n % 2 == 0:
        n //= 2
    return n == 1


def sprawdz(napis, start):
    # Dane: napis - ciąg n liter, start - litera nowo powstałego organizmu
    # Wynik: czy napis to organizm, który można uzyskać z organizmu start
    print("sprawdz(", napis, ",", start, ")")
    if not potega_dwojki(len(napis)):
        return False
    if napis == start:
        return True
    if len(napis) == 1:
        return False
    polowa = len(napis) // 2
    napis1 = napis[:polowa]
    napis2 = napis[polowa:]
    for lewa, prawa in reguly:
        if lewa == start:
            wynik1 = sprawdz(napis1, prawa[0])
            wynik2 = sprawdz(napis2, prawa[1])
            if wynik1 and wynik2:
                return True
    return False


print(sprawdz("BCAAADCD", "A"))
```

```java
    static double kosztD2(double N) { return 0.5 * Math.sqrt(N) + 5 * f(N); }

    static double zaokraglij2(double x) { return Math.round(x * 100) / 100.0; }

    public static void main(String[] args) {
        // sprawdzamy wszystkie N z [1, 7000] z krokiem 0,01
        String poprzednia = "";
        for (int k = 100; k <= 700000; k++) {
            double N = k / 100.0;
            String firma = kosztD1(N) <= kosztD2(N) ? "D1" : "D2";
            if (!firma.equals(poprzednia)) {
                System.out.println("od N = " + N + " wybieramy " + firma);
                poprzednia = firma;
            }
        }

        System.out.println();
        System.out.println("N D1 D2");
        for (int N : new int[]{100, 1000, 5000})
            System.out.println(N + " " + zaokraglij2(kosztD1(N)) + " " + zaokraglij2(kosztD2(N)));
    }
}
```

> **Treść zadania:** Najpopularniejszy element ciągu to element występujący w nim najwięcej razy (przy remisie można podać dowolny z nich, np. dla 1, 3, 5, 1, 3 poprawne są 1 i 3). def najpopularniejszy(ciag):
```py
    # sortowanie O(n log n), potem jedno przejście po posortowanym ciągu O(n)
    ciag = sorted(ciag)
    najlepszy, najwiecej = ciag[0], 0
    licznik = 0
    for i in range(len(ciag)):
        if i > 0 and ciag[i] == ciag[i - 1]:
            licznik += 1
        else:
            licznik = 1
        if licznik > najwiecej:
            najwiecej = licznik
            najlepszy = ciag[i]
    return najlepszy, najwiecej


for nazwa in ["dane5-1.txt", "dane5-2.txt", "dane5-3.txt"]:
```
