import math


def ln2(eps):
    # Dane: eps > 0, Wynik: przybliżona wartość ln 2
    n = 1
    r = 2 / 81           # różnica l1 - l0
    l = 2 / 3 + r        # l1
    while r >= eps:
        n += 1
        r = r * (2 * n - 1) / ((2 * n + 1) * 9)
        l = l + r
    return l


for eps in [0.1, 0.001, 0.000001, 0.000000001]:
    print(eps, ln2(eps), math.log(2))
```

**Odpowiedź:** algorytm sumuje kolejne wyrazy r(n), dopóki różnica dwóch ostatnich przybliżeń (czyli r(n)) nie jest mniejsza niż ε; wynik to ostatnie przybliżenie l(n).

def wiek(n):
    # organizm o długości n istnieje tylko, gdy n jest potęgą dwójki; wiek = log2(n)
    lata = 0
    while n % 2 == 0:
        n //= 2
        lata += 1
    if n == 1:
        return lata
    return -1  # nie ma takiego organizmu


def ile_wielkosci(n, m):
    # liczba potęg dwójki w przedziale (n, m]
    ile = 0
    potega = 1
    while potega <= m:
        if potega > n:
            ile += 1
        potega *= 2
    return ile


print(wiek(1), wiek(8), wiek(12), wiek(1024))
print(ile_wielkosci(3, 100))
```

**Odpowiedź:** organizm o długości n istnieje wtedy i tylko wtedy, gdy n = 2^k, a jego wiek to k = log2 n; w przedziale (n, m] jest ⌊log2 m⌋ − ⌊log2 n⌋ poprawnych wielkości.

reguly = [("A", "BC"), ("A", "CD"), ("B", "AD"), ("C", "BA"), ("D", "AA"), ("D", "BB")]


def potega_dwojki(n):
    while n % 2 == 0:
        n //= 2
    return n == 1


def sprawdz(napis, start):
    # Dane: napis - ciąg n liter, start - litera nowo powstałego organizmu
    # Wynik: czy napis to organizm, który można uzyskać z organizmu start
    print("sprawdz(", napis, ",", start, ")")
    if not potega_dwojki(len(napis)):
        return False
    if napis == start:
        return True
    if len(napis) == 1:
        return False
    polowa = len(napis) // 2
    napis1 = napis[:polowa]
    napis2 = napis[polowa:]
    for lewa, prawa in reguly:
        if lewa == start:
            wynik1 = sprawdz(napis1, prawa[0])
            wynik2 = sprawdz(napis2, prawa[1])
            if wynik1 and wynik2:
                return True
    return False


print(sprawdz("BCAAADCD", "A"))
```

**Odpowiedź:** wywołania: (BCAAADCD, A), (BCAA, B), (BC, A), (B, B), (C, C), (AA, D), (A, A), (A, A), (ADCD, C), (AD, B), (A, A), (D, D), (CD, A), (C, B), (D, C), (C, C), (D, D); wynik: TAK

import math


def f(N):
    m = 0.0001 * N
    return 10 * m ** 3 + 7 * m ** 2 + 0.1 * m + 0.1


def koszt_d1(N):
    return 0.01 * N + f(N)


def koszt_d2(N):
    return 0.5 * math.sqrt(N) + 5 * f(N)


# sprawdzamy wszystkie N z [1, 7000] z krokiem 0,01
poprzednia = None
for k in range(100, 700001):
    N = k / 100
    firma = "D1" if koszt_d1(N) <= koszt_d2(N) else "D2"
    if firma != poprzednia:
        print("od N =", N, "wybieramy", firma)
        poprzednia = firma

print()
print("N", "D1", "D2")
for N in [100, 1000, 5000]:
    print(N, round(koszt_d1(N), 2), round(koszt_d2(N), 2))
```

def najlepsza_suma(ciag):
    najlepsza = ciag[0]
    biezaca = 0
    for x in ciag:
        biezaca = max(biezaca + x, x)
        najlepsza = max(najlepsza, biezaca)
    return najlepsza

```py
def najpopularniejszy(ciag):
    # sortowanie O(n log n), potem jedno przejście po posortowanym ciągu O(n)
    ciag = sorted(ciag)
    najlepszy, najwiecej = ciag[0], 0
    licznik = 0
    for i in range(len(ciag)):
        if i > 0 and ciag[i] == ciag[i - 1]:
            licznik += 1
        else:
            licznik = 1
        if licznik > najwiecej:
            najwiecej = licznik
            najlepszy = ciag[i]
    return najlepszy, najwiecej


for nazwa in ["dane5-1.txt", "dane5-2.txt", "dane5-3.txt"]:
    with open(nazwa) as file:
        ciag = [int(linia) for linia in file if linia.strip()]
    print(nazwa, *najpopularniejszy(ciag))
```

**Odpowiedź:** dane5-1.txt: -22 (lub -15, 6, 19, 24), dane5-2.txt: -18 (lub 18), dane5-3.txt: 22; czas O(n log n)
