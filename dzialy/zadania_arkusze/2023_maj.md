```py
# biblioteczka jako słownik: (półka, przegródka) -> numer książki
def wstaw(B, ksiazka):
    i, j = 0, 1
    while (i, j) in B:
        if ksiazka < B[(i, j)]:
            i, j = i + 1, 2 * j - 1
        else:
            i, j = i + 1, 2 * j
    B[(i, j)] = ksiazka


def A(B, i, j, wynik):
    wynik.append(B[(i, j)])
    if (i + 1, 2 * j - 1) in B:
        A(B, i + 1, 2 * j - 1, wynik)
    if (i + 1, 2 * j) in B:
        A(B, i + 1, 2 * j, wynik)
    return wynik


# 1.1
B = {}
for k in [14, 18, 12, 9, 20, 15, 17]:
    wstaw(B, k)
for i in range(5):
    print(i, [(j, B[(i, j)]) for j in range(1, 2 ** i + 1) if (i, j) in B])

# 1.2 - minimum: półki zapełnione do końca (2^0 + 2^1 + ... >= n), maksimum: każda książka na innej półce
for n in [1, 3, 4, 7, 16, 31, 32]:
    polki = 0
    miejsca = 0
    while miejsca < n:
        miejsca += 2 ** polki
        polki += 1
    print(n, polki, n)

# 1.3
a = {(0, 1): 9, (1, 1): 2, (1, 2): 12, (2, 3): 10, (2, 4): 14, (3, 7): 13, (3, 8): 15}
b = {(0, 1): 10, (1, 1): 8, (1, 2): 15, (2, 1): 4, (2, 3): 12, (3, 2): 6, (3, 6): 13}
print(A(a, 0, 1, []))
print(A(b, 0, 1, []))
```

**Odpowiedź:** 1.1: półka 0: 14; półka 1: 12, 18; półka 2: B[2,1] = 9, B[2,3] = 15, B[2,4] = 20; półka 3: B[3,6] = 17. 1.2 (min, max): n = 7: 3, 7; n = 16: 5, 16; n = 31: 5, 31; n = 32: 6, 32; n = 2^k − 1: k, 2^k − 1. 1.3: a) 9, 2, 12, 10, 14, 13, 15; b) 10, 8, 4, 6, 15, 12, 13.

```py
# 2.1 - liczba bloków bez funkcji wbudowanych
def bloki(n):
    b = 1
    poprzednia = n % 2
    n = n // 2
    while n > 0:
        cyfra = n % 2
        if cyfra != poprzednia:
            b += 1
            poprzednia = cyfra
        n = n // 2
    return b


def na_dziesietny(napis):
    liczba = 0
    for c in napis:
        liczba = liczba * 2 + int(c)
    return liczba


def na_binarny(n):
    if n == 0:
        return "0"
    wynik = ""
    while n > 0:
        wynik = str(n % 2) + wynik
        n = n // 2
    return wynik


print("2.1", bloki(67), bloki(245))

with open("bin.txt") as file:
    liczby = [linia.strip() for linia in file if linia.strip()]

# 2.2
print("2.2", sum(1 for x in liczby if bloki(na_dziesietny(x)) <= 2))

# 2.3
print("2.3", na_binarny(max(na_dziesietny(x) for x in liczby)))

# 2.4
print("2.4", (123 ^ 0b101101) ^ 0x2D)

# 2.5
with open("wyniki2_5.txt", "w") as wyniki:
    for x in liczby:
        p = na_dziesietny(x)
        wyniki.write(na_binarny(p ^ (p // 2)) + "\n")
```

**Odpowiedź:** 2.1: funkcja `bloki` (dla 67 → 3, dla 245 → 5). 2.2: 10. 2.3: 1110100011100011100. 2.4: 123. 2.5: plik `wyniki2_5.txt`: 1001011000, 1111000001, 1110001000, 1101100001111010, … (100 wierszy).

```py
def rosnaco_malejacy(c):
    n = len(c)
    if n < 4:
        return False
    for k in range(2, n - 1):  # a1..ak rosnący, a(k+1)..an malejący
        rosnie = all(c[i] < c[i + 1] for i in range(k - 1))
        maleje = all(c[i] > c[i + 1] for i in range(k, n - 1))
        if rosnie and maleje:
            return True
    return False


# 3.3

# 3.4 - ciąg rosnąco-malejący cyfr ma najwyżej 20 elementów (10 rosnących + 10 malejących)

najlepsza_dlugosc = 0
for i in range(len(cyfry)):
    for dlugosc in range(4, 21):
        if i + dlugosc <= len(cyfry) and dlugosc > najlepsza_dlugosc:

print("3.4", najlepszy_start + 1)
```

```py
def na_dziesietny(napis, podstawa):
    liczba = 0
    for c in napis:
        liczba = liczba * podstawa + int(c)
    return liczba


for (x, px), (y, py) in [(("2011", 3), ("134", 6)), (("134", 5), ("134", 6)), (("2222", 3), ("1111", 6))]:
    a, b = na_dziesietny(x, px), na_dziesietny(y, py)
    print(a, "<" if a < b else (">" if a > b else "="), b)
```

**Odpowiedź:** (2011)₃ = (134)₆ (58 = 58), (134)₅ < (134)₆ (44 < 58), (2222)₃ < (1111)₆ (80 < 259).
