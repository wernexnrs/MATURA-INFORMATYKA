## 1.1, 1.2 i 1.3

> **Treść zadania (Biblioteczka):** Półki są numerowane od 0 (najwyższa); półka i ma 2ⁱ przegródek numerowanych 1..2ⁱ, a B[i, j] to j-ta przegródka na i-tej półce (jedna książka na przegródkę). Adam wstawia książkę, zaczynając od B[0, 1]: jeśli przegródka jest pusta, kładzie ją tam; w przeciwnym razie, gdy numer wstawianej książki jest mniejszy od numeru stojącej, próbuje B[i+1, 2j−1], a gdy większy – B[i+1, 2j]. Np. po wstawieniu 10, 2, 15, 13, 1, 5, 25 półka 0 to [10], półka 1 to [2, 15], półka 2 to [1, 5, 13, 25].
>
> - **1.1** — Podaj zawartość biblioteczki po wstawieniu (do pustej) kolejno książek 14, 18, 12, 9, 20, 15, 17.
> - **1.2** — Uzupełnij tabelę: ile minimalnie, a ile maksymalnie półek musi mieć biblioteczka, by zmieścić n książek, tak aby na ostatniej półce była co najmniej jedna książka – dla n = 4, 7, 16, 32 oraz n = 2ᵏ − 1 (k > 0). Przykłady: n=1 → 1 i 1; n=2 → 2 i 2; n=3 → 2 i 3.
> - **1.3** — Algorytm `A(i, j)` (start: `A(0, 1)`) wypisuje numer książki z B[i, j], a następnie, jeśli niepuste, wywołuje `A(i+1, 2j−1)` i `A(i+1, 2j)` (dla przykładu wyżej wypisze 10, 2, 1, 5, 15, 13, 25). Podaj, co wypisze dla dwóch biblioteczek z arkusza: a) półki [9], [2, 12], [10, 14], [13, 15]; b) [10], [8, 15], [4, 12], [6, 13] (rozmieszczenie w przegródkach zgodne z regułą wstawiania).

```py
# biblioteczka jako słownik: (półka, przegródka) -> numer książki
def wstaw(B, ksiazka):
    i, j = 0, 1
    while (i, j) in B:
        if ksiazka < B[(i, j)]:
            i, j = i + 1, 2 * j - 1
        else:
            i, j = i + 1, 2 * j
    B[(i, j)] = ksiazka


def A(B, i, j, wynik):
    wynik.append(B[(i, j)])
    if (i + 1, 2 * j - 1) in B:
        A(B, i + 1, 2 * j - 1, wynik)
    if (i + 1, 2 * j) in B:
        A(B, i + 1, 2 * j, wynik)
    return wynik
```

```java
import java.util.*;

public class Zadanie {
    // biblioteczka jako słownik: (półka, przegródka) -> numer książki
    record Miejsce(int i, int j) {}

    static void wstaw(Map<Miejsce, Integer> B, int ksiazka) {
        int i = 0, j = 1;
        while (B.containsKey(new Miejsce(i, j))) {
            if (ksiazka < B.get(new Miejsce(i, j)))
                j = 2 * j - 1;
            else
                j = 2 * j;
            i = i + 1;
        }
        B.put(new Miejsce(i, j), ksiazka);
    }

    static List<Integer> A(Map<Miejsce, Integer> B, int i, int j, List<Integer> wynik) {
        wynik.add(B.get(new Miejsce(i, j)));
        if (B.containsKey(new Miejsce(i + 1, 2 * j - 1))) A(B, i + 1, 2 * j - 1, wynik);
        if (B.containsKey(new Miejsce(i + 1, 2 * j))) A(B, i + 1, 2 * j, wynik);
        return wynik;
    }

    static Map<Miejsce, Integer> biblioteczka(int[][] dane) {
        var B = new HashMap<Miejsce, Integer>();
        for (var d : dane) B.put(new Miejsce(d[0], d[1]), d[2]);
        return B;
    }

    public static void main(String[] args) throws Exception {
        // 1.1
        var B = new HashMap<Miejsce, Integer>();
        for (int k : new int[] {14, 18, 12, 9, 20, 15, 17}) wstaw(B, k);
        for (int i = 0; i < 5; i++) {
            var polka = new ArrayList<String>();  // "(przegródka, książka)"
            for (int j = 1; j <= (1 << i); j++)
                if (B.containsKey(new Miejsce(i, j))) polka.add("(" + j + ", " + B.get(new Miejsce(i, j)) + ")");
            System.out.println(i + " " + polka);
        }

        // 1.2 - minimum: półki zapełnione do końca (2^0 + 2^1 + ... >= n), maksimum: każda książka na innej półce
        for (int n : new int[] {1, 3, 4, 7, 16, 31, 32}) {
            int polki = 0;
            int miejsca = 0;
            while (miejsca < n) {
                miejsca += 1 << polki;
                polki++;
            }
            System.out.println(n + " " + polki + " " + n);
        }

        // 1.3
        var a = biblioteczka(new int[][] {{0, 1, 9}, {1, 1, 2}, {1, 2, 12}, {2, 3, 10}, {2, 4, 14}, {3, 7, 13}, {3, 8, 15}});

        System.out.println(A(a, 0, 1, new ArrayList<>()));
        System.out.println(A(b, 0, 1, new ArrayList<>()));
    }
}
```

```java
import java.io.*;
import java.nio.file.*;
import java.util.*;

public class Zadanie {
    // 2.1 - liczba bloków bez funkcji wbudowanych
    static int bloki(long n) {
        int b = 1;
        long poprzednia = n % 2;
        n = n / 2;
        while (n > 0) {
            long cyfra = n % 2;
            if (cyfra != poprzednia) {
                b++;
                poprzednia = cyfra;
            }
            n = n / 2;
        }
        return b;

    static long naDziesietny(String napis) {

        return wynik;
```

```cpp
// C++23
#include <algorithm>
#include <fstream>
#include <print>
#include <vector>

bool rosnaco_malejacy(const std::vector<int>& c) {
    int n = c.size();
    if (n < 4) return false;
    for (int k = 2; k < n - 1; k++) {  // a1..ak rosnący, a(k+1)..an malejący
        bool rosnie = true, maleje = true;
        for (int i = 0; i < k - 1; i++)
            if (!(c[i] < c[i + 1])) rosnie = false;
        for (int i = k; i < n - 1; i++)
            if (!(c[i] > c[i + 1])) maleje = false;
        if (rosnie && maleje) return true;
    }
    return false;
}
```

```java
    static boolean rosnacoMalejacy(List<Integer> c) {
        int n = c.size();
        if (n < 4) return false;
        for (int k = 2; k < n - 1; k++) {  // a1..ak rosnący, a(k+1)..an malejący

            for (int i = 0; i < k - 1; i++)
                if (!(c.get(i) < c.get(i + 1))) rosnie = false;
            for (int i = k; i < n - 1; i++)
                if (!(c.get(i) > c.get(i + 1))) maleje = false;
            if (rosnie && maleje) return true;
        }
        return false;
    }
```
