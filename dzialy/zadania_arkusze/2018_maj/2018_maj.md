```py
def algorytm(n):
    p = 1
    q = n
    while p < q:
        s = (p + q) // 2
        if s * s * s < n:
            p = s + 1
        else:
            q = s
    return p


# 1.1
for n in [28, 64, 80]:
    print(n, algorytm(n))

# 1.2 - najmniejsze i największe n, dla którego p = 10
liczby = [n for n in range(1, 2000) if algorytm(n) == 10]
print(min(liczby), max(liczby))
```

**Odpowiedź:** 1.1: 28 → 4, 64 → 4, 80 → 5; 1.2: najmniejsza 730, największa 1000

```py
def skrajnie_lewy(n, X, Y):
    k = 0
    for i in range(1, n):
        if X[i] / Y[i] < X[k] / Y[k]:
            k = i
    return X[k], Y[k]


X = [3, -2, 5, -4]
Y = [4, 3, 2, 1]
print(skrajnie_lewy(len(X), X, Y))
```

**Odpowiedź:** przykładowy algorytm – szukamy szczytu o najmniejszym ilorazie X[i]/Y[i] (dla przykładu w kodzie: (-4, 1))

```py
def sortuj_szczyty(n, X, Y):
    # sortowanie przez wybór - złożoność kwadratowa
    for i in range(n - 1):
        m = i
        for j in range(i + 1, n):
            if X[j] / Y[j] < X[m] / Y[m]:
                m = j
        X[i], X[m] = X[m], X[i]
        Y[i], Y[m] = Y[m], Y[i]


X = [3, -2, 5, -4]
Y = [4, 3, 2, 1]
sortuj_szczyty(len(X), X, Y)
print(X)
print(Y)
```

**Odpowiedź:** przykładowy algorytm – sortowanie (np. przez wybór) względem X[i]/Y[i], zamieniając jednocześnie elementy X i Y

```py
import math

with open("woda.txt") as file:
    dane = [line.split() for line in file]


def symulacja(z_wypuszczaniem):
    woda = 500000
    pierwsze_przepelnienie = None
    dni_800 = 0
    maks = 0
    for data, doplyw in dane:
        pomiar = woda  # pomiar zaraz po północy
        maks = max(maks, pomiar)
        if pomiar >= 800000:
            dni_800 += 1
        if z_wypuszczaniem and woda > 1000000:
            if pierwsze_przepelnienie is None:
                pierwsze_przepelnienie = data
            woda = 1000000
        woda -= math.ceil(pomiar * 2 / 100)  # o 8 rano wypuszczamy 2% wody z pomiaru
        woda += int(doplyw)
    return pierwsze_przepelnienie, dni_800, maks


a, b, _ = symulacja(True)
_, _, c = symulacja(False)
print("a)", a)
print("b)", b)
print("c)", c)
```

**Odpowiedź:** a) 2015-04-19, b) 188 dni, c) 1 399 242 m3

```py
from datetime import datetime

czas_awarii = {}
with open("awarie.txt") as file:
    file.readline()
    for line in file:
        zgloszenie, komputer, data, godzina, priorytet = line.split()
        czas_awarii[zgloszenie] = datetime.fromisoformat(data + " " + godzina)

ostatnia_naprawa = {}
with open("naprawy.txt") as file:
    file.readline()
    for line in file:
        zgloszenie, data, godzina, rodzaj = line.split()
        czas = datetime.fromisoformat(data + " " + godzina)
        if zgloszenie not in ostatnia_naprawa or czas > ostatnia_naprawa[zgloszenie]:
            ostatnia_naprawa[zgloszenie] = czas

najdluzej = max(ostatnia_naprawa, key=lambda z: ostatnia_naprawa[z] - czas_awarii[z])
print(najdluzej, czas_awarii[najdluzej], ostatnia_naprawa[najdluzej])
```

**Odpowiedź:** zgłoszenie 2087, awaria 2015-11-06 12:38:46, koniec ostatniej naprawy 2015-11-13 12:38:32

#### Kwerenda o nazwie 6_4(wynik)
```sql
SELECT TOP 1 [Naprawy]![Czas_naprawy]-[Awarie]![Czas_awarii] AS Wyr1, Naprawy.Numer_zgloszenia, Naprawy.Czas_naprawy, Awarie.Czas_awarii
FROM Komputery INNER JOIN (Awarie INNER JOIN Naprawy ON Awarie.Numer_zgloszenia = Naprawy.Numer_zgloszenia) ON Komputery.Numer_komputera = Awarie.Numer_komputera
```
