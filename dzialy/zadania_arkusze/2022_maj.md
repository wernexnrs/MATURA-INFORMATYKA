## 1.1 i 1.2

```py
def do_podmiany(A):
    n = len(A)
    wystapienia = [0] * (n + 1)   # wystapienia[x] - ile razy liczba x (1..n) jest w A
    k = 0
    for x in A:
        if 1 <= x <= n:
            wystapienia[x] += 1
        else:
            k += 1                # liczba spoza [1, n] na pewno do podmiany
    for x in range(1, n + 1):
        if wystapienia[x] > 1:
            k += wystapienia[x] - 1   # wszystkie powtórzenia poza jednym
    return k


for A in [[1, 3, 1], [1, 4, 2, 5], [2, 2, 2, 2, 2], [4, 2, 3, 1], [5, 4, 1, 5, 6, 8], [8, 4, 9, 6, 5, 7]]:
    print(A, do_podmiany(A))
```

**Odpowiedź:** 1.1: (1,3,1) → 1, (1,4,2,5) → 1, (2,2,2,2,2) → 4, (4,2,3,1) → 0, (5,4,1,5,6,8) → 2, (8,4,9,6,5,7) → 3. 1.2: do podmiany są liczby spoza [1, n] oraz wszystkie powtórzenia (poza jednym wystąpieniem) liczb z [1, n] — funkcja `do_podmiany`.

## 2.1, 2.2 i 2.3

```py
def algorytm(s):
    n = len(s)
    s = " " + s                   # indeksowanie od 1 jak w treści
    A = [0] * (n + 2)
    for i in range(1, n + 1):
        if s[i] == "a":
            A[i] = A[i - 1] + 1
        else:
            A[i] = A[i - 1]
    B = [0] * (n + 2)
    for j in range(n, 0, -1):
        if s[j] == "b":
            B[j] = B[j + 1] + 1
        else:
            B[j] = B[j + 1]
    k = 1
    for i in range(0, n + 1):
        if A[i] + B[i + 1] > k:
            k = A[i] + B[i + 1]
    return k


# 2.1
for s in ["aabab", "ab", "aaa", "aababb", "baabbaaab"]:
    print(s, algorytm(s))

# 2.2 - k = najdłuższy podciąg postaci a...ab...b; szukamy przykładów
print("aaaaabbbbb", algorytm("aaaaabbbbb"))
print("bbbbbaaaaa", algorytm("bbbbbaaaaa"))

# 2.3
print(algorytm("a" * 300 + "b" * 550 + "a" * 300 + "b" * 7 + "a" * 280 + "b" * 110))
```

**Odpowiedź:** 2.1: aababb → 5, baabbaaab → 6. 2.2: k = 10 np. aaaaabbbbb (wszystkie a przed wszystkimi b), k = 5 np. bbbbbaaaaa. 2.3: k = 990

## 4.1, 4.2 i 4.3

```py
liczby = []
with open("liczby.txt") as file:
    for line in file:
        liczby.append(int(line))

# 4.1
ile = 0
pierwsza = 0
for x in liczby:
    napis = str(x)
    if napis[0] == napis[-1]:
        if ile == 0:
            pierwsza = x
        ile += 1
print("4.1", ile, pierwsza)


# 4.2
def czynniki(x):
    wszystkie = 0
    rozne = 0
    d = 2
    while d * d <= x:
        if x % d == 0:
            rozne += 1
            while x % d == 0:
                x //= d
                wszystkie += 1
        d += 1
    if x > 1:
        wszystkie += 1
        rozne += 1
    return wszystkie, rozne


max_wsz, liczba_wsz = 0, 0
max_roz, liczba_roz = 0, 0
for x in liczby:
    wszystkie, rozne = czynniki(x)
    if wszystkie > max_wsz:
        max_wsz, liczba_wsz = wszystkie, x
    if rozne > max_roz:
        max_roz, liczba_roz = rozne, x
print("4.2", liczba_wsz, max_wsz, liczba_roz, max_roz)

# 4.3 - liczby w pliku są różne, więc wystarczy x | y, y | z i x != y != z
trojki = 0
with open("trojki.txt", "w") as wyjscie:
    for x in liczby:
        for y in liczby:
            if y != x and y % x == 0:
                for z in liczby:
                    if z != y and z % y == 0:
                        trojki += 1
                        print(x, y, z, file=wyjscie)

# dobre piątki - liczymy rekurencyjnie łańcuchy podzielności długości 5
def lancuchy(ostatnia, dlugosc):
    if dlugosc == 5:
        return 1
    wynik = 0
    for nastepna in liczby:
        if nastepna != ostatnia and nastepna % ostatnia == 0:
            wynik += lancuchy(nastepna, dlugosc + 1)
    return wynik


piatki = 0
for u in liczby:
    piatki += lancuchy(u, 1)
print("4.3", trojki, piatki)
```
