```py
def czy_pierwsza(n):
    if n < 2:
        return False
    i = 2
    while i * i <= n:
        if n % i == 0:
            return False

    return True


def suma_cyfr(n):

        n //= 10
```

```cpp
#include <fstream>
#include <print>
#include <string>
#include <utility>
#include <vector>

bool czy_pierwsza(int n) {
    if (n < 2) return false;
    for (int i = 2; i * i <= n; i++)
        if (n % i == 0) return false;
    return true;
}
```

```java
// Java 25
import java.io.PrintWriter;
import java.util.*;

public class Zadanie {
    static boolean czyPierwsza(int n) {
        if (n < 2) return false;
        for (int i = 2; i * i <= n; i++)
            if (n % i == 0) return false;
        return true;
    }

    static int sumaCyfr(int n) {
        int suma = 0;
        while (n > 0) {
            suma += n % 10;
            n /= 10;
        }
        return suma;
    }

    static boolean superBPierwsza(int n) {
        // Integer.bitCount - liczba jedynek w zapisie binarnym
        return czyPierwsza(n) && czyPierwsza(sumaCyfr(n)) && czyPierwsza(Integer.bitCount(n));
    }

    public static void main(String[] args) throws Exception {
        int[][] przedzialy = {{2, 1000}, {100, 10000}, {1000, 100000}};
        for (int nr = 0; nr < przedzialy.length; nr++) {
            int a = przedzialy[nr][0], b = przedzialy[nr][1];
            var liczby = new ArrayList<Integer>();
            for (int n = a; n <= b; n++)
                if (superBPierwsza(n)) liczby.add(n);
            try (var wyjscie = new PrintWriter((nr + 1) + ".txt")) {
                for (int n : liczby)
                    wyjscie.println(n);
            }
            System.out.println((nr + 1) + " " + liczby.size());
        }
    }
}
```

> **Treść zadania:** Na podstawie pliku `telefony.txt` utwórz zestawienie: w pierwszej kolumnie numery, z których dzwoniono co najmniej 2 razy, w drugiej — liczba połączeń z danego numeru (zapisz w `zad_6.txt`). ```py

```py
with open("telefony.txt") as file:
    for linia in file:
        numer = linia.strip()
```

```cpp

#include <algorithm>
#include <fstream>
#include <map>
#include <print>
#include <string>
#include <vector>

int main() {
    std::map<std::string, int> polaczenia;
    std::vector<std::string> numery; // w kolejności pierwszego wystąpienia
    std::ifstream plik("telefony.txt");
    std::string numer;
    while (plik >> numer) {
        if (polaczenia[numer] == 0) numery.push_back(numer);
        polaczenia[numer]++;
    }

    // sortowanie stabilne malejąco po liczbie połączeń (jak sorted w Pythonie)
    std::stable_sort(numery.begin(), numery.end(), [&](const std::string& a, const std::string& b) {
        return polaczenia[a] > polaczenia[b];
    });
    for (const auto& n : numery)
        if (polaczenia[n] >= 2) std::println("{}\t{}", n, polaczenia[n]);
```
