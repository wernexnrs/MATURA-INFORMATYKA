## 2a i 2c

```py
def suma_cyfr(k):
    s = 0
    while k > 0:
        s += k % 10
        k //= 10
    return s


def osiagalna(n):
    # z podpunktu b): dla n czterocyfrowego s(k) <= 36, więc sprawdzamy k = n-1, ..., n-36
    for i in range(1, 37):
        k = n - i
        if k > 0 and k + suma_cyfr(k) == n:
            return k
    return "NIE"


# a)
for n in [505, 20, 28, 31]:
    k = osiagalna(n)
    if k == "NIE":
        print(n, "-", "-", "NIE")
    else:
        print(n, k, suma_cyfr(k), "TAK")

# c)
print(osiagalna(2012), osiagalna(1000))
```

**Odpowiedź:** a) n = 28: k = 23, s(k) = 5, TAK; n = 31: NIE; b) suma cyfr liczby co najwyżej czterocyfrowej nie przekracza 4·9 = 36

def szyfruj(tekst, klucz):
    wynik = ""
    for i in range(len(tekst)):
        numer = ord(klucz[i % len(klucz)]) - 64  # numer litery klucza (A = 1)
        kod = ord(tekst[i]) + numer
        if kod > 90:
            kod -= 26
        wynik += chr(kod)
    return wynik

## 6e

*Zadanie na Excel/Access — rozwiązanie w Pythonie.*

```py
from decimal import Decimal, ROUND_HALF_UP

powiaty = {}
with open("tablice.txt") as file:
    file.readline()
    for line in file:
        ozn, powiat, siedziba, typ = line.strip().split("\t")
        powiaty[ozn] = powiat

firmy = {}
with open("nip_firm.txt") as file:
    file.readline()
    for line in file:
        nip, firma = line.strip().split("\t")
        firmy[nip] = firma

uslugi = []
with open("uslugi.txt") as file:
    file.readline()
    for line in file:
        nip, ozn, nr, rodzaj, rata = line.strip().split("\t")
        uslugi.append((nip, ozn, int(rata)))

konin = set()
for nip, ozn, rata in uslugi:
    if powiaty[ozn] == "Konin":
        konin.add(nip)

srednie = []
for nip in konin:
    raty = [rata for n, ozn, rata in uslugi if n == nip]
    srednia = Decimal(sum(raty)) / len(raty)
    # zaokrąglenie do grosza "od połowy w górę", jak w arkuszu (2590,125 -> 2590,13)
    srednie.append((srednia.quantize(Decimal("0.01"), ROUND_HALF_UP), firmy[nip]))

for srednia, firma in sorted(srednie):
    print(firma, srednia)
```

**Odpowiedź:** BARTEX 2325,64 zł; WIET-IND 2556,43 zł; REFAMUR 2590,13 zł
