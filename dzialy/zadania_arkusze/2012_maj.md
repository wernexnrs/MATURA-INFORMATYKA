## 2a i 2c

```py
def suma_cyfr(k):
    s = 0
    while k > 0:
        s += k % 10
        k //= 10
    return s


def osiagalna(n):
    # z podpunktu b): dla n czterocyfrowego s(k) <= 36, więc sprawdzamy k = n-1, ..., n-36
    for i in range(1, 37):
        k = n - i
        if k > 0 and k + suma_cyfr(k) == n:
            return k
    return "NIE"


# a)
for n in [505, 20, 28, 31]:
    k = osiagalna(n)
    if k == "NIE":
        print(n, "-", "-", "NIE")
    else:
        print(n, k, suma_cyfr(k), "TAK")

# c)
print(osiagalna(2012), osiagalna(1000))
```

**Odpowiedź:** a) n = 28: k = 23, s(k) = 5, TAK; n = 31: NIE; b) suma cyfr liczby co najwyżej czterocyfrowej nie przekracza 4·9 = 36

def szyfruj(tekst, klucz):
    wynik = ""
    for i in range(len(tekst)):
        numer = ord(klucz[i % len(klucz)]) - 64  # numer litery klucza (A = 1)
        kod = ord(tekst[i]) + numer
        if kod > 90:
            kod -= 26
        wynik += chr(kod)
    return wynik


print(szyfruj("LATO", "WODA"), szyfruj("MARTA", "TOR"))  # IPXP GPJNP

with open("tj.txt") as file:

## 5d

*Zadanie na arkusz kalkulacyjny — rozwiązanie w Pythonie.*

```py
trojkat = [[1]]
for w in range(1, 30):
    poprzedni = trojkat[-1]
    wiersz = [1]
    for j in range(1, w):
        wiersz.append(poprzedni[j - 1] + poprzedni[j])
    wiersz.append(1)
    trojkat.append(wiersz)

# X - liczba podzielna przez 3, spacja - pozostałe (trójkąt prostokątny)
with open("wynik5d.txt", "w") as wynik:
    for wiersz in trojkat:
        linia = ""
        for liczba in wiersz:
            if liczba % 3 == 0:
                linia += "X"
            else:
                linia += " "
        wynik.write(linia + "\n")
        print(linia)
```

**Odpowiedź:** obraz w pliku wynik5d.txt – pierwsze X w wierszu 4 („ XX ”), wzór podobny do trójkąta Sierpińskiego
