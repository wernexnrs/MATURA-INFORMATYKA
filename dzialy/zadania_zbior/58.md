```py
def wczytaj(nazwa, podstawa):
    zegary = []
    temperatury = []
    with open(nazwa) as file:
        for line in file:
            zegar, temperatura = line.split()
            zegary.append(int(zegar, podstawa))
            temperatury.append(int(temperatura, podstawa))  # int() obsługuje znak "-"
    return zegary, temperatury


def na_dwojkowy(x):
    if x < 0:
        return "-" + bin(-x)[2:]
    return bin(x)[2:]


stacje = [("dane_systemy1.txt", 2), ("dane_systemy2.txt", 4), ("dane_systemy3.txt", 8)]

for i in range(3):
    zegary, temperatury = wczytaj(stacje[i][0], stacje[i][1])
    print("S" + str(i + 1), na_dwojkowy(min(temperatury)))
```
**Odpowiedź:** S1: -1011, S2: -1001100, S3: -1001011

```py
def wczytaj(nazwa, podstawa):
    zegary = []
    temperatury = []
    with open(nazwa) as file:
        for line in file:
            zegar, temperatura = line.split()
            zegary.append(int(zegar, podstawa))
            temperatury.append(int(temperatura, podstawa))
    return zegary, temperatury


z1, t1 = wczytaj("dane_systemy1.txt", 2)
z2, t2 = wczytaj("dane_systemy2.txt", 4)
z3, t3 = wczytaj("dane_systemy3.txt", 8)

max1, max2, max3 = t1[0], t2[0], t3[0]
dni = 1  # pierwszy pomiar jest zawsze rekordem

for i in range(1, len(t1)):
    rekord = False
    if t1[i] > max1:
        max1 = t1[i]
        rekord = True
    if t2[i] > max2:
        max2 = t2[i]
        rekord = True
    if t3[i] > max3:
        max3 = t3[i]
        rekord = True
    if rekord:
        dni += 1

print(dni)
```
**Odpowiedź:** 42

```py
temperatury = []

with open("dane_systemy1.txt") as file:
    for line in file:
        zegar, temperatura = line.split()
        temperatury.append(int(temperatura, 2))

najwiekszy = 0
for i in range(len(temperatury)):
    for j in range(i + 1, len(temperatury)):
        r = (temperatury[i] - temperatury[j]) ** 2
        skok = (r + (j - i) - 1) // (j - i)  # zaokrąglenie w górę
        if skok > najwiekszy:
            najwiekszy = skok

print(najwiekszy)
```
**Odpowiedź:** 25
