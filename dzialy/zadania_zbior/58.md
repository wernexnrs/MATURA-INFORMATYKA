## 58.1

> **Treść zadania:** Trzy stacje pogodowe S1, S2, S3 codziennie w południe zapisują stan zegara (liczbę godzin od uruchomienia stacji) i temperaturę. Stacja S1 zapisuje wartości w systemie dwójkowym, S2 — czwórkowym, S3 — ósemkowym; temperatury ujemne poprzedza znak „-”. Pliki `dane_systemy1.txt`, `dane_systemy2.txt`, `dane_systemy3.txt` zawierają po 1095 kolejnych pomiarów: w każdym wierszu stan zegara i temperatura, oddzielone spacją (np. zegar 36 i temperatura -7 to w S1 `100100 -111`, w S2 `210 -13`, w S3 `44 -7`). Odpowiedzi zapisz w pliku `wyniki_systemy.txt`, każdą w osobnym wierszu, poprzedzoną numerem zadania.
>
> Dla każdej stacji podaj najniższą zarejestrowaną temperaturę. Wszystkie trzy wyniki zapisz w systemie dwójkowym.

```py
def wczytaj(nazwa, podstawa):
    zegary = []
    temperatury = []
    with open(nazwa) as file:
        for line in file:
            zegar, temperatura = line.split()
            zegary.append(int(zegar, podstawa))
            temperatury.append(int(temperatura, podstawa))  # int() obsługuje znak "-"
    return zegary, temperatury


def na_dwojkowy(x):
    if x < 0:
        return "-" + bin(-x)[2:]
    return bin(x)[2:]


stacje = [("dane_systemy1.txt", 2), ("dane_systemy2.txt", 4), ("dane_systemy3.txt", 8)]

for i in range(3):
    zegary, temperatury = wczytaj(stacje[i][0], stacje[i][1])
    print("S" + str(i + 1), na_dwojkowy(min(temperatury)))
```

```cpp
// C++23
#include <algorithm>
#include <format>
#include <fstream>
#include <print>
#include <string>
#include <utility>

// std::stoi(napis, nullptr, podstawa) działa jak int(napis, podstawa) w Pythonie (obsługuje znak "-")
void wczytaj(const std::string& nazwa, int podstawa, std::vector<int>& zegary, std::vector<int>& temperatury) {
    std::ifstream plik(nazwa);
    std::string zegar, temperatura;
    while (plik >> zegar >> temperatura) {
        zegary.push_back(std::stoi(zegar, nullptr, podstawa));
        temperatury.push_back(std::stoi(temperatura, nullptr, podstawa));
    }
}

std::string na_dwojkowy(int x) {
    if (x < 0) {
        return "-" + na_dwojkowy(-x);
    }
    return std::format("{:b}", x);
}

int main() {
    std::vector<std::pair<std::string, int>> stacje = {{"dane_systemy1.txt", 2}, {"dane_systemy2.txt", 4}, {"dane_systemy3.txt", 8}};

    for (int i = 0; i < 3; i++) {
        std::vector<int> zegary, temperatury;
        wczytaj(stacje[i].first, stacje[i].second, zegary, temperatury);
        std::println("S{} {}", i + 1, na_dwojkowy(std::ranges::min(temperatury)));
    }
}
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Zadanie {
    // Integer.parseInt(napis, podstawa) działa jak int(napis, podstawa) w Pythonie (obsługuje znak "-")
    static void wczytaj(String nazwa, int podstawa, List<Integer> zegary, List<Integer> temperatury) throws Exception {
        for (String linia : Files.readAllLines(Path.of(nazwa))) {
            if (linia.isBlank()) {
                continue;
            }
            String[] t = linia.trim().split("\\s+");
            zegary.add(Integer.parseInt(t[0], podstawa));
            temperatury.add(Integer.parseInt(t[1], podstawa));
        }
    }

    static String naDwojkowy(int x) {
        if (x < 0) {
            return "-" + Integer.toBinaryString(-x);
        }
        return Integer.toBinaryString(x);
    }

    public static void main(String[] args) throws Exception {
        String[] nazwy = {"dane_systemy1.txt", "dane_systemy2.txt", "dane_systemy3.txt"};
        int[] podstawy = {2, 4, 8};

        for (int i = 0; i < 3; i++) {
            List<Integer> zegary = new ArrayList<>();
            List<Integer> temperatury = new ArrayList<>();
            wczytaj(nazwy[i], podstawy[i], zegary, temperatury);
            System.out.println("S" + (i + 1) + " " + naDwojkowy(Collections.min(temperatury)));
        }
    }
}
```

## 58.4

> **Treść zadania:** Dane jak w zadaniu 58.1; dotyczy tylko stacji S1 (kolejne temperatury t1, t2, t3, …).
>
> Skokiem temperatury między i-tym a j-tym pomiarem nazywamy zaokrąglenie w górę do liczby całkowitej wartości (t_i − t_j)² / |i − j|. Podaj w systemie dziesiętnym największy skok temperatury w stacji S1.
>
> Przykład: dla temperatur 3, 5, 4, 7 skoki wynoszą 4, 1, 6, 1, 2, 9 — największy to 9.

```py
temperatury = []

with open("dane_systemy1.txt") as file:
    for line in file:
        zegar, temperatura = line.split()
        temperatury.append(int(temperatura, 2))

najwiekszy = 0
for i in range(len(temperatury)):
    for j in range(i + 1, len(temperatury)):
        r = (temperatury[i] - temperatury[j]) ** 2
        skok = (r + (j - i) - 1) // (j - i)  # zaokrąglenie w górę
        if skok > najwiekszy:
            najwiekszy = skok

print(najwiekszy)
```

```cpp
// C++23
#include <fstream>
#include <print>
#include <string>
#include <vector>

int main() {
    std::vector<long long> temperatury;

    std::ifstream plik("dane_systemy1.txt");
    std::string zegar, temperatura;
    while (plik >> zegar >> temperatura) {
        temperatury.push_back(std::stoi(temperatura, nullptr, 2));
    }

    long long najwiekszy = 0;
    for (size_t i = 0; i < temperatury.size(); i++) {
        for (size_t j = i + 1; j < temperatury.size(); j++) {
            long long r = (temperatury[i] - temperatury[j]) * (temperatury[i] - temperatury[j]);
            long long odstep = j - i;
            long long skok = (r + odstep - 1) / odstep;  // zaokrąglenie w górę
            if (skok > najwiekszy) {
                najwiekszy = skok;
            }
        }
    }

    std::println("{}", najwiekszy);
}
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    public static void main(String[] args) throws Exception {
        List<Long> temperatury = new ArrayList<>();

        for (String linia : Files.readAllLines(Path.of("dane_systemy1.txt"))) {
            if (linia.isBlank()) {
                continue;
            }
            String[] t = linia.trim().split("\\s+");
            temperatury.add(Long.parseLong(t[1], 2));
        }

        long najwiekszy = 0;
        for (int i = 0; i < temperatury.size(); i++) {
            for (int j = i + 1; j < temperatury.size(); j++) {
                long roznica = temperatury.get(i) - temperatury.get(j);
                long r = roznica * roznica;
                long odstep = j - i;
                long skok = (r + odstep - 1) / odstep;  // zaokrąglenie w górę
                if (skok > najwiekszy) {
                    najwiekszy = skok;
                }
            }
        }

        System.out.println(najwiekszy);
    }
}
```
**Odpowiedź:** 25
