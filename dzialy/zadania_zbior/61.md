## 61.1

```py
ciagi = []

with open("ciagi.txt") as file:
    wiersze = file.read().split("\n")
for i in range(1, len(wiersze), 2):  # co drugi wiersz zawiera wyrazy ciągu
    ciagi.append(list(map(int, wiersze[i].split())))

count = 0
najwieksza_roznica = 0
for ciag in ciagi:
    r = ciag[1] - ciag[0]
    arytmetyczny = True
    for j in range(1, len(ciag)):
        if ciag[j] - ciag[j - 1] != r:
            arytmetyczny = False
            break
    if arytmetyczny:
        count += 1
        najwieksza_roznica = max(najwieksza_roznica, r)

print(count)
print(najwieksza_roznica)
```
**Odpowiedź:** 44 ciągi arytmetyczne, największa różnica: 246849

## 61.2

```py
def czy_szescian(x):
    a = round(x ** (1 / 3))
    return a ** 3 == x


ciagi = []

with open("ciagi.txt") as file:
    wiersze = file.read().split("\n")
for i in range(1, len(wiersze), 2):
    ciagi.append(list(map(int, wiersze[i].split())))

for ciag in ciagi:
    najwiekszy = 0
    for liczba in ciag:
        if czy_szescian(liczba) and liczba > najwiekszy:
            najwiekszy = liczba
    if najwiekszy > 0:
        print(najwiekszy)
```
**Odpowiedź:** 1, 27, 551368, 1000000, 250047, 1000, 35937, 531441

## 61.3

```py
ciagi = []

with open("bledne.txt") as file:
    wiersze = file.read().split("\n")
for i in range(1, len(wiersze), 2):
    ciagi.append(list(map(int, wiersze[i].split())))

for ciag in ciagi:
    # różnica ciągu to najczęściej występująca różnica kolejnych wyrazów
    roznice = {}
    for j in range(1, len(ciag)):
        d = ciag[j] - ciag[j - 1]
        roznice[d] = roznice.get(d, 0) + 1
    r = max(roznice, key=roznice.get)

    # pierwszy wyraz: najczęstsza wartość ciag[j] - j * r
    pierwsze = {}
    for j in range(len(ciag)):
        a = ciag[j] - j * r
        pierwsze[a] = pierwsze.get(a, 0) + 1
    a1 = max(pierwsze, key=pierwsze.get)

    for j in range(len(ciag)):
        if ciag[j] != a1 + j * r:
            print(ciag[j])
```
