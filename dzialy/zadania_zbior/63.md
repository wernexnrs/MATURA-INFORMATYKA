## 63.3

```py
def liczba_czynnikow(n):  # liczba czynników pierwszych (z powtórzeniami)
    ile = 0
    i = 2
    while i * i <= n:
        while n % i == 0:
            ile += 1
            n //= i
        i += 1
    if n > 1:
        ile += 1
    return ile


polpierwsze = []

with open("ciagi.txt") as file:
    for line in file:
        liczba = int(line.strip(), 2)
        if liczba_czynnikow(liczba) == 2:
            polpierwsze.append(liczba)

print(len(polpierwsze))
print("min:", min(polpierwsze), "max:", max(polpierwsze))
```
**Odpowiedź:** 259 liczb półpierwszych, min = 6, max = 248667
