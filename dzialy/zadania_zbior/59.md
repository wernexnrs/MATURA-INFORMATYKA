## 59.1

```py
def rozne_czynniki(n):
    czynniki = set()
    i = 2
    while i * i <= n:
        while n % i == 0:
            czynniki.add(i)
            n //= i
        i += 1
    if n > 1:
        czynniki.add(n)
    return czynniki


count = 0

with open("liczby.txt") as file:
    for line in file:
        czynniki = rozne_czynniki(int(line))
        if len(czynniki) == 3 and 2 not in czynniki:
            count += 1

print(count)
```
**Odpowiedź:** 114

```py
def iloczyn_cyfr(x):
    wynik = 1
    for cyfra in str(x):
        wynik *= int(cyfra)
    return wynik


def moc(x):
    k = 0
    while x > 9:
        x = iloczyn_cyfr(x)
        k += 1
    return k


ile = [0] * 9
moc_1 = []

with open("liczby.txt") as file:
    for line in file:
        liczba = int(line)
        m = moc(liczba)
        ile[m] += 1
        if m == 1:
            moc_1.append(liczba)

for m in range(1, 9):
    print(m, ile[m])
print("min:", min(moc_1), "max:", max(moc_1))
```
