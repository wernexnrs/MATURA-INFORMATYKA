## 60.2

```py
def dzielniki(n):
    male = []
    duze = []
    i = 1
    while i * i <= n:
        if n % i == 0:
            male.append(i)
            if i != n // i:
                duze.append(n // i)
        i += 1
    return male + duze[::-1]  # rosnąco


with open("liczby.txt") as file:
    for line in file:
        liczba = int(line)
        lista = dzielniki(liczba)
        if len(lista) == 18:
            print(liczba)
            print(*lista)
```

## 60.3

```py
def nwd(a, b):
    while b > 0:
        a, b = b, a % b
    return a


liczby = []

with open("liczby.txt") as file:
    for line in file:
        liczby.append(int(line))

najwieksza = 0
for i in range(len(liczby)):
    wzglednie_pierwsza = True
    for j in range(len(liczby)):
        if i != j and nwd(liczby[i], liczby[j]) > 1:
            wzglednie_pierwsza = False
            break
    if wzglednie_pierwsza and liczby[i] > najwieksza:
        najwieksza = liczby[i]

print(najwieksza)
```
**Odpowiedź:** 988027
