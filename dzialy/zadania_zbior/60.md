## 60.1

> **Treść zadania:** Plik `liczby.txt` zawiera 200 różnych liczb całkowitych z przedziału [2, 1 000 000], po jednej w wierszu. Odpowiedzi zapisz w pliku `wyniki.txt`.
>
> Policz, ile liczb w pliku jest mniejszych niż 1000, oraz podaj dwie takie liczby, które występują w pliku jako ostatnie (można założyć, że są co najmniej dwie).

```py
male = []

with open("liczby.txt") as file:
    for line in file:
        liczba = int(line)
        if liczba < 1000:
            male.append(liczba)

print(len(male))
print(male[-2], male[-1])
```

## 60.2

> **Treść zadania:** Dane jak w zadaniu 60.1 (200 liczb w pliku `liczby.txt`).
>
> Znajdź liczby, które mają dokładnie 18 dzielników naturalnych (licząc 1 i samą liczbę). Dla każdej z nich wypisz jej wartość oraz rosnąco posortowaną listę wszystkich jej dzielników.

```py
def dzielniki(n):
    male = []
    duze = []
    i = 1
    while i * i <= n:
        if n % i == 0:
            male.append(i)
            if i != n // i:
                duze.append(n // i)
        i += 1
    return male + duze[::-1]  # rosnąco


with open("liczby.txt") as file:
    for line in file:
        liczba = int(line)
        lista = dzielniki(liczba)
        if len(lista) == 18:
            print(liczba)
            print(*lista)
```

```cpp
// C++23
#include <fstream>
#include <print>
#include <vector>

std::vector<int> dzielniki(int n) {
    std::vector<int> male;
    std::vector<int> duze;
    int i = 1;
    while (i * i <= n) {
        if (n % i == 0) {
            male.push_back(i);
            if (i != n / i) {
                duze.push_back(n / i);
            }
        }
        i += 1;
    }
    male.insert(male.end(), duze.rbegin(), duze.rend());  // rosnąco
    return male;
}

int main() {
    std::ifstream plik("liczby.txt");
    int liczba;
    while (plik >> liczba) {
        std::vector<int> lista = dzielniki(liczba);
        if (lista.size() == 18) {
            std::println("{}", liczba);
            for (size_t i = 0; i < lista.size(); i++) {
                std::print("{}{}", i > 0 ? " " : "", lista[i]);
            }
            std::println();
        }
    }
}
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class Zadanie {
    static List<Integer> dzielniki(int n) {
        List<Integer> male = new ArrayList<>();
        List<Integer> duze = new ArrayList<>();
        int i = 1;
        while (i * i <= n) {
            if (n % i == 0) {
                male.add(i);
                if (i != n / i) {
                    duze.add(n / i);
                }
            }
            i += 1;
        }
        Collections.reverse(duze);
        male.addAll(duze);  // rosnąco
        return male;
    }

    public static void main(String[] args) throws Exception {
        for (String linia : Files.readAllLines(Path.of("liczby.txt"))) {
            int liczba = Integer.parseInt(linia.trim());
            List<Integer> lista = dzielniki(liczba);
            if (lista.size() == 18) {
                System.out.println(liczba);
                List<String> napisy = new ArrayList<>();
                for (int d : lista) {
                    napisy.add(String.valueOf(d));
                }
                System.out.println(String.join(" ", napisy));
            }
        }
    }
}
```

## 60.3

> **Treść zadania:** Dane jak w zadaniu 60.1.
>
> Znajdź największą liczbę z pliku, która jest względnie pierwsza ze wszystkimi pozostałymi liczbami, tzn. z żadną z nich nie ma wspólnego dzielnika większego niż 1.

```py
def nwd(a, b):
    while b > 0:
        a, b = b, a % b
    return a


liczby = []

with open("liczby.txt") as file:
    for line in file:
        liczby.append(int(line))

najwieksza = 0
for i in range(len(liczby)):
    wzglednie_pierwsza = True
    for j in range(len(liczby)):
        if i != j and nwd(liczby[i], liczby[j]) > 1:
            wzglednie_pierwsza = False
            break
    if wzglednie_pierwsza and liczby[i] > najwieksza:
        najwieksza = liczby[i]

print(najwieksza)
```

```cpp
            if (i != j && nwd(liczby[i], liczby[j]) > 1) {
                wzglednie_pierwsza = false;
                break;
            }
        }
        if (wzglednie_pierwsza && liczby[i] > najwieksza) {
            najwieksza = liczby[i];
        }
    }

    std::println("{}", najwieksza);
}
```

```java
// Java 25
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;

public class Zadanie {
    static int nwd(int a, int b) {
        while (b > 0) {
            int reszta = a % b;
            a = b;
            b = reszta;
        }
        return a;
    }

    public static void main(String[] args) throws Exception {
        List<Integer> liczby = new ArrayList<>();

        for (String linia : Files.readAllLines(Path.of("liczby.txt"))) {
            liczby.add(Integer.parseInt(linia.trim()));
        }

        int najwieksza = 0;
        for (int i = 0; i < liczby.size(); i++) {
            boolean wzgledniePierwsza = true;
            for (int j = 0; j < liczby.size(); j++) {
                if (i != j && nwd(liczby.get(i), liczby.get(j)) > 1) {
                    wzgledniePierwsza = false;
                    break;
                }
            }
            if (wzgledniePierwsza && liczby.get(i) > najwieksza) {
                najwieksza = liczby.get(i);
            }
        }

        System.out.println(najwieksza);
    }
}
```
